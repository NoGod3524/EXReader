cmake_minimum_required(VERSION 3.24)
option(EXR_READER_SELECTOR_BYTES_CANDIDATE "Explicit raw store companion candidate; never default" OFF)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  project(EXReaderSession VERSION 0.2.0 LANGUAGES CXX)
else()
  project(EXReaderSession VERSION 0.1.0 LANGUAGES CXX)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(EXR_SYNC_VERSION 0.1.0)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  set(EXR_SYNC_VERSION 0.2.0)
endif()
find_package(EXReaderSync ${EXR_SYNC_VERSION} EXACT CONFIG REQUIRED)
option(EXR_READER_SESSION_MANAGED "Explicit Linux raw session managed staging" OFF)
if(EXR_READER_SESSION_MANAGED)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT EXR_READER_SELECTOR_BYTES_CANDIDATE)
    message(FATAL_ERROR "Managed session requires the Linux raw family")
  endif()
  find_package(EXReaderWebDAVManaged 0.1.0 EXACT CONFIG REQUIRED)
endif()
find_package(Threads REQUIRED)
get_target_property(_raw_dependency_defs EXReaderSync::reader_sync INTERFACE_COMPILE_DEFINITIONS)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  if(NOT "EXR_READER_SELECTOR_BYTES_CANDIDATE=1" IN_LIST _raw_dependency_defs)
    message(FATAL_ERROR "Exact dependency target belongs to the legacy capability family")
  endif()
else()
  if("EXR_READER_SELECTOR_BYTES_CANDIDATE=1" IN_LIST _raw_dependency_defs)
    message(FATAL_ERROR "Default profile cannot consume a raw candidate dependency target")
  endif()
endif()
add_library(reader_session STATIC src/session.cpp)
target_include_directories(reader_session PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(reader_session PUBLIC EXReaderSync::reader_sync PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(reader_session PRIVATE /W4 /WX)
else()
  target_compile_options(reader_session PRIVATE -Wall -Wextra -Werror)
endif()
include(CMakePackageConfigHelpers)
install(TARGETS reader_session EXPORT EXReaderSessionTargets ARCHIVE DESTINATION lib)
install(FILES include/reader_session.h DESTINATION include)
install(EXPORT EXReaderSessionTargets NAMESPACE EXReaderSession:: DESTINATION lib/cmake/EXReaderSession)
configure_package_config_file(EXReaderSessionConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionConfig.cmake"
  INSTALL_DESTINATION lib/cmake/EXReaderSession)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionConfigVersion.cmake"
  DESTINATION lib/cmake/EXReaderSession)

if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  target_compile_definitions(reader_session PUBLIC EXR_READER_SELECTOR_BYTES_CANDIDATE=1)
endif()

if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  _exr_raw_guard_target(EXReaderSync::reader_sync "${EXReaderSync_DIR}"
    reader_sync.h READER_SYNC_VERSION 0.2.0 TRUE)
endif()

if(EXR_READER_SESSION_MANAGED)
  target_compile_definitions(reader_session PUBLIC EXR_READER_SESSION_MANAGED=1)
  install(FILES include/reader_session_managed.h DESTINATION include)
  configure_package_config_file(EXReaderSessionManagedConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionManagedConfig.cmake INSTALL_DESTINATION lib/cmake/EXReaderSessionManaged)
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionManagedConfigVersion.cmake VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionManagedConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/EXReaderSessionManagedConfigVersion.cmake DESTINATION lib/cmake/EXReaderSessionManaged)
endif()
