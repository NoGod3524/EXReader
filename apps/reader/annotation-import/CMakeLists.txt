cmake_minimum_required(VERSION 3.24)
option(EXR_READER_SELECTOR_BYTES_CANDIDATE "Explicit raw0.2 portable import family; never default" OFF)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  project(EXReaderAnnotationImport VERSION 0.2.0 LANGUAGES C CXX)
  set(EXR_REVISIONS_VERSION 0.4.0)
  set(EXR_READING_STATE_VERSION 0.2.0)
else()
  project(EXReaderAnnotationImport VERSION 0.1.0 LANGUAGES C CXX)
  set(EXR_REVISIONS_VERSION 0.3.0)
endif()
# Standalone and not included by reading-library or host default configuration.
option(READER_ANNOTATION_IMPORT_BUILD "Build optional annotation importer" OFF)
if(NOT READER_ANNOTATION_IMPORT_BUILD)
  return()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(EXReaderRevisions ${EXR_REVISIONS_VERSION} EXACT CONFIG REQUIRED)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  find_package(EXReaderReadingState ${EXR_READING_STATE_VERSION} EXACT CONFIG REQUIRED)
endif()
add_library(reader_annotation_import STATIC src/annotation_import.cpp)
target_include_directories(reader_annotation_import PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(reader_annotation_import PUBLIC EXReaderRevisions::reader_revisions)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  target_link_libraries(reader_annotation_import PUBLIC EXReaderReadingState::reader_reading_state_codec)
  target_compile_definitions(reader_annotation_import PUBLIC EXR_READER_SELECTOR_BYTES_CANDIDATE=1)
endif()
if(MSVC)
  target_compile_options(reader_annotation_import PRIVATE /W4 /WX /utf-8)
  target_compile_definitions(reader_annotation_import PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(reader_annotation_import PRIVATE -Wall -Wextra -Werror)
endif()
include(CMakePackageConfigHelpers)
install(TARGETS reader_annotation_import EXPORT EXReaderAnnotationImportTargets ARCHIVE DESTINATION lib)
install(FILES include/reader_annotation_import.h DESTINATION include)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  install(FILES include/reader_annotation_import_raw3.h DESTINATION include)
endif()
install(EXPORT EXReaderAnnotationImportTargets NAMESPACE EXReaderAnnotationImport:: DESTINATION lib/cmake/EXReaderAnnotationImport)
configure_package_config_file(EXReaderAnnotationImportConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EXReaderAnnotationImportConfig.cmake" INSTALL_DESTINATION lib/cmake/EXReaderAnnotationImport)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EXReaderAnnotationImportConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderAnnotationImportConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EXReaderAnnotationImportConfigVersion.cmake" DESTINATION lib/cmake/EXReaderAnnotationImport)

if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  _exr_public_boundary_guard(EXReaderRevisions::reader_revisions "${EXReaderRevisions_DIR}"
    reader_selector_bytes_v1.h READER_REV_SELECTOR_BYTES_API_VERSION EXR_READER_PUBLIC_SELECTOR_BYTES_V1)
  _exr_public_boundary_guard(EXReaderReadingState::reader_reading_state_codec "${EXReaderReadingState_DIR}"
    reader_reading_state_retained_raw_v1.h READER_RS_RETAINED_RAW_API_VERSION EXR_READER_PUBLIC_RETAINED_RAW_V1)
endif()
