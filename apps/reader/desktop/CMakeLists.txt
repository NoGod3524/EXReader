cmake_minimum_required(VERSION 3.24)
project(EXReaderDesktopProduct LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core library
  PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
set(READER_METADATA_PREFIX "" CACHE PATH "Installed desktop metadata commands and their shared product provider")
if(READER_METADATA_PREFIX)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  list(PREPEND CMAKE_PREFIX_PATH "${READER_METADATA_PREFIX}")
  find_package(EXReaderDesktopMetadata CONFIG REQUIRED)
  # Commands, revisions and both hosts must resolve one installed product
  # catalog/SQLite provider, instead of adding a second private catalog build.
  add_library(reader_catalog ALIAS EXReaderProductCatalog::reader_catalog)
  add_library(reader_catalog_sqlite ALIAS EXReaderProductCatalog::reader_catalog_sqlite)
else()
  add_subdirectory(../store/native native-catalog)
endif()
add_library(reader_catalog_sdk STATIC catalog_sdk.c)
target_include_directories(reader_catalog_sdk PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reader_catalog_sdk PUBLIC reader_catalog EXReader::exr_library)
if(WIN32)
  target_link_libraries(reader_catalog_sdk PRIVATE bcrypt normaliz)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  target_link_libraries(reader_catalog_sdk PRIVATE PkgConfig::GLIB)
endif()
if(MSVC)
  target_compile_options(reader_catalog_sdk PRIVATE /W4 /WX)
else()
  target_compile_options(reader_catalog_sdk PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/native-check" "${CMAKE_CURRENT_BINARY_DIR}/sdk-check")
add_executable(reader-desktop-catalog-check ../../../tests/reader_catalog/desktop_sdk_check.c)
target_link_libraries(reader-desktop-catalog-check PRIVATE reader_catalog_sdk)
target_link_libraries(reader-desktop-catalog-check PRIVATE reader_catalog_sqlite)
if(MSVC)
  target_link_options(reader-desktop-catalog-check PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/sdk-link.map")
else()
  target_link_options(reader-desktop-catalog-check PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/sdk-link.map" "-Wl,--trace-symbol=sqlite3_open")
endif()
add_test(NAME desktop-public-sdk-catalog COMMAND reader-desktop-catalog-check
  "${CMAKE_CURRENT_BINARY_DIR}/sdk-check" "${EXR_READER_FIXTURES}")
add_executable(reader-native-catalog-check ../../../tests/reader_catalog/native_catalog_check.c)
target_link_libraries(reader-native-catalog-check PRIVATE reader_catalog reader_catalog_sqlite Threads::Threads)
add_test(NAME desktop-native-catalog COMMAND reader-native-catalog-check
  "${CMAKE_CURRENT_BINARY_DIR}/native-check" "${CMAKE_CURRENT_SOURCE_DIR}/../store/001_catalog.sql")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shelf-check")
add_executable(reader-native-shelf-check ../../../tests/reader_catalog/native_shelf_check.c)
target_link_libraries(reader-native-shelf-check PRIVATE reader_catalog reader_catalog_sqlite)
add_test(NAME desktop-bidirectional-shelf COMMAND reader-native-shelf-check "${CMAKE_CURRENT_BINARY_DIR}/shelf-check")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/organization-check")
add_executable(reader-native-organization-check ../../../tests/reader_catalog/native_organization_check.c)
target_link_libraries(reader-native-organization-check PRIVATE reader_catalog reader_catalog_sqlite Threads::Threads)
add_test(NAME desktop-organization-views COMMAND reader-native-organization-check "${CMAKE_CURRENT_BINARY_DIR}/organization-check")
option(EXR_READER_BUILD_GTK "Build the real GTK product host over the installed SDK" OFF)
option(EXR_READER_GTK_ACCOUNT "Enable native GTK secure credentials and asynchronous account scans" OFF)
set(READER_ACCOUNT_PREFIX "" CACHE PATH "Installed optional session, credentials and sideload modules")
if(WIN32)
  add_executable(EXReaderProduct WIN32 ../../../examples/reader.c)
  target_compile_definitions(EXReaderProduct PRIVATE EXR_READER_PRODUCT_CATALOG=1
    "EXR_READER_WINDOWS_PRODUCT_HEADER=\"${CMAKE_CURRENT_SOURCE_DIR}/windows_catalog.inc\"")
  target_link_libraries(EXReaderProduct PRIVATE reader_catalog_sdk EXReader::exr_core
    comdlg32 gdi32 shell32 user32)
  if(READER_METADATA_PREFIX)
    target_compile_definitions(EXReaderProduct PRIVATE EXR_READER_DESKTOP_METADATA=1)
    target_link_libraries(EXReaderProduct PRIVATE EXReaderDesktopMetadata::reader_desktop_metadata)
  endif()
  if(MSVC)
    target_link_options(EXReaderProduct PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/windows-host.map")
  endif()
  option(EXR_READER_WINDOWS_SHELF_TESTS "Build isolated worker/cache ownership regression, not installed" OFF)
  if(EXR_READER_WINDOWS_SHELF_TESTS)
    add_executable(reader-windows-shelf-seed ../../../tests/windows_host/seed_shelf.c)
    target_link_libraries(reader-windows-shelf-seed PRIVATE EXReader::exr_library)
    add_executable(EXReaderShelfReview WIN32 ../../../examples/reader.c)
    target_compile_definitions(EXReaderShelfReview PRIVATE EXR_READER_PRODUCT_CATALOG=1
      EXR_HOST_WORKER_REVIEW=1 EXR_HOST_SHELF_REVIEW=1
      "EXR_READER_WINDOWS_PRODUCT_HEADER=\"${CMAKE_CURRENT_SOURCE_DIR}/windows_catalog.inc\""
      "EXR_HOST_WORKER_REVIEW_HEADER=\"${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/windows_host/worker_review/hooks.h\"")
    target_link_libraries(EXReaderShelfReview PRIVATE reader_catalog_sdk EXReader::exr_core comdlg32 gdi32 shell32 user32)
    if(READER_METADATA_PREFIX)
      target_compile_definitions(EXReaderShelfReview PRIVATE EXR_READER_DESKTOP_METADATA=1)
      target_link_libraries(EXReaderShelfReview PRIVATE EXReaderDesktopMetadata::reader_desktop_metadata)
    endif()
  endif()
endif()
if(EXR_READER_BUILD_GTK)
  if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "The GTK product host requires Linux")
  endif()
  add_subdirectory(../../../examples/linux gtk-host)
  target_compile_definitions(exreader-linux-reading PRIVATE EXR_READER_PRODUCT_CATALOG=1)
  target_compile_definitions(exreader-linux PRIVATE EXR_READER_PRODUCT_CATALOG=1)
  target_link_libraries(exreader-linux-reading PRIVATE reader_catalog_sdk)
  if(READER_METADATA_PREFIX)
    target_compile_definitions(exreader-linux-reading PRIVATE EXR_READER_DESKTOP_METADATA=1)
    target_compile_definitions(exreader-linux PRIVATE EXR_READER_DESKTOP_METADATA=1)
    target_link_libraries(exreader-linux-reading PRIVATE
      EXReaderDesktopMetadata::reader_desktop_metadata PkgConfig::GLIB)
  endif()
  if(EXR_READER_GTK_ACCOUNT)
    if(NOT READER_METADATA_PREFIX OR NOT READER_ACCOUNT_PREFIX)
      message(FATAL_ERROR "GTK account integration requires the installed metadata and account module prefixes")
    endif()
    list(PREPEND CMAKE_PREFIX_PATH "${READER_ACCOUNT_PREFIX}")
    find_package(EXReaderCredentials 0.1.0 EXACT CONFIG REQUIRED)
    find_package(EXReaderSession 0.1.0 EXACT CONFIG REQUIRED)
    find_package(EXReaderSideload CONFIG REQUIRED)
    target_compile_definitions(exreader-linux-reading PRIVATE EXR_READER_ACCOUNT=1)
    target_compile_definitions(exreader-linux PRIVATE EXR_READER_ACCOUNT=1)
    target_link_libraries(exreader-linux-reading PRIVATE EXReaderCredentials::exr_credentials
      EXReaderSideload::exr_sideload EXReaderWebDAV::exr_webdav)
    target_link_libraries(exreader-linux PRIVATE EXReaderSession::reader_session EXReaderCredentials::exr_credentials)
  endif()
  add_executable(reader-gtk-shelf-check ../../../tests/linux_host/shelf_backend.c)
  target_compile_options(reader-gtk-shelf-check PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(reader-gtk-shelf-check PRIVATE exreader-linux-reading reader_catalog_sqlite)
endif()
