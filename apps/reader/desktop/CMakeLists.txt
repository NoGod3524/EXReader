cmake_minimum_required(VERSION 3.24)
project(EXReaderDesktopProduct LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core library
  PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
add_subdirectory(../store/native native-catalog)
add_library(reader_catalog_sdk STATIC catalog_sdk.c)
target_include_directories(reader_catalog_sdk PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reader_catalog_sdk PUBLIC reader_catalog EXReader::exr_library)
if(WIN32)
  target_link_libraries(reader_catalog_sdk PRIVATE bcrypt normaliz)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  target_link_libraries(reader_catalog_sdk PRIVATE PkgConfig::GLIB)
endif()
if(MSVC)
  target_compile_options(reader_catalog_sdk PRIVATE /W4 /WX)
else()
  target_compile_options(reader_catalog_sdk PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/native-check" "${CMAKE_CURRENT_BINARY_DIR}/sdk-check")
add_executable(reader-desktop-catalog-check ../../../tests/reader_catalog/desktop_sdk_check.c)
target_link_libraries(reader-desktop-catalog-check PRIVATE reader_catalog_sdk)
target_link_libraries(reader-desktop-catalog-check PRIVATE reader_catalog_sqlite)
if(MSVC)
  target_link_options(reader-desktop-catalog-check PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/sdk-link.map")
else()
  target_link_options(reader-desktop-catalog-check PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/sdk-link.map" "-Wl,--trace-symbol=sqlite3_open")
endif()
add_test(NAME desktop-public-sdk-catalog COMMAND reader-desktop-catalog-check
  "${CMAKE_CURRENT_BINARY_DIR}/sdk-check" "${EXR_READER_FIXTURES}")
add_executable(reader-native-catalog-check ../../../tests/reader_catalog/native_catalog_check.c)
target_link_libraries(reader-native-catalog-check PRIVATE reader_catalog reader_catalog_sqlite Threads::Threads)
add_test(NAME desktop-native-catalog COMMAND reader-native-catalog-check
  "${CMAKE_CURRENT_BINARY_DIR}/native-check" "${CMAKE_CURRENT_SOURCE_DIR}/../store/001_catalog.sql")
option(EXR_READER_BUILD_GTK "Build the real GTK product host over the installed SDK" OFF)
if(WIN32)
  add_executable(EXReaderProduct WIN32 ../../../examples/reader.c)
  target_compile_definitions(EXReaderProduct PRIVATE EXR_READER_PRODUCT_CATALOG=1
    "EXR_READER_WINDOWS_PRODUCT_HEADER=\"${CMAKE_CURRENT_SOURCE_DIR}/windows_catalog.inc\"")
  target_link_libraries(EXReaderProduct PRIVATE reader_catalog_sdk EXReader::exr_core
    comdlg32 gdi32 shell32 user32)
  if(MSVC)
    target_link_options(EXReaderProduct PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/windows-host.map")
  endif()
endif()
if(EXR_READER_BUILD_GTK)
  if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "The GTK product host requires Linux")
  endif()
  add_subdirectory(../../../examples/linux gtk-host)
  target_compile_definitions(exreader-linux-reading PRIVATE EXR_READER_PRODUCT_CATALOG=1)
  target_link_libraries(exreader-linux-reading PRIVATE reader_catalog_sdk)
endif()
