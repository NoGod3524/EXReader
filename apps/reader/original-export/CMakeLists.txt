cmake_minimum_required(VERSION 3.24)
project(EXReaderOriginalExport VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
# Independent opt-in package; never added to SDK/default reading configuration.
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS library)
find_package(Threads REQUIRED)
add_library(reader_original_export STATIC src/original_export.cpp)
target_include_directories(reader_original_export PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(reader_original_export PUBLIC EXReader::exr_library PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(reader_original_export PRIVATE /W4 /WX)
  target_compile_definitions(reader_original_export PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(reader_original_export PRIVATE -Wall -Wextra -Werror)
endif()
include(CMakePackageConfigHelpers)
install(TARGETS reader_original_export EXPORT EXReaderOriginalExportTargets ARCHIVE DESTINATION lib)
install(FILES include/reader_original_export.h DESTINATION include)
install(EXPORT EXReaderOriginalExportTargets NAMESPACE EXReaderOriginalExport:: DESTINATION lib/cmake/EXReaderOriginalExport)
configure_package_config_file(EXReaderOriginalExportConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EXReaderOriginalExportConfig.cmake" INSTALL_DESTINATION lib/cmake/EXReaderOriginalExport)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EXReaderOriginalExportConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderOriginalExportConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EXReaderOriginalExportConfigVersion.cmake" DESTINATION lib/cmake/EXReaderOriginalExport)
