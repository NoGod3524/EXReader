cmake_minimum_required(VERSION 3.24)
option(EXR_READER_SELECTOR_BYTES_CANDIDATE "Explicit raw store companion candidate; never default" OFF)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  project(EXReaderSync VERSION 0.2.0 LANGUAGES CXX)
else()
  project(EXReaderSync VERSION 0.1.0 LANGUAGES CXX)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(EXR_REVISIONS_VERSION 0.3.0)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  set(EXR_REVISIONS_VERSION 0.4.0)
endif()
find_package(EXReaderRevisions ${EXR_REVISIONS_VERSION} EXACT CONFIG REQUIRED)
find_package(EXReaderWebDAV 0.2.0 EXACT CONFIG REQUIRED)
get_target_property(_raw_dependency_defs EXReaderRevisions::reader_revisions INTERFACE_COMPILE_DEFINITIONS)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  if(NOT "EXR_READER_SELECTOR_BYTES_CANDIDATE=1" IN_LIST _raw_dependency_defs)
    message(FATAL_ERROR "Exact dependency target belongs to the legacy capability family")
  endif()
else()
  if("EXR_READER_SELECTOR_BYTES_CANDIDATE=1" IN_LIST _raw_dependency_defs)
    message(FATAL_ERROR "Default profile cannot consume a raw candidate dependency target")
  endif()
endif()
add_library(reader_sync STATIC src/sync.cpp)
target_include_directories(reader_sync PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(reader_sync PUBLIC EXReaderRevisions::reader_revisions EXReaderWebDAV::exr_webdav)
if(WIN32)
  target_link_libraries(reader_sync PRIVATE bcrypt)
elseif(ANDROID)
  # One explicit installed hash owner, independent of producer prefix overrides.
  set(_sync_hash_prefix "${READER_ORIGINAL_HASH_PREFIX}")
  set(_sync_hash_env "$ENV{EXREADER_ORIGINAL_HASH_PREFIX}")
  if(NOT _sync_hash_prefix)
    set(_sync_hash_prefix "${_sync_hash_env}")
  endif()
  if(NOT IS_ABSOLUTE "${_sync_hash_prefix}" OR NOT IS_DIRECTORY "${_sync_hash_prefix}")
    message(FATAL_ERROR "Sync requires an explicit installed OPDSHash prefix")
  endif()
  file(REAL_PATH "${_sync_hash_prefix}" _sync_hash_real)
  if(NOT "${_sync_hash_prefix}" STREQUAL "${_sync_hash_real}")
    message(FATAL_ERROR "Sync hash prefix must be canonical")
  endif()
  if(_sync_hash_env)
    file(REAL_PATH "${_sync_hash_env}" _sync_hash_env_real)
    if(NOT IS_ABSOLUTE "${_sync_hash_env}" OR NOT "${_sync_hash_env}" STREQUAL "${_sync_hash_env_real}" OR
        NOT "${_sync_hash_env_real}" STREQUAL "${_sync_hash_real}")
      message(FATAL_ERROR "Conflicting Sync hash prefix selectors")
    endif()
  endif()
  set(_sync_hash_config "${_sync_hash_real}/lib/cmake/EXReaderOPDSHash")
  if(DEFINED EXReaderOPDSHash_DIR AND NOT "${EXReaderOPDSHash_DIR}" STREQUAL "${_sync_hash_config}")
    message(FATAL_ERROR "Foreign cached Sync OPDSHash package directory")
  endif()
  find_package(EXReaderOPDSHash 0.1.0 EXACT CONFIG REQUIRED
    PATHS "${_sync_hash_config}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  if(NOT "${EXReaderOPDSHash_DIR}" STREQUAL "${_sync_hash_config}")
    message(FATAL_ERROR "Sync hash discovery escaped its selected prefix")
  endif()
  exreader_opds_hash_validate("${_sync_hash_real}")
  target_link_libraries(reader_sync PRIVATE EXReaderOPDSHash::opds_sha256_stream)
elseif(APPLE)
  # CommonCrypto is provided by libSystem; no separate crypto package.
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(reader_sync PRIVATE OpenSSL::Crypto)
endif()
if(MSVC)
  target_compile_options(reader_sync PRIVATE /W4 /WX)
  target_compile_definitions(reader_sync PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(reader_sync PRIVATE -Wall -Wextra -Werror)
endif()
include(CMakePackageConfigHelpers)
install(TARGETS reader_sync EXPORT EXReaderSyncTargets ARCHIVE DESTINATION lib)
install(FILES include/reader_sync.h DESTINATION include)
install(EXPORT EXReaderSyncTargets NAMESPACE EXReaderSync:: DESTINATION lib/cmake/EXReaderSync)
configure_package_config_file(EXReaderSyncConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncConfig.cmake" INSTALL_DESTINATION lib/cmake/EXReaderSync)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncConfigVersion.cmake" DESTINATION lib/cmake/EXReaderSync)

if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  target_compile_definitions(reader_sync PUBLIC EXR_READER_SELECTOR_BYTES_CANDIDATE=1)
endif()

if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  _exr_raw_guard_target(EXReaderRevisions::reader_revisions "${EXReaderRevisions_DIR}"
    reader_revisions.h READER_REVISIONS_VERSION 0.4.0 TRUE)
endif()
