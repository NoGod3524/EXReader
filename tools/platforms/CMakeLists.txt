cmake_minimum_required(VERSION 3.24)
project(EXReaderRuntimeProbe LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(NOT EXR_LIBRARY)
  message(FATAL_ERROR "Set EXR_LIBRARY to the actual shared library to exercise")
endif()
add_executable(platform-probe probe.c)
target_include_directories(platform-probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_link_libraries(platform-probe PRIVATE "${EXR_LIBRARY}")
if(NOT MSVC)
  target_compile_options(platform-probe PRIVATE -Wall -Wextra -Werror)
endif()
