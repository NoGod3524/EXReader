cmake_minimum_required(VERSION 3.24)
project(EXReaderPlatformAcceptance LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(EXReader CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_language(OBJC)
  if(NOT EXR_SMOKE_INPUT)
    message(FATAL_ERROR "Set EXR_SMOKE_INPUT to prepare_smoke.py output")
  endif()
  add_executable(exreader-acceptance MACOSX_BUNDLE acceptance.c ios_main.m "${EXR_SMOKE_INPUT}/smoke.epub" "${EXR_SMOKE_INPUT}/font.otf")
  set_source_files_properties("${EXR_SMOKE_INPUT}/smoke.epub" "${EXR_SMOKE_INPUT}/font.otf" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(exreader-acceptance PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.exreader.acceptance MACOSX_BUNDLE_BUNDLE_VERSION 1 MACOSX_BUNDLE_SHORT_VERSION_STRING 1.0)
  target_compile_definitions(exreader-acceptance PRIVATE EXR_ACCEPTANCE_IOS)
  if(EXR_ACCEPTANCE_ARCHIVE)
    target_link_libraries(exreader-acceptance PRIVATE "${EXR_ACCEPTANCE_ARCHIVE}")
  endif()
  target_link_libraries(exreader-acceptance PRIVATE EXReader::exr_core "-framework UIKit")
else()
  add_executable(exreader-acceptance acceptance.c)
  target_link_libraries(exreader-acceptance PRIVATE EXReader::exreader)
endif()
if(NOT MSVC)
  target_compile_options(exreader-acceptance PRIVATE -Wall -Wextra -Werror)
endif()
