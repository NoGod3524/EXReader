cmake_minimum_required(VERSION 3.24)
project(exreader_consumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SDK_ROOT "" CACHE PATH "Installed SDK containing include/ and lib/")
set(MODE "shared" CACHE STRING "static, shared, runtime, or headers-only")
if(NOT EXISTS "${SDK_ROOT}/include/exr/version.h")
  message(FATAL_ERROR "SDK_ROOT must contain include/exr/version.h")
endif()
add_executable(consumer main.c)
target_include_directories(consumer PRIVATE "${SDK_ROOT}/include")
target_compile_definitions(consumer PRIVATE CONSUMER_MODE="${MODE}")
set_target_properties(consumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(consumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
if(MSVC)
  set_property(TARGET consumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(consumer PRIVATE /W4 /utf-8)
  target_compile_definitions(consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(MODE STREQUAL "static")
  # The release has component archives, not an aggregate libexreader.a.
  # Link their published dependency closure without importing the DLL's bin/ path.
  set(CMAKE_FIND_LIBRARY_SUFFIXES "${CMAKE_STATIC_LIBRARY_SUFFIX}")
  foreach(component render layout formats doc text graphics base)
    find_library(SDK_${component} NAMES exr_${component}
      PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    list(APPEND archives "${SDK_${component}}")
  endforeach()
  file(GLOB vendors "${SDK_ROOT}/lib/exreader/${CMAKE_STATIC_LIBRARY_PREFIX}*${CMAKE_STATIC_LIBRARY_SUFFIX}")
  list(APPEND archives ${vendors})
  add_library(exreader INTERFACE)
  target_compile_definitions(exreader INTERFACE EXR_STATIC)
  if(CMAKE_C_LINK_GROUP_USING_RESCAN_SUPPORTED OR CMAKE_LINK_GROUP_USING_RESCAN_SUPPORTED)
    list(JOIN archives "," archive_group)
    target_link_libraries(exreader INTERFACE "$<LINK_GROUP:RESCAN,${archive_group}>")
  else()
    target_link_libraries(exreader INTERFACE ${archives})
  endif()
  if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(exreader INTERFACE Threads::Threads m ${CMAKE_DL_LIBS})
  elseif(MINGW)
    target_link_libraries(exreader INTERFACE m)
  endif()
  target_link_libraries(consumer PRIVATE exreader)
elseif(MODE STREQUAL "shared")
  add_library(exreader SHARED IMPORTED)
  if(WIN32)
    find_file(SDK_DLL NAMES libexreader.dll exreader.dll
      PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    find_library(SDK_IMPORT_LIBRARY NAMES exreader libexreader.dll
      PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    set_target_properties(exreader PROPERTIES IMPORTED_LOCATION "${SDK_DLL}"
      IMPORTED_IMPLIB "${SDK_IMPORT_LIBRARY}")
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E
      copy_if_different "${SDK_DLL}" "$<TARGET_FILE_DIR:consumer>")
  else()
    # Only the shared suffix is eligible; never silently select a static archive.
    set(CMAKE_FIND_LIBRARY_SUFFIXES "${CMAKE_SHARED_LIBRARY_SUFFIX}")
    find_library(SDK_SHARED_LIBRARY NAMES exreader
      PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    set_target_properties(exreader PROPERTIES IMPORTED_LOCATION "${SDK_SHARED_LIBRARY}")
  endif()
  target_link_libraries(consumer PRIVATE exreader)
elseif(MODE STREQUAL "runtime")
  target_compile_definitions(consumer PRIVATE CONSUMER_RUNTIME EXR_STATIC)
  target_link_libraries(consumer PRIVATE ${CMAKE_DL_LIBS})
elseif(MODE STREQUAL "headers-only")
  # Deliberately reach the linker: public headers alone cannot provide the SDK.
  target_link_directories(consumer PRIVATE "${SDK_ROOT}/lib")
  target_link_libraries(consumer PRIVATE exreader)
else()
  message(FATAL_ERROR "Unknown MODE=${MODE}")
endif()
