cmake_minimum_required(VERSION 3.24)
project(EXReaderE1 LANGUAGES C)
# Keep SDK feature defaults. Only omit unrelated executables and tests.
set(EXR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXR_BUILD_JNI OFF CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(EXR_PERF_SDK_DLL)
  # Reuse a byte-identical SDK for measurement-only experiments.
  add_library(exreader UNKNOWN IMPORTED)
  set_target_properties(exreader PROPERTIES IMPORTED_LOCATION "${EXR_PERF_SDK_DLL}"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/../../include;${EXR_PERF_SDK_INCLUDE}")
  set(EXR_BUILD_SHARED_COMPAT ON)
else()
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." sdk)
endif()

# Reuse the actual R3 operation plan, hashes and oversized-page fallback.
# Adapt only the OS counters, process affinity and selectable viewport.
file(READ "${CMAKE_CURRENT_LIST_DIR}/../reading_real_benchmark.c" probe)
set(old "config.page.width=800;config.page.height=1000;config.page.margin=32;config.page.font_size=24;")
string(FIND "${probe}" "${old}" matched)
if(matched LESS 0)
  message(FATAL_ERROR "R3 viewport adapter no longer matches; review the probe")
endif()
string(REPLACE "${old}" "config.page.width=e1_dimension(\"E1_WIDTH\",800);config.page.height=e1_dimension(\"E1_HEIGHT\",1000);config.page.margin=e1_dimension(\"E1_MARGIN\",32);config.page.font_size=24;" probe "${probe}")
string(REPLACE "#include <sys/resource.h>" "#include \"e1_platform.h\"" probe "${probe}")
string(REPLACE "int main(int argc,char **argv) {" "int main(int argc,char **argv) {\n  if(!e1_pin()) return 3;" probe "${probe}")

# Measurement-only paired probe; keep the existing E1 operation plan untouched.
set(ratio "${probe}")
string(REPLACE "int main(int argc,char **argv) {" "#include \"ratio_document.inc\"\nint main(int argc,char **argv) {" ratio "${ratio}")
string(REPLACE "static int text_sample(" "static exr_document *ratio_document;\nstatic int text_sample(" ratio "${ratio}")
string(REPLACE "  return exr_pager_read_text(pager,item->text_start,item->text_end,text_sink,item,NULL,error)==EXR_OK;" "  if(ratio_document) {\n    size_t bytes=0;const char *text=exr_document_text(ratio_document,&bytes);\n    if(!text || item->text_start>item->text_end || item->text_end>bytes) return 0;\n    item->text_bytes=item->text_end-item->text_start;\n    hash_bytes(&item->text_hash,text+item->text_start,item->text_bytes);return 1;\n  }\n  return exr_pager_read_text(pager,item->text_start,item->text_end,text_sink,item,NULL,error)==EXR_OK;" ratio "${ratio}")
string(REPLACE "if(!is_pdf && strcmp(argv[1],\"epub\"))" "if(!is_pdf && strcmp(argv[1],\"epub\") && strcmp(argv[1],\"mobi\"))" ratio "${ratio}")
string(REPLACE "} else {\n    if(exr_publication_open" "} else if(!strcmp(argv[1],\"mobi\")) {\n    if(exr_document_open(argv[2],&ratio_document,&error)!=EXR_OK || !ratio_document_pager(ratio_document,font,&pager,&error)) goto finish;\n    part_count=1;current_part=part;\n  } else {\n    if(exr_publication_open" ratio "${ratio}")
string(REPLACE "exr_publication_free(publication);exr_fixed_free" "exr_publication_free(publication);exr_document_free(ratio_document);exr_fixed_free" ratio "${ratio}")
string(REPLACE "#define SEQUENTIAL 32" "#define SEQUENTIAL 16" ratio "${ratio}")
string(REPLACE "#define RANDOM 16" "#define RANDOM 0" ratio "${ratio}")
string(REPLACE "double layout_ms,render_ms,total_ms;" "double layout_ms,render_ms,total_ms,reference_ms;" ratio "${ratio}")
string(REPLACE "static double now_ms(void)" "double perf_reference_ms(void);\nstatic double now_ms(void)" ratio "${ratio}")
string(REPLACE "double at=now_ms();int ok;" "item->reference_ms=perf_reference_ms();\n  double at=now_ms();int ok;" ratio "${ratio}")
string(REPLACE "exr_bitmap bitmap={0};double at=now_ms();int ok;" "exr_bitmap bitmap={0};item->reference_ms=perf_reference_ms();double at=now_ms();int ok;" ratio "${ratio}")
string(REPLACE "\\\"total_ms\\\":%.6f," "\\\"total_ms\\\":%.6f,\\\"target_ms\\\":%.6f,\\\"reference_ms\\\":%.6f,\\\"ratio\\\":%.9f," ratio "${ratio}")
string(REPLACE "s->layout_ms,s->render_ms,s->total_ms,s->width" "s->layout_ms,s->render_ms,s->total_ms,s->total_ms,s->reference_ms,s->total_ms/s->reference_ms,s->width" ratio "${ratio}")
# Warm up the frozen reference in the same process before opening the font/book.
string(REPLACE "if(!e1_pin()) return 3;" "if(!e1_pin()) return 3;\n  (void)perf_reference_ms();" ratio "${ratio}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ratio_probe.c" "${ratio}")
add_executable(ratio-probe "${CMAKE_CURRENT_BINARY_DIR}/ratio_probe.c" reference.c)
target_include_directories(ratio-probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(ratio-probe PRIVATE exreader)
add_executable(perf-reference reference_main.c reference.c)
if(MINGW)
  # Pin executable identity across builds: PE wall-clock timestamps add no workload.
  target_link_options(perf-reference PRIVATE "-Wl,--no-insert-timestamp")
endif()
if(WIN32)
  target_link_libraries(ratio-probe PRIVATE psapi)
  target_link_libraries(perf-reference PRIVATE psapi)
  if(EXR_BUILD_SHARED_COMPAT)
    add_custom_command(TARGET ratio-probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:exreader>" "$<TARGET_FILE_DIR:ratio-probe>")
  endif()
else()
  target_link_libraries(ratio-probe PRIVATE m)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/load_probe.cmake")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/e1_probe.c" "${probe}")
add_executable(e1-probe "${CMAKE_CURRENT_BINARY_DIR}/e1_probe.c")
target_include_directories(e1-probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(e1-probe PRIVATE exreader)
if(WIN32)
  target_link_libraries(e1-probe PRIVATE psapi)
  if(EXR_BUILD_SHARED_COMPAT)
    add_custom_command(TARGET e1-probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:exreader>" "$<TARGET_FILE_DIR:e1-probe>")
  endif()
else()
  target_link_libraries(e1-probe PRIVATE m)
endif()
