cmake_minimum_required(VERSION 3.24)
project(EXReaderWindowsRestartReader LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(EXReader 3.42.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_library(exreader-migration-windows-restart-reader STATIC src/lm_win_restart_reader.c)
target_include_directories(exreader-migration-windows-restart-reader PUBLIC include "${RECORD_PREFIX}/include/exreader-migration-candidate" "${LEASE_PREFIX}/include/exreader-migration-candidate")
target_compile_definitions(exreader-migration-windows-restart-reader PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(exreader-migration-windows-restart-reader PRIVATE /W4 /WX /utf-8)
target_link_libraries(exreader-migration-windows-restart-reader PUBLIC "${LEASE_PREFIX}/lib/exreader-migration-windows-lease.lib" "${RECORD_PREFIX}/lib/exreader-migration-windows-record.lib" EXReader::exr_library advapi32 psapi)
set_target_properties(exreader-migration-windows-restart-reader PROPERTIES C_STANDARD 11)
install(TARGETS exreader-migration-windows-restart-reader ARCHIVE DESTINATION lib)
install(FILES include/lm_win_restart_reader.h DESTINATION include/exreader-migration-candidate)
