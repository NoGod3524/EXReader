function(pc02_owner target)
  add_executable(${target} src/pc02_main.c src/pc02_io.c src/pc02_journal.c)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${WH3_PREFIX}/include/exreader-coordinator-cas" "${WH_PREFIX}/include/exreader-migration-coordinator-hardening")
  target_link_libraries(${target} PRIVATE pc02-transform "${WH3_PREFIX}/lib/libexreader-coordinator-cas.a" EXReaderC02::Admission)
endfunction()
