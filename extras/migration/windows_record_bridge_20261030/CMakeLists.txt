cmake_minimum_required(VERSION 3.24)
project(EXReaderWindowsRecordBridge LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(EXReader 3.42.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_library(exreader-migration-windows-record-bridge STATIC src/lm_win_record_bridge.c)
target_include_directories(exreader-migration-windows-record-bridge PUBLIC include "${RECORD_PREFIX}/include/exreader-migration-candidate" "${PUBLISH_PREFIX}/include/exreader-migration-candidate")
target_compile_definitions(exreader-migration-windows-record-bridge PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(exreader-migration-windows-record-bridge PRIVATE /W4 /WX /utf-8)
target_link_libraries(exreader-migration-windows-record-bridge PUBLIC "${PUBLISH_PREFIX}/lib/exreader-migration-windows-publish.lib" "${RECORD_PREFIX}/lib/exreader-migration-windows-record.lib" EXReader::exr_library advapi32 psapi)
set_target_properties(exreader-migration-windows-record-bridge PROPERTIES C_STANDARD 11)
install(TARGETS exreader-migration-windows-record-bridge ARCHIVE DESTINATION lib)
install(FILES include/lm_win_record_bridge.h DESTINATION include/exreader-migration-candidate)
