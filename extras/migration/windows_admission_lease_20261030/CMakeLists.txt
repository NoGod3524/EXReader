cmake_minimum_required(VERSION 3.24)
project(EXReaderWindowsAdmissionLease LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(exreader-migration-windows-lease STATIC src/lm_win_lease.c)
target_include_directories(exreader-migration-windows-lease PUBLIC include)
target_compile_definitions(exreader-migration-windows-lease PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(exreader-migration-windows-lease PRIVATE /W4 /WX /utf-8)
set_target_properties(exreader-migration-windows-lease PROPERTIES C_STANDARD 11)
install(TARGETS exreader-migration-windows-lease ARCHIVE DESTINATION lib)
install(FILES include/lm_win_lease.h DESTINATION include/exreader-migration-candidate)
