cmake_minimum_required(VERSION 3.24)
project(EXReaderDesktopDisplay VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(exr_desktop_display STATIC src/common.c)
add_library(EXReaderDesktopDisplay::display ALIAS exr_desktop_display)
set_target_properties(exr_desktop_display PROPERTIES EXPORT_NAME display POSITION_INDEPENDENT_CODE ON)
target_include_directories(exr_desktop_display PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(WIN32)
  target_sources(exr_desktop_display PRIVATE src/windows.c)
  target_link_libraries(exr_desktop_display PUBLIC user32 comctl32)
  target_compile_options(exr_desktop_display PRIVATE /W4 /utf-8)
else()
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Only Win32 and Linux GTK3 are supported")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  target_sources(exr_desktop_display PRIVATE src/linux.c)
  target_link_libraries(exr_desktop_display PUBLIC PkgConfig::GTK3)
  target_compile_options(exr_desktop_display PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(EXReaderDesktopDisplayConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/EXReaderDesktopDisplayConfig.cmake INSTALL_DESTINATION lib/cmake/EXReaderDesktopDisplay)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EXReaderDesktopDisplayConfigVersion.cmake VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
install(TARGETS exr_desktop_display EXPORT EXReaderDesktopDisplayTargets ARCHIVE DESTINATION lib)
install(FILES include/exr_desktop_display.h DESTINATION include)
install(EXPORT EXReaderDesktopDisplayTargets NAMESPACE EXReaderDesktopDisplay:: DESTINATION lib/cmake/EXReaderDesktopDisplay)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EXReaderDesktopDisplayConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/EXReaderDesktopDisplayConfigVersion.cmake DESTINATION lib/cmake/EXReaderDesktopDisplay)
set(EXR_DISPLAY_SOURCE_SHA "unversioned" CACHE STRING "Exact source commit compiled into the module")
target_compile_definitions(exr_desktop_display PRIVATE EXR_DISPLAY_SOURCE_SHA="${EXR_DISPLAY_SOURCE_SHA}")
