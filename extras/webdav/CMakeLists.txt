cmake_minimum_required(VERSION 3.24)
project(EXReaderWebDAV VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
add_library(exr_webdav STATIC src/webdav.cpp)
add_library(EXReaderWebDAV::exr_webdav ALIAS exr_webdav)
target_include_directories(exr_webdav PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(WIN32)
  target_link_libraries(exr_webdav PRIVATE winhttp xmllite shlwapi)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
  find_package(LibXml2 REQUIRED)
  target_link_libraries(exr_webdav PRIVATE PkgConfig::CURL LibXml2::LibXml2)
endif()
if(MSVC)
  target_compile_options(exr_webdav PRIVATE /W4 /WX)
  target_compile_definitions(exr_webdav PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(exr_webdav PRIVATE -Wall -Wextra -Werror)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS exr_webdav EXPORT EXReaderWebDAVTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/exr_webdav.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT EXReaderWebDAVTargets NAMESPACE EXReaderWebDAV:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderWebDAV)
configure_package_config_file(EXReaderWebDAVConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderWebDAV)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderWebDAV)
