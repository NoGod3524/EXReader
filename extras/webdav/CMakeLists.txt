cmake_minimum_required(VERSION 3.24)
project(EXReaderWebDAV VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(_apple_native_default OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(_apple_native_default ON)
endif()
option(EXR_WEBDAV_APPLE_URLSESSION "Use Apple Foundation HTTPS transport" ${_apple_native_default})
if(EXR_WEBDAV_APPLE_URLSESSION AND NOT APPLE)
  message(FATAL_ERROR "Foundation HTTPS requires an Apple target")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
add_library(exr_webdav STATIC src/webdav.cpp)
add_library(EXReaderWebDAV::exr_webdav ALIAS exr_webdav)
target_include_directories(exr_webdav PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(WIN32)
  target_link_libraries(exr_webdav PRIVATE winhttp xmllite shlwapi)
elseif(ANDROID)
  if(DEFINED ANDROID_PLATFORM_LEVEL AND ANDROID_PLATFORM_LEVEL LESS 26)
    message(FATAL_ERROR "Android WebDAV requires API 26 or newer")
  endif()
  set_target_properties(exr_webdav PROPERTIES POSITION_INDEPENDENT_CODE ON)
  option(EXR_WEBDAV_ANDROID_TESTING "Enable Android acceptance counters (Debug only)" OFF)
  if(EXR_WEBDAV_ANDROID_TESTING)
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
      message(FATAL_ERROR "Android acceptance hooks are forbidden outside Debug")
    endif()
    target_compile_definitions(exr_webdav PRIVATE EXR_WEBDAV_ANDROID_TESTING=1)
  endif()
elseif(EXR_WEBDAV_APPLE_URLSESSION)
  enable_language(OBJCXX)
  target_sources(exr_webdav PRIVATE src/apple_transport.mm)
  set_target_properties(exr_webdav PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
  target_compile_options(exr_webdav PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  target_compile_definitions(exr_webdav PRIVATE EXR_WEBDAV_APPLE_URLSESSION=1)
  find_package(LibXml2 REQUIRED)
  target_link_libraries(exr_webdav PRIVATE LibXml2::LibXml2 "-framework Foundation" "-framework CoreFoundation")
else()
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
  find_package(LibXml2 REQUIRED)
  target_link_libraries(exr_webdav PRIVATE PkgConfig::CURL LibXml2::LibXml2 Threads::Threads)
endif()
if(MSVC)
  target_compile_options(exr_webdav PRIVATE /W4 /WX)
  target_compile_definitions(exr_webdav PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(exr_webdav PRIVATE -Wall -Wextra -Werror)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS exr_webdav EXPORT EXReaderWebDAVTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/exr_webdav.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(ANDROID)
  install(FILES include/exr_webdav_android.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
install(FILES ../../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/exreader-webdav RENAME LICENSE.txt)
install(EXPORT EXReaderWebDAVTargets NAMESPACE EXReaderWebDAV:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderWebDAV)
configure_package_config_file(EXReaderWebDAVConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderWebDAV)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderWebDAV)
