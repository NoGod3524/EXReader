cmake_minimum_required(VERSION 3.24)
project(SharedSQLiteConsumer LANGUAGES C)
set(SDK_CONTRACT_VERSION "3.41.0" CACHE STRING "Exact SDK line under contract verification")
find_package(EXReader ${SDK_CONTRACT_VERSION} EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderProductCatalog CONFIG REQUIRED)
find_package(EXReaderSQLiteProvider 1.0.0 EXACT CONFIG REQUIRED)
if(NOT EXReader_USE_SHARED_SQLITE_PROVIDER OR NOT EXReaderProductCatalog_USE_SHARED_SQLITE_PROVIDER)
  message(FATAL_ERROR "The integration consumer requires both SDK and ProductCatalog shared-provider profiles")
endif()
add_executable(shared-sqlite-consumer main.c)
target_link_libraries(shared-sqlite-consumer PRIVATE EXReader::exreader
  EXReaderProductCatalog::reader_catalog EXReaderSQLiteProvider::sqlite3)
if(MSVC)
  target_link_options(shared-sqlite-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/consumer.map")
elseif(APPLE)
  target_link_options(shared-sqlite-consumer PRIVATE "LINKER:-map,${CMAKE_CURRENT_BINARY_DIR}/consumer.map")
else()
  target_link_options(shared-sqlite-consumer PRIVATE "LINKER:-Map,${CMAKE_CURRENT_BINARY_DIR}/consumer.map")
endif()
enable_testing()
add_test(NAME shared-sqlite-sdk-product COMMAND shared-sqlite-consumer)
set_tests_properties(shared-sqlite-sdk-product PROPERTIES LABELS "FTS5;SDKContract")
