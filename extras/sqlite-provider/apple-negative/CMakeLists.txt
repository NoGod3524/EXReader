cmake_minimum_required(VERSION 3.24)
project(SharedSQLiteAppleNegatives LANGUAGES C)
if(NOT APPLE)
  message(FATAL_ERROR "These are real Mach-O negative controls; build them on Apple")
endif()
find_package(EXReader CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderSQLiteProvider 1.0.0 EXACT CONFIG REQUIRED)
if(NOT SQLITE_SOURCE_DIR OR NOT PROVIDER_PREFIX)
  message(FATAL_ERROR "Supply the verified SQLite source directory and installed provider prefix")
endif()
file(READ "${PROVIDER_PREFIX}/share/exreader-sqlite/provider.json" provider_json)
string(JSON expected_source GET "${provider_json}" source_sha256)
file(SHA256 "${SQLITE_SOURCE_DIR}/sqlite3.c" actual_source)
if(NOT actual_source STREQUAL expected_source)
  message(FATAL_ERROR "Negative control source differs from the actual provider")
endif()
# Compile a complete second implementation once, then embed it in both mutants.
add_library(embedded_sqlite OBJECT "${SQLITE_SOURCE_DIR}/sqlite3.c")
set_target_properties(embedded_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(embedded_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0
  SQLITE_DEFAULT_FOREIGN_KEYS=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_MAX_LENGTH=2147483647 SQLITE_ENABLE_FTS5)
add_executable(embedded_main main.c $<TARGET_OBJECTS:embedded_sqlite>)
target_link_libraries(embedded_main PRIVATE EXReader::exreader EXReaderSQLiteProvider::sqlite3)
target_link_options(embedded_main PRIVATE "LINKER:-map,${CMAKE_CURRENT_BINARY_DIR}/embedded-main.map")
add_library(embedded_sdk SHARED $<TARGET_OBJECTS:embedded_sqlite>)
set_target_properties(embedded_sdk PROPERTIES OUTPUT_NAME exreader VERSION ${EXReader_VERSION}
  SOVERSION 1 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bad-sdk" INSTALL_NAME_DIR "@rpath"
  BUILD_WITH_INSTALL_NAME_DIR ON LINKER_LANGUAGE C)
target_link_libraries(embedded_sdk PRIVATE m)
