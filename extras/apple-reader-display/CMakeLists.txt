cmake_minimum_required(VERSION 3.21)
project(AppleReaderDisplay VERSION 1.0.0 LANGUAGES OBJC)
if(NOT APPLE)
  message(FATAL_ERROR "AppleReaderDisplay requires an Apple SDK")
endif()
set(ADR_SOURCE_SHA "unversioned" CACHE STRING "Source revision embedded in the library")
add_library(AppleReaderDisplay STATIC src/ADRDisplayLease.m)
target_include_directories(AppleReaderDisplay PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(AppleReaderDisplay PRIVATE -fobjc-arc -Wall -Wextra)
target_compile_definitions(AppleReaderDisplay PRIVATE ADR_SOURCE_SHA="${ADR_SOURCE_SHA}")
target_link_libraries(AppleReaderDisplay PUBLIC "-framework Foundation")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_link_libraries(AppleReaderDisplay PUBLIC "-framework UIKit")
else()
  target_link_libraries(AppleReaderDisplay PUBLIC "-framework IOKit")
endif()
install(TARGETS AppleReaderDisplay EXPORT AppleReaderDisplayTargets ARCHIVE DESTINATION lib)
install(FILES include/ADRDisplayLease.h DESTINATION include)
install(EXPORT AppleReaderDisplayTargets NAMESPACE AppleReaderDisplay:: DESTINATION lib/cmake/AppleReaderDisplay)
install(FILES AppleReaderDisplayConfig.cmake DESTINATION lib/cmake/AppleReaderDisplay)
