# Independent optional module; the default reading-core/library build never adds it.
cmake_minimum_required(VERSION 3.22)
project(EXReaderCredentials VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(public_headers include/exr_credentials.h)
if(WIN32)
  set(backend src/windows.c)
elseif(APPLE)
  set(backend src/apple.c)
elseif(ANDROID)
  set(backend src/android.c)
  list(APPEND public_headers include/exr_credentials_android.h)
else()
  set(backend src/linux.c)
endif()
add_library(exr_credentials STATIC src/store.c ${backend})
add_library(EXReaderCredentials::exr_credentials ALIAS exr_credentials)
target_include_directories(exr_credentials PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(WIN32)
  target_link_libraries(exr_credentials PRIVATE advapi32)
elseif(APPLE)
  target_link_libraries(exr_credentials PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(NOT ANDROID)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBSECRET REQUIRED IMPORTED_TARGET libsecret-1>=0.19)
  target_link_libraries(exr_credentials PRIVATE PkgConfig::LIBSECRET)
endif()
if(MSVC)
  target_compile_options(exr_credentials PRIVATE /W4 /WX)
else()
  target_compile_options(exr_credentials PRIVATE -Wall -Wextra -Werror)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS exr_credentials EXPORT EXReaderCredentialsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES ${public_headers} DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ../../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/exreader-credentials RENAME LICENSE.txt)
install(EXPORT EXReaderCredentialsTargets NAMESPACE EXReaderCredentials:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderCredentials)
configure_package_config_file(EXReaderCredentialsConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/EXReaderCredentialsConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderCredentials)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EXReaderCredentialsConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EXReaderCredentialsConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/EXReaderCredentialsConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EXReaderCredentials)
