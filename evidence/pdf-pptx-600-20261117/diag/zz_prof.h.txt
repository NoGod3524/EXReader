/* TEMP profiling - do not commit */
#ifndef ZZ_PROF_H
#define ZZ_PROF_H
#include <windows.h>
#include <stdlib.h>
#include <string.h>
__attribute__((weak)) double zz_t[24]; __attribute__((weak)) long long zz_n[24];
static inline long long zz_now(void){LARGE_INTEGER x;QueryPerformanceCounter(&x);return x.QuadPart;}
static inline void zz_add(int k,long long s){LARGE_INTEGER f;QueryPerformanceFrequency(&f);zz_t[k]+=(double)(zz_now()-s)*1e3/(double)f.QuadPart;zz_n[k]++;}
static inline int zz_skip(const char *w){const char *v=getenv("ZZ_SKIP");return v&&strstr(v,w)!=NULL;}
#endif
