cmake_minimum_required(VERSION 3.24)
project(InstalledCXX98Consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 98)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderProductCatalog CONFIG REQUIRED)
find_package(EXReaderSQLiteProvider 1.0.0 EXACT CONFIG REQUIRED)
foreach(t EXReader::exreader EXReaderProductCatalog::reader_original_import EXReaderProductCatalog::reader_catalog EXReaderSQLiteProvider::sqlite3)
 get_target_property(features ${t} INTERFACE_COMPILE_FEATURES)
 message(STATUS "consumer interface ${t}: ${features}")
 if(features MATCHES "cxx_std_17")
  message(FATAL_ERROR "Unexpected public C++17 requirement")
 endif()
endforeach()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE EXReader::exreader EXReaderProductCatalog::reader_original_import EXReaderSQLiteProvider::sqlite3)
