cmake_minimum_required(VERSION 3.24)
project(EXReaderHostExample LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReader 3.35.1 EXACT CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
set(exr_examples construct anchor lazy text font page shaping compat hyphen)
if(TARGET EXReader::exr_pdf)
  list(APPEND exr_examples fixed)
endif()
if(TARGET EXReader::exr_audio)
  list(APPEND exr_examples wave pcm)
endif()
if(TARGET EXReader::exr_textindex)
  list(APPEND exr_examples search)
endif()
foreach(example IN LISTS exr_examples)
  add_executable(exreader-${example}-example ${example}.c)
  target_link_libraries(exreader-${example}-example PRIVATE EXReader::exreader)
endforeach()
target_compile_definitions(exreader-compat-example PRIVATE EXR_EXPECTED_ABI=${EXReader_ABI_VERSION})
if(WIN32)
  foreach(example IN LISTS exr_examples)
    target_link_libraries(exreader-${example}-example PRIVATE shell32)
  endforeach()
  get_target_property(exr_aggregate_type EXReader::exreader TYPE)
  if(exr_aggregate_type STREQUAL "SHARED_LIBRARY")
    find_file(EXR_RUNTIME NAMES exreader.dll libexreader.dll PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
    add_custom_target(exreader-example-runtime
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:exreader-construct-example>"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-construct-example>"
      VERBATIM)
    foreach(example IN LISTS exr_examples)
      add_dependencies(exreader-${example}-example exreader-example-runtime)
    endforeach()
  endif()
endif()
