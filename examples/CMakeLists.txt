cmake_minimum_required(VERSION 3.24)
project(EXReaderHostExample LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
include("${EXR_SDK}/lib/cmake/EXReader/EXReaderConfigVersion.cmake")
find_package(EXReader ${PACKAGE_VERSION} EXACT CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
set(exr_examples construct anchor lazy text font page shaping compat hyphen window bench_batch)
if(WIN32)
  list(APPEND exr_examples reader)
endif()
if(TARGET EXReader::exr_pdf)
  list(APPEND exr_examples fixed)
endif()
if(TARGET EXReader::exr_audio)
  list(APPEND exr_examples wave pcm)
endif()
if(TARGET EXReader::exr_textindex)
  list(APPEND exr_examples search)
endif()
if(WIN32)
  # The 2a shell surface is compiled on every Windows build so it stays buildable
  # on its own; the reader links it only when EXR_READER_WINDOWS_SHELL is ON.
  enable_language(CXX)
  add_library(exreader-shell-d2d STATIC windows/shell_d2d.cpp)
  target_compile_features(exreader-shell-d2d PRIVATE cxx_std_17)
  target_compile_options(exreader-shell-d2d PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>,/W4 /WX,-Wall -Wextra -Werror>)
  target_link_libraries(exreader-shell-d2d PUBLIC d2d1 dwrite advapi32 comctl32)
  set(EXR_READER_WINDOWS_SHELL ON) # The retired native toolbar has no build path.
endif()
foreach(example IN LISTS exr_examples)
  if(example STREQUAL "reader")
    add_executable(exreader-${example}-example WIN32 ${example}.c)
    target_link_libraries(exreader-${example}-example PRIVATE comdlg32 gdi32 uiautomationcore crypt32 winhttp)
    if(EXR_READER_WINDOWS_SHELL)
      target_compile_definitions(exreader-${example}-example PRIVATE EXR_READER_WINDOWS_SHELL=1)
      target_link_libraries(exreader-${example}-example PRIVATE exreader-shell-d2d)
      target_sources(exreader-${example}-example PRIVATE windows/shell_speech.cpp windows/shell_epub.cpp)
      target_link_libraries(exreader-${example}-example PRIVATE ole32 uuid EXReader::vendor_miniz EXReader::vendor_expat)
      if(NOT MSVC)
        target_link_options(exreader-${example}-example PRIVATE -static)
      endif()
    endif()
    if(NOT MSVC)
      target_link_libraries(exreader-${example}-example PRIVATE m)
    endif()
  else()
    add_executable(exreader-${example}-example ${example}.c)
  endif()
  target_link_libraries(exreader-${example}-example PRIVATE EXReader::exreader)
endforeach()
set_target_properties(exreader-bench_batch-example PROPERTIES OUTPUT_NAME bench_batch)

target_compile_definitions(exreader-compat-example PRIVATE EXR_EXPECTED_ABI=${EXReader_ABI_VERSION})
if(WIN32)
  foreach(example IN LISTS exr_examples)
    target_link_libraries(exreader-${example}-example PRIVATE shell32)
  endforeach()
  get_target_property(exr_aggregate_type EXReader::exreader TYPE)
  if(exr_aggregate_type STREQUAL "SHARED_LIBRARY")
    find_file(EXR_RUNTIME NAMES exreader.dll libexreader.dll PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
    add_custom_target(exreader-example-runtime
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:exreader-construct-example>"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-construct-example>"
      VERBATIM)
    foreach(example IN LISTS exr_examples)
      add_dependencies(exreader-${example}-example exreader-example-runtime)
    endforeach()
  endif()
endif()
