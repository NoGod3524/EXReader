cmake_minimum_required(VERSION 3.24)
project(EXReaderHostExample LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReader 0.93 CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
foreach(example construct anchor lazy text font page shaping fixed wave compat search)
  add_executable(exreader-${example}-example ${example}.c)
  target_link_libraries(exreader-${example}-example PRIVATE EXReader::exreader)
endforeach()
target_compile_definitions(exreader-compat-example PRIVATE EXR_EXPECTED_ABI=${EXReader_ABI_VERSION})
if(WIN32)
  target_link_libraries(exreader-search-example PRIVATE shell32)
  target_link_libraries(exreader-shaping-example PRIVATE shell32)
  target_link_libraries(exreader-fixed-example PRIVATE shell32)
  target_link_libraries(exreader-wave-example PRIVATE shell32)
  target_link_libraries(exreader-construct-example PRIVATE shell32)
  target_link_libraries(exreader-lazy-example PRIVATE shell32)
  target_link_libraries(exreader-font-example PRIVATE shell32)
  target_link_libraries(exreader-page-example PRIVATE shell32)
  find_file(EXR_RUNTIME NAMES exreader.dll libexreader.dll PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
  add_custom_target(exreader-example-runtime
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:exreader-construct-example>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-construct-example>"
    VERBATIM)
  foreach(example construct anchor lazy text font page shaping fixed wave compat search)
    add_dependencies(exreader-${example}-example exreader-example-runtime)
  endforeach()
endif()
