cmake_minimum_required(VERSION 3.24)
project(EXReaderHostExample LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_path(EXR_INCLUDE exreader.h PATHS "${EXR_SDK}/include" NO_DEFAULT_PATH REQUIRED)
find_library(EXR_LIBRARY NAMES exreader libexreader PATHS "${EXR_SDK}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(exreader-construct-example construct.c)
add_executable(exreader-anchor-example anchor.c)
add_executable(exreader-lazy-example lazy.c)
add_executable(exreader-text-example text.c)
add_executable(exreader-font-example font.c)
add_executable(exreader-page-example page.c)
add_executable(exreader-shaping-example shaping.c)
target_include_directories(exreader-shaping-example PRIVATE "${EXR_INCLUDE}")
target_link_libraries(exreader-shaping-example PRIVATE "${EXR_LIBRARY}")
target_include_directories(exreader-construct-example PRIVATE "${EXR_INCLUDE}")
target_include_directories(exreader-anchor-example PRIVATE "${EXR_INCLUDE}")
target_include_directories(exreader-lazy-example PRIVATE "${EXR_INCLUDE}")
target_include_directories(exreader-text-example PRIVATE "${EXR_INCLUDE}")
target_include_directories(exreader-font-example PRIVATE "${EXR_INCLUDE}")
target_include_directories(exreader-page-example PRIVATE "${EXR_INCLUDE}")
target_link_libraries(exreader-construct-example PRIVATE "${EXR_LIBRARY}")
target_link_libraries(exreader-anchor-example PRIVATE "${EXR_LIBRARY}")
target_link_libraries(exreader-lazy-example PRIVATE "${EXR_LIBRARY}")
target_link_libraries(exreader-text-example PRIVATE "${EXR_LIBRARY}")
target_link_libraries(exreader-font-example PRIVATE "${EXR_LIBRARY}")
target_link_libraries(exreader-page-example PRIVATE "${EXR_LIBRARY}")
if(WIN32)
  target_link_libraries(exreader-shaping-example PRIVATE shell32)
  target_link_libraries(exreader-construct-example PRIVATE shell32)
  target_link_libraries(exreader-lazy-example PRIVATE shell32)
  target_link_libraries(exreader-font-example PRIVATE shell32)
  target_link_libraries(exreader-page-example PRIVATE shell32)
  find_file(EXR_RUNTIME NAMES exreader.dll libexreader.dll PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
  add_custom_command(TARGET exreader-shaping-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-shaping-example>")
  add_custom_command(TARGET exreader-page-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-page-example>")
  add_custom_command(TARGET exreader-font-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-font-example>")
  add_custom_command(TARGET exreader-text-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-text-example>")
  add_custom_command(TARGET exreader-lazy-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-lazy-example>")
  add_custom_command(TARGET exreader-anchor-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-anchor-example>")
  add_custom_command(TARGET exreader-construct-example POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:exreader-construct-example>")
endif()
