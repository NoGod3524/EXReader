cmake_minimum_required(VERSION 3.22.1)
project(exreader_host C CXX)
option(EXR_HOST_SYNC "Install the optional native account/sync host integration" OFF)
option(EXR_HOST_SEARCH "Install the optional public in-book search host integration" OFF)
option(EXR_HOST_NAVIGATION "Install the optional public navigation host integration" OFF)
option(EXR_HOST_READING "Install optional typed product reading UI integration" OFF)
option(EXR_HOST_PREFERENCES "Install optional device-local reading settings" OFF)
option(EXR_HOST_PAGE_THEME "Explicit SDK 3.41 whole-page theme candidate" OFF)
option(EXR_READER_SELECTOR_BYTES_CANDIDATE "Explicit independent raw3 native host candidate" OFF)
option(EXR_HOST_WAVE3 "Install the reviewed annotation/font/original-export/jump-history/sync-status host integration" OFF)
set(EXR_HOST_SDK_VERSION "3.41.0" CACHE STRING "Exact installed Android host SDK version")
if(NOT EXR_HOST_SDK_VERSION MATCHES "^3\\.(40|41|42)\\.0$")
  message(FATAL_ERROR "Unsupported Android host SDK version: ${EXR_HOST_SDK_VERSION}")
endif()
if(EXR_HOST_SDK_VERSION STREQUAL "3.40.0" AND (EXR_HOST_PAGE_THEME OR EXR_READER_SELECTOR_BYTES_CANDIDATE OR EXR_HOST_WAVE3))
  message(FATAL_ERROR "SDK 3.40 host cannot enable 3.41 candidate features")
endif()
if(EXR_READER_SELECTOR_BYTES_CANDIDATE AND (NOT EXR_HOST_READING OR NOT EXR_HOST_SYNC))
  message(FATAL_ERROR "Raw3 host requires the complete reading/sync family")
endif()
option(EXR_ANDROID_RECORD_INTEROP_REVIEW "Debug-only real retained-wire review probe" OFF)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(NOT EXR_SDK)
  message(FATAL_ERROR "EXR_SDK must point at an installed EXReader SDK")
endif()

# The Android toolchain otherwise restricts package lookup to the NDK sysroot.
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE BOTH)
list(APPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
set(EXReader_DIR "${EXR_SDK}/lib/cmake/EXReader" CACHE PATH "Installed EXReader package" FORCE)
find_package(EXReader ${EXR_HOST_SDK_VERSION} EXACT CONFIG REQUIRED)

add_library(exreader_host SHARED exreader_host_jni.c)
target_link_libraries(exreader_host PRIVATE EXReader::exreader)
# Consume installed admission archives without pulling a second static SDK
# provider into the JNI process. Every SDK call continues through libexreader.so.
if(NOT EXR_C02_PREFIX)
  message(FATAL_ERROR "EXR_C02_PREFIX must contain installed C02 admission")
endif()
foreach(_c02_library exreader-posix-c02-sdk-admission exreader-posix-c02-admission exreader-migration-coordinator-hardening)
  if(NOT EXISTS "${EXR_C02_PREFIX}/lib/lib${_c02_library}.a")
    message(FATAL_ERROR "Missing installed C02 archive: ${_c02_library}")
  endif()
  target_link_libraries(exreader_host PRIVATE "${EXR_C02_PREFIX}/lib/lib${_c02_library}.a")
endforeach()
target_include_directories(exreader_host PRIVATE "${EXR_C02_PREFIX}/include/exreader-c02-admission")
option(EXR_HOST_C02_TEST "Debug app-UID C02 process actors" OFF)
if(EXR_HOST_C02_TEST)
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    message(FATAL_ERROR "C02 process actors require a Debug build")
  endif()
  target_compile_definitions(exreader_host PRIVATE READER_C02_TEST=1)
endif()
if(EXR_HOST_PAGE_THEME)
  if(NOT EXR_HOST_PREFERENCES OR NOT EXR_HOST_READING OR NOT EXReader_VERSION VERSION_EQUAL "3.41.0")
    message(FATAL_ERROR "Page theme requires preferences, reading and installed SDK 3.41.0")
  endif()
  target_compile_definitions(exreader_host PRIVATE READER_HOST_PAGE_THEME=1)
endif()
if(EXR_HOST_PREFERENCES)
  if(NOT EXR_HOST_READING)
    message(FATAL_ERROR "Reading settings require the public reading anchor bridge")
  endif()
  list(APPEND CMAKE_PREFIX_PATH "${EXR_PREFERENCES_PREFIX}")
  find_package(EXReaderPreferences 0.1.0 EXACT CONFIG REQUIRED)
  target_link_libraries(exreader_host PRIVATE EXReaderPreferences::reader_preferences)
  target_compile_definitions(exreader_host PRIVATE READER_HOST_PREFERENCES=1)
endif()
target_compile_options(exreader_host PRIVATE -Wall -Wextra)
target_link_options(exreader_host PRIVATE -Wl,-z,max-page-size=16384)

if(EXR_HOST_SEARCH)
  list(APPEND CMAKE_PREFIX_PATH "${EXR_SEARCH_PREFIX}")
  find_package(EXReaderSearch 0.1.0 EXACT CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  target_compile_definitions(exreader_host PRIVATE READER_HOST_SEARCH=1)
  target_include_directories(exreader_host PRIVATE
    $<TARGET_PROPERTY:EXReaderSearch::reader_search,INTERFACE_INCLUDE_DIRECTORIES>)
  # Keep the existing shared SDK provider; never link Search's static core closure.
  target_link_libraries(exreader_host PRIVATE $<TARGET_FILE:EXReaderSearch::reader_search> Threads::Threads dl)
  set_target_properties(exreader_host PROPERTIES LINKER_LANGUAGE CXX)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(exreader_host PRIVATE READER_SEARCH_FIXTURES=1)
  endif()
endif()

if(EXR_HOST_NAVIGATION)
  list(APPEND CMAKE_PREFIX_PATH "${EXR_NAVIGATION_PREFIX}")
  find_package(EXReaderNavigation 0.1.0 EXACT CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  target_compile_definitions(exreader_host PRIVATE READER_HOST_NAVIGATION=1)
  target_include_directories(exreader_host PRIVATE
    $<TARGET_PROPERTY:EXReaderNavigation::reader_navigation,INTERFACE_INCLUDE_DIRECTORIES>)
  target_link_libraries(exreader_host PRIVATE $<TARGET_FILE:EXReaderNavigation::reader_navigation> Threads::Threads dl)
  set_target_properties(exreader_host PROPERTIES LINKER_LANGUAGE CXX)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(exreader_host PRIVATE READER_NAVIGATION_FIXTURES=1)
  endif()
endif()

# Installed product modules use their own SQLite provider; the frozen SDK stays
# in the original host library. Hide archive symbols against Android interpose.
if(NOT EXR_PRODUCT)
  message(FATAL_ERROR "EXR_PRODUCT must point at installed native product packages")
endif()
list(APPEND CMAKE_PREFIX_PATH "${EXR_PRODUCT}")
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  find_package(EXReaderRevisions 0.4.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderReadingState 0.2.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderCommands 0.2.0 EXACT CONFIG REQUIRED)
else()
  find_package(EXReaderCommands 0.1.0 EXACT CONFIG REQUIRED)
endif()
add_library(exreader_product SHARED catalog_jni.cpp)
target_compile_features(exreader_product PRIVATE cxx_std_17)
target_link_libraries(exreader_product PRIVATE EXReaderCommands::reader_commands
  EXReaderProductCatalog::reader_catalog EXReaderProductCatalog::reader_catalog_sqlite dl)
target_compile_options(exreader_product PRIVATE -Wall -Wextra -fvisibility=hidden)
if(EXR_ANDROID_RECORD_INTEROP_REVIEW AND CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(exreader_product PRIVATE READER_RECORD_INTEROP_REVIEW=1)
  target_include_directories(exreader_product PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../tests/reader_android_record_interop/cpp")
  if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
    target_compile_definitions(exreader_product PRIVATE READER_RAW3_REVIEW=1)
    target_include_directories(exreader_product PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../tests/reader_android_raw3/cpp")
  endif()
endif()
target_link_options(exreader_product PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL
  -Wl,-Bsymbolic -Wl,--trace-symbol=sqlite3_open_v2 "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/product-link.map")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(exreader_product PRIVATE READER_PRODUCT_FIXTURES=1)
endif()

if(EXR_HOST_SYNC)
  list(APPEND CMAKE_PREFIX_PATH "${EXR_DAV}" "${EXR_SIDELOAD}")
  if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
    find_package(EXReaderSync 0.2.0 EXACT CONFIG REQUIRED)
    find_package(EXReaderSession 0.2.0 EXACT CONFIG REQUIRED)
  else()
    find_package(EXReaderSession 0.1.0 EXACT CONFIG REQUIRED)
  endif()
  find_package(EXReaderSideload CONFIG REQUIRED)
  target_sources(exreader_product PRIVATE sync_jni.cpp)
  target_link_libraries(exreader_product PRIVATE EXReaderSession::reader_session)
  target_compile_definitions(exreader_host PRIVATE READER_HOST_SYNC=1)
  # The existing shared SDK also serves exreader-jni. Consume the installed
  # public sideload archive without adding its static SDK dependency closure.
  target_include_directories(exreader_host PRIVATE
    $<TARGET_PROPERTY:EXReaderSideload::exr_sideload,INTERFACE_INCLUDE_DIRECTORIES>)
  target_link_libraries(exreader_host PRIVATE $<TARGET_FILE:EXReaderSideload::exr_sideload> dl)
  target_link_options(exreader_host PRIVATE -Wl,--exclude-libs,ALL
    "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/sdk-host-link.map")
endif()

if(EXR_HOST_READING)
  if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
    find_package(EXReaderReadingCommands 0.2.0 EXACT CONFIG REQUIRED)
    find_package(EXReaderAnnotationView 0.2.0 EXACT CONFIG REQUIRED)
  else()
    find_package(EXReaderReadingCommands 0.1.0 EXACT CONFIG REQUIRED)
    find_package(EXReaderAnnotationView 0.1.0 EXACT CONFIG REQUIRED)
  endif()
  target_link_libraries(exreader_product PRIVATE EXReaderReadingCommands::reader_reading_commands
    EXReaderAnnotationView::reader_annotation_view)
  target_compile_definitions(exreader_host PRIVATE READER_HOST_READING=1)
  target_compile_definitions(exreader_product PRIVATE READER_HOST_READING=1)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(exreader_host PRIVATE READER_READING_FIXTURES=1)
    target_compile_definitions(exreader_product PRIVATE READER_READING_FIXTURES=1)
  endif()
endif()

if(EXR_HOST_WAVE3)
  if(NOT EXR_READER_SELECTOR_BYTES_CANDIDATE OR NOT EXR_HOST_READING OR NOT EXR_HOST_SYNC)
    message(FATAL_ERROR "Wave3 host integration requires the raw0.2 family with reading and sync")
  endif()
  list(APPEND CMAKE_PREFIX_PATH "${EXR_WAVE3_PREFIX}")
  find_package(EXReaderAnnotationImport 0.2.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderAnnotationExport 0.2.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderJumpHistory 0.2.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderSyncView 0.2.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderFontLibrary 0.1.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderOriginalExport 0.1.0 EXACT CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  # Product side: SDK-free archives only. Jump history is linked by file so that its
  # SDK-resolving reading-state companion never enters the product library.
  target_link_libraries(exreader_product PRIVATE EXReaderAnnotationImport::reader_annotation_import
    EXReaderAnnotationExport::reader_annotation_export EXReaderSyncView::reader_sync_view
    $<TARGET_FILE:EXReaderJumpHistory::reader_jump_history>)
  target_include_directories(exreader_product PRIVATE
    $<TARGET_PROPERTY:EXReaderJumpHistory::reader_jump_history,INTERFACE_INCLUDE_DIRECTORIES>)
  target_compile_definitions(exreader_product PRIVATE READER_HOST_WAVE3=1)
  # SDK side: archives are consumed by file so the existing shared SDK stays the only provider.
  target_compile_definitions(exreader_host PRIVATE READER_HOST_WAVE3=1)
  target_include_directories(exreader_host PRIVATE
    $<TARGET_PROPERTY:EXReaderFontLibrary::reader_font_library,INTERFACE_INCLUDE_DIRECTORIES>
    $<TARGET_PROPERTY:EXReaderOriginalExport::reader_original_export,INTERFACE_INCLUDE_DIRECTORIES>)
  target_link_libraries(exreader_host PRIVATE $<TARGET_FILE:EXReaderFontLibrary::reader_font_library>
    $<TARGET_FILE:EXReaderOriginalExport::reader_original_export> Threads::Threads dl)
  set_target_properties(exreader_host PROPERTIES LINKER_LANGUAGE CXX)
endif()
