# Independent consumer build for the Windows reference host.
#
# examples/CMakeLists.txt is owned by the integration branch, so this project
# consumes the installed SDK directly and reuses ../reader.c without touching
# the shared example entry point. It also works from the installed tree
# (stage/examples/windows).
cmake_minimum_required(VERSION 3.24)
project(EXReaderWindowsHost LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT EXR_SDK)
  message(FATAL_ERROR "Set -DEXR_SDK=<installed SDK prefix>")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
include("${EXR_SDK}/lib/cmake/EXReader/EXReaderConfigVersion.cmake")
find_package(EXReader ${PACKAGE_VERSION} EXACT CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_executable(EXReader WIN32 ../reader.c)
option(EXR_READER_TEST_DRIVERS "Compile acceptance commands (test builds only)" OFF)
if(EXR_READER_TEST_DRIVERS)
  target_compile_definitions(EXReader PRIVATE EXR_READER_TEST_DRIVERS=1)
endif()
target_link_libraries(EXReader PRIVATE EXReader::exreader comdlg32 gdi32 uiautomationcore shell32 user32 winhttp crypt32)
if(NOT MSVC)
  target_link_libraries(EXReader PRIVATE m)
endif()
set(EXR_READER_WINDOWS_SHELL ON) # The retired native toolbar has no build path.
if(EXR_READER_WINDOWS_SHELL)
  enable_language(CXX)
  target_sources(EXReader PRIVATE shell_d2d.cpp shell_epub.cpp shell_speech.cpp)
  target_link_libraries(EXReader PRIVATE ole32 uuid)
  # The same warning gate as the standalone exreader-shell-d2d library, on this
  # one file only: the C sources of EXReader keep their own flags.
  if(MSVC)
    set_source_files_properties(shell_d2d.cpp PROPERTIES COMPILE_OPTIONS "/W4;/WX")
  else()
    set_source_files_properties(shell_d2d.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
  endif()
  target_compile_features(EXReader PRIVATE cxx_std_17)
  target_compile_definitions(EXReader PRIVATE EXR_READER_WINDOWS_SHELL=1)
  target_link_libraries(EXReader PRIVATE d2d1 dwrite advapi32 winhttp EXReader::vendor_miniz EXReader::vendor_expat)
  add_executable(check_shell_epub EXCLUDE_FROM_ALL check_shell_epub.cpp)
  target_compile_features(check_shell_epub PRIVATE cxx_std_17)
  target_link_libraries(check_shell_epub PRIVATE EXReader::exreader EXReader::vendor_miniz EXReader::vendor_expat)
  add_executable(check_shell_opds EXCLUDE_FROM_ALL check_shell_opds.c)
  target_link_libraries(check_shell_opds PRIVATE EXReader::exreader winhttp ws2_32)
  add_executable(check_shell_geometry EXCLUDE_FROM_ALL check_shell_geometry.c)
  target_link_libraries(check_shell_geometry PRIVATE EXReader::exreader)
  if(MSVC)
    target_compile_options(check_shell_geometry PRIVATE /UNDEBUG)
  else()
    target_compile_options(check_shell_geometry PRIVATE -UNDEBUG)
    target_link_libraries(check_shell_geometry PRIVATE m)
  endif()
  add_executable(check_shell_anchor EXCLUDE_FROM_ALL check_shell_anchor.c shell_d2d.cpp shell_epub.cpp shell_speech.cpp)
  target_link_libraries(check_shell_anchor PRIVATE ole32 uuid)
  target_compile_features(check_shell_anchor PRIVATE cxx_std_17)
  target_compile_definitions(check_shell_anchor PRIVATE EXR_READER_WINDOWS_SHELL=1)
  target_link_libraries(check_shell_anchor PRIVATE EXReader::exreader comdlg32 gdi32 uiautomationcore shell32 user32 winhttp crypt32 d2d1 dwrite advapi32 EXReader::vendor_miniz EXReader::vendor_expat)
  if(MSVC)
    target_compile_options(check_shell_anchor PRIVATE /UNDEBUG)
  else()
    target_compile_options(check_shell_anchor PRIVATE -UNDEBUG)
    target_link_options(check_shell_anchor PRIVATE -static)
    target_link_libraries(check_shell_anchor PRIVATE m)
  endif()
  # Console acceptance: exercises the real editor and D2D into a memory DC, no HWND.
  add_executable(check_shell_docedit EXCLUDE_FROM_ALL check_shell_docedit.c shell_d2d.cpp shell_epub.cpp shell_speech.cpp)
  add_executable(check_shell_table EXCLUDE_FROM_ALL check_shell_table.c shell_d2d.cpp shell_epub.cpp shell_speech.cpp)
  target_compile_features(check_shell_table PRIVATE cxx_std_17)
  target_compile_definitions(check_shell_table PRIVATE EXR_READER_WINDOWS_SHELL=1)
  target_link_libraries(check_shell_table PRIVATE EXReader::exreader comdlg32 gdi32 uiautomationcore shell32 user32
    winhttp crypt32 d2d1 dwrite advapi32 ole32 uuid EXReader::vendor_miniz EXReader::vendor_expat)
  if(MSVC)
    target_compile_options(check_shell_table PRIVATE /UNDEBUG)
  else()
    target_compile_options(check_shell_table PRIVATE -UNDEBUG)
    target_link_options(check_shell_table PRIVATE -static)
    target_link_libraries(check_shell_table PRIVATE m)
  endif()
  enable_testing()
  add_test(NAME windows-table-ui COMMAND check_shell_table "${CMAKE_CURRENT_BINARY_DIR}/table-evidence"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/rich.docx")
  target_compile_features(check_shell_docedit PRIVATE cxx_std_17)
  target_compile_definitions(check_shell_docedit PRIVATE EXR_READER_WINDOWS_SHELL=1)
  target_link_libraries(check_shell_docedit PRIVATE EXReader::exreader comdlg32 gdi32 uiautomationcore shell32 user32
    winhttp crypt32 d2d1 dwrite advapi32 ole32 uuid EXReader::vendor_miniz EXReader::vendor_expat)
  if(MSVC)
    target_compile_options(check_shell_docedit PRIVATE /UNDEBUG)
  else()
    target_compile_options(check_shell_docedit PRIVATE -UNDEBUG)
    target_link_options(check_shell_docedit PRIVATE -static)
    target_link_libraries(check_shell_docedit PRIVATE m)
  endif()
  if(NOT MSVC)
    target_link_options(EXReader PRIVATE -static)
  endif()
  # Test-only WebDAV server for the cloud-sync acceptance (rclone is not required).
  add_executable(mock_webdav EXCLUDE_FROM_ALL mock_webdav_main.c)
  target_link_libraries(mock_webdav PRIVATE ws2_32)
  if(NOT MSVC)
    target_link_options(mock_webdav PRIVATE -municode)
  endif()
endif()
get_target_property(exr_type EXReader::exreader TYPE)
if(exr_type STREQUAL "SHARED_LIBRARY")
  find_file(EXR_RUNTIME NAMES exreader.dll libexreader.dll
            PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
  add_custom_command(TARGET EXReader POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:EXReader>"
    VERBATIM)
endif()
