cmake_minimum_required(VERSION 3.24)
project(EXReaderFindAcceptance LANGUAGES C CXX)
find_package(EXReader CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
set(EXR_READER_TEST_DRIVERS ON CACHE BOOL "" FORCE)
add_subdirectory(.. host)
get_target_property(find_links check_shell_docedit LINK_LIBRARIES)
add_executable(check_shell_docedit_find ../check_shell_docedit_find.c ../shell_d2d.cpp ../shell_epub.cpp ../shell_speech.cpp
  ../../../src/docx_edit.c)
target_compile_features(check_shell_docedit_find PRIVATE c_std_11 cxx_std_17)
target_include_directories(check_shell_docedit_find PRIVATE ../../../include)
target_compile_definitions(check_shell_docedit_find PRIVATE EXR_READER_WINDOWS_SHELL=1 EXR_STATIC)
target_link_libraries(check_shell_docedit_find PRIVATE ${find_links})
if(MSVC)
  target_compile_options(check_shell_docedit_find PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(check_shell_docedit_find PRIVATE -O0 -UNDEBUG -ffunction-sections -fdata-sections)
  target_link_options(check_shell_docedit_find PRIVATE -static -Wl,--gc-sections)
endif()
find_file(find_runtime NAMES exreader.dll libexreader.dll PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH)
if(find_runtime)
  add_custom_command(TARGET check_shell_docedit_find POST_BUILD COMMAND ${CMAKE_COMMAND} -E
    copy_if_different "${find_runtime}" "$<TARGET_FILE_DIR:check_shell_docedit_find>")
endif()
enable_testing()
add_test(NAME shell_docedit_find COMMAND check_shell_docedit_find)
