cmake_minimum_required(VERSION 3.24)
project(EXReaderLinuxHost LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(EXR_SDK_VERSION "3.40.0" CACHE STRING "Exact installed SDK: frozen 3.40.0 or explicit candidate")
if(NOT EXR_SDK_VERSION MATCHES "^3\\.(40|41|42)\\.0$")
  message(FATAL_ERROR "Unsupported exact reader SDK version: ${EXR_SDK_VERSION}")
endif()
find_package(EXReader ${EXR_SDK_VERSION} EXACT CONFIG REQUIRED COMPONENTS library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
add_library(exreader-linux-reading STATIC reading.c)
target_compile_options(exreader-linux-reading PRIVATE -Wall -Wextra -Werror)
target_link_libraries(exreader-linux-reading PUBLIC EXReader::exr_core EXReader::exr_library)
option(EXR_MIGRATION_HOST_TEST "Build transaction migration lifecycle probe" OFF)
if(EXR_MIGRATION_HOST_TEST)
  add_executable(exreader-linux-migration-lifecycle ../../tests/migration/linux_host_lifecycle.c)
  target_compile_options(exreader-linux-migration-lifecycle PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(exreader-linux-migration-lifecycle PRIVATE exreader-linux-reading)
endif()
option(EXR_READER_PAGE_BACKGROUND "Use the explicitly selected SDK 3.41 page-canvas API" OFF)
if(EXR_READER_PAGE_BACKGROUND)
  if(NOT EXR_SDK_VERSION STREQUAL "3.41.0")
    message(FATAL_ERROR "Page background requires the explicit SDK 3.41.0 candidate")
  endif()
  target_compile_definitions(exreader-linux-reading PUBLIC EXR_READER_PAGE_BACKGROUND=1)
endif()
add_executable(exreader-linux reader.c)
option(EXR_READER_GTK_COMPACT "Use optional compact native library and reading views" OFF)
if(EXR_READER_GTK_COMPACT)
  target_compile_definitions(exreader-linux PRIVATE EXR_READER_GTK_COMPACT=1)
endif()
target_compile_options(exreader-linux PRIVATE -Wall -Wextra -Werror)
target_link_libraries(exreader-linux PRIVATE exreader-linux-reading PkgConfig::GTK)
install(TARGETS exreader-linux RUNTIME DESTINATION bin)
option(EXR_LINUX_REVIEW_TESTS "Build isolated fault regressions, not installed" OFF)
if(EXR_LINUX_REVIEW_TESTS)
  add_executable(exreader-linux-bounded-backend ../../tests/linux_host/bounded_backend.c)
  target_compile_options(exreader-linux-bounded-backend PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(exreader-linux-bounded-backend PRIVATE exreader-linux-reading)
  foreach(test notes_failure font_restore_mutant)
    add_executable(exreader-linux-${test} ../../tests/linux_host/${test}.c)
    target_compile_options(exreader-linux-${test} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(exreader-linux-${test} PRIVATE EXReader::exr_core EXReader::exr_library PkgConfig::GTK)
    if(test STREQUAL "font_restore_mutant")
      target_link_libraries(exreader-linux-${test} PRIVATE exreader-linux-reading)
    endif()
  endforeach()
endif()
option(EXR_LINUX_FIXED_TESTS "Build fixed-page public SDK host regression" OFF)
if(EXR_LINUX_FIXED_TESTS)
  add_executable(exreader-linux-fixed-backend ../../tests/linux_host/fixed_backend.c)
  target_compile_options(exreader-linux-fixed-backend PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(exreader-linux-fixed-backend PRIVATE exreader-linux-reading)
  add_executable(exreader-linux-fixed-import ../../tests/linux_host/fixed_import_snapshot.c)
  target_compile_options(exreader-linux-fixed-import PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(exreader-linux-fixed-import PRIVATE EXReader::exr_core EXReader::exr_library)
endif()
