cmake_minimum_required(VERSION 3.24)
project(AnnotationImportConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderAnnotationImport 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderAnnotationExport 0.1.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(annotation-import-c api.c)
set_target_properties(annotation-import-c PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(annotation-import-c PRIVATE EXReaderAnnotationImport::reader_annotation_import)
add_executable(annotation-import-probe probe.cpp)
target_link_libraries(annotation-import-probe PRIVATE EXReaderAnnotationImport::reader_annotation_import EXReaderAnnotationExport::reader_annotation_export EXReaderProductCatalog::reader_catalog_sqlite Threads::Threads)
if(MSVC)
  target_link_options(annotation-import-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/annotation-import-link.map")
elseif(APPLE)
  target_link_options(annotation-import-probe PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/annotation-import-link.map")
else()
  target_link_options(annotation-import-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/annotation-import-link.map")
endif()
