cmake_minimum_required(VERSION 3.24)
project(WindowsAdmissionLeaseConsumer LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(windows-lease-check probe.c)
target_include_directories(windows-lease-check PRIVATE "${LEASE_PREFIX}/include/exreader-migration-candidate")
target_compile_definitions(windows-lease-check PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00 "SOURCE_SHA=\"${SOURCE_SHA}\"")
target_compile_options(windows-lease-check PRIVATE /W4 /WX /utf-8)
target_link_libraries(windows-lease-check PRIVATE "${LEASE_PREFIX}/lib/exreader-migration-windows-lease.lib")
