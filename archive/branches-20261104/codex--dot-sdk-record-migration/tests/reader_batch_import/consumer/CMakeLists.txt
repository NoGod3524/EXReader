cmake_minimum_required(VERSION 3.24)
project(reader_batch_import_consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(EXR_SDK_VERSION "3.40.0" CACHE STRING "Exact installed SDK version used by the batch importer")
find_package(EXReader ${EXR_SDK_VERSION} EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderSideload CONFIG REQUIRED)
find_package(EXReaderBatchImport 0.1.0 EXACT CONFIG REQUIRED)
add_executable(batch-probe batch_probe.c)
target_link_libraries(batch-probe PRIVATE EXReaderBatchImport::reader_batch_import)
set_target_properties(batch-probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
add_executable(batch-cpp-consumer batch_cpp_consumer.cpp)
target_link_libraries(batch-cpp-consumer PRIVATE EXReaderBatchImport::reader_batch_import)
find_package(Threads REQUIRED)
add_executable(batch-review-probe batch_review_probe.cpp)
target_link_libraries(batch-review-probe PRIVATE EXReaderBatchImport::reader_batch_import Threads::Threads)
