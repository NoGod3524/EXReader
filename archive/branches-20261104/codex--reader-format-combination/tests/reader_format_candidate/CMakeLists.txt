cmake_minimum_required(VERSION 3.24)
project(EXReaderFormatCandidate LANGUAGES C)
option(FORMAT_ONLY_ARGS "Build only independent argument adapter checks" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
add_executable(format-argv-probe argv_utf8.c)
target_compile_definitions(format-argv-probe PRIVATE FORMAT_ARGV_PROBE)
add_executable(format-argv-self-test argv_utf8.c)
target_compile_definitions(format-argv-self-test PRIVATE FORMAT_ARGV_SELF_TEST)
foreach(target format-argv-probe format-argv-self-test)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(WIN32)
    target_link_libraries(${target} PRIVATE shell32)
  endif()
endforeach()
if(NOT FORMAT_ONLY_ARGS)
  list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
  find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED)
  configure_file("${FORMAT_ORIGINAL_CONSUMER}" "${CMAKE_CURRENT_BINARY_DIR}/f5-consumer-original.c" COPYONLY)
  add_executable(format-f5-consumer argv_utf8.c)
  target_include_directories(format-f5-consumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(format-f5-consumer PRIVATE EXReader::exreader ${CMAKE_DL_LIBS})
  set_target_properties(format-f5-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(WIN32)
    target_link_libraries(format-f5-consumer PRIVATE shell32)
    add_custom_command(TARGET format-f5-consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:EXReader::exreader>" "$<TARGET_FILE_DIR:format-f5-consumer>")
  endif()
endif()
