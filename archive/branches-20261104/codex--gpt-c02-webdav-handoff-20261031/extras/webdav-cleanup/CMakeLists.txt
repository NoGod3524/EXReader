cmake_minimum_required(VERSION 3.24)
project(EXReaderWebDAVCleanup VERSION 0.1.0 LANGUAGES C)
if(WIN32 OR NOT UNIX)
  message(FATAL_ERROR "EXReaderWebDAVCleanup is POSIX only")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(exr_webdav_cleanup STATIC src/cleanup_posix.c)
add_library(EXReaderWebDAVCleanup::exr_webdav_cleanup ALIAS exr_webdav_cleanup)
target_include_directories(exr_webdav_cleanup PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(exr_webdav_cleanup PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CMakePackageConfigHelpers)
install(TARGETS exr_webdav_cleanup EXPORT EXReaderWebDAVCleanupTargets ARCHIVE DESTINATION lib)
install(FILES include/exr_webdav_cleanup.h DESTINATION include)
install(FILES ../../LICENSE DESTINATION share/exreader-webdav-cleanup RENAME LICENSE.txt)
install(EXPORT EXReaderWebDAVCleanupTargets NAMESPACE EXReaderWebDAVCleanup:: DESTINATION lib/cmake/EXReaderWebDAVCleanup)
configure_package_config_file(EXReaderWebDAVCleanupConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVCleanupConfig.cmake INSTALL_DESTINATION lib/cmake/EXReaderWebDAVCleanup)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVCleanupConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVCleanupConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/EXReaderWebDAVCleanupConfigVersion.cmake DESTINATION lib/cmake/EXReaderWebDAVCleanup)
