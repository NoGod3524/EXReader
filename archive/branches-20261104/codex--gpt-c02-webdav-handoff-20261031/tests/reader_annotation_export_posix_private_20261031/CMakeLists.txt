cmake_minimum_required(VERSION 3.24)
project(AnnotationPrivate LANGUAGES CXX)
if(NOT UNIX)
  message(FATAL_ERROR "POSIX only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(EXReaderAnnotationExport 0.1.0 EXACT CONFIG REQUIRED)
add_executable(annotation-private probe.cpp)
target_link_libraries(annotation-private PRIVATE EXReaderAnnotationExport::reader_annotation_export EXReaderProductCatalog::reader_catalog_sqlite)
target_compile_definitions(annotation-private PRIVATE "SOURCE_SHA=\"${SOURCE_SHA}\"")
target_compile_options(annotation-private PRIVATE -Wall -Wextra -Werror)
if(APPLE)
  target_link_options(annotation-private PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/private.map")
else()
  target_link_options(annotation-private PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/private.map")
endif()
