cmake_minimum_required(VERSION 3.24)
project(ReaderStartupProfile LANGUAGES C)
if(NOT EXR_SDK_VERSION)
  set(EXR_SDK_VERSION 3.41.0)
endif()
if(NOT READER_SOURCE_ROOT)
  get_filename_component(READER_SOURCE_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
find_package(EXReader ${EXR_SDK_VERSION} EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_subdirectory("${READER_SOURCE_ROOT}/apps/reader/store/native" catalog)
add_library(profile_adapter STATIC "${READER_SOURCE_ROOT}/apps/reader/desktop/catalog_sdk.c")
target_include_directories(profile_adapter PUBLIC "${READER_SOURCE_ROOT}/apps/reader/desktop")
target_link_libraries(profile_adapter PUBLIC reader_catalog EXReader::exr_library)
if(WIN32)
  target_link_libraries(profile_adapter PRIVATE bcrypt normaliz)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  target_link_libraries(profile_adapter PRIVATE PkgConfig::GLIB)
endif()
add_executable(startup_profile profile.c)
target_link_libraries(startup_profile PRIVATE profile_adapter reader_catalog_sqlite)
add_executable(seed_shelf ../windows_host/seed_shelf.c)
target_link_libraries(seed_shelf PRIVATE EXReader::exr_library)
add_executable(reader-desktop-catalog-check ../reader_catalog/desktop_sdk_check.c)
target_link_libraries(reader-desktop-catalog-check PRIVATE profile_adapter reader_catalog_sqlite)
if(MSVC)
  target_link_options(reader-desktop-catalog-check PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/sdk-link.map")
endif()
add_executable(reader-native-catalog-check ../reader_catalog/native_catalog_check.c)
target_link_libraries(reader-native-catalog-check PRIVATE reader_catalog reader_catalog_sqlite)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../reader_catalog/desktop_sdk_check.c" diagnostic_source)
string(REPLACE "if(result!=SQLITE_OK) fprintf(stderr,\"SQL: %s\\n\",message?message:\"\");" "if(result!=SQLITE_OK) fprintf(stderr,\"SQL statement: %s; result: %d; extended: %d; system: %d; message: %s\\n\",text,result,sqlite3_extended_errcode(db),sqlite3_system_errno(db),message?message:\"\");" diagnostic_source "${diagnostic_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/diagnostic_catalog.c" "${diagnostic_source}")
add_executable(reader-diagnostic-catalog-check "${CMAKE_CURRENT_BINARY_DIR}/diagnostic_catalog.c")
target_link_libraries(reader-diagnostic-catalog-check PRIVATE profile_adapter reader_catalog_sqlite)
add_executable(reader-wal-crash-check wal_crash.c)
target_link_libraries(reader-wal-crash-check PRIVATE profile_adapter reader_catalog_sqlite)
