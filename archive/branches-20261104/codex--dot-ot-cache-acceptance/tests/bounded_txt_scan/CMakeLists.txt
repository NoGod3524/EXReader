cmake_minimum_required(VERSION 3.24)
project(bounded_txt_scan C CXX)
find_package(EXReader CONFIG REQUIRED)
add_executable(bounded_txt_scan check.c)
target_link_libraries(bounded_txt_scan PRIVATE EXReader::exr_core)
add_executable(bounded_txt_cpp cpp.cpp)
target_link_libraries(bounded_txt_cpp PRIVATE EXReader::exr_core)
enable_testing()
add_test(NAME bounded_txt_cpp COMMAND bounded_txt_cpp)
add_test(NAME bounded_txt_scan COMMAND bounded_txt_scan)
add_executable(bounded_txt_view view.c)
target_link_libraries(bounded_txt_view PRIVATE EXReader::exr_core)
add_test(NAME bounded_txt_view COMMAND bounded_txt_view)

if(WIN32)
  add_executable(bounded_txt_large_pager large_pager.c)
  target_link_libraries(bounded_txt_large_pager PRIVATE EXReader::exr_core psapi)
endif()

add_executable(bounded_txt_library_boundary library_boundary.c)
target_link_libraries(bounded_txt_library_boundary PRIVATE EXReader::exr_library)
add_test(NAME bounded_txt_library_boundary COMMAND bounded_txt_library_boundary)

add_executable(bounded_txt_ot_compare ot_compare.c)
target_link_libraries(bounded_txt_ot_compare PRIVATE EXReader::exr_core)

add_executable(bounded_native_shaping ../test_shaping.c)
add_executable(bounded_native_complex_layout ../test_complex_layout.c)
add_executable(bounded_native_line_boundary ../test_line_boundary.c)
foreach(target bounded_native_shaping bounded_native_complex_layout bounded_native_line_boundary)
  target_link_libraries(${target} PRIVATE EXReader::exr_core)
  if(UNIX)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
add_test(NAME bounded_native_shaping COMMAND bounded_native_shaping ${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/shaping)
add_test(NAME bounded_native_complex_layout COMMAND bounded_native_complex_layout ${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/shaping)
add_test(NAME bounded_native_line_boundary COMMAND bounded_native_line_boundary ${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/shaping ${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/fonts)
