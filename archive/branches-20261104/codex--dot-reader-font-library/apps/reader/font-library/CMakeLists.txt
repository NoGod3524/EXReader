cmake_minimum_required(VERSION 3.24)
project(EXReaderFontLibrary VERSION 0.1.0 LANGUAGES C CXX)
# Standalone, default OFF, never wired into the product/default SDK build.
option(READER_FONT_LIBRARY_BUILD "Build the optional local font library" OFF)
if(NOT READER_FONT_LIBRARY_BUILD)
  return()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core)
find_package(Threads REQUIRED)
add_library(reader_font_library STATIC src/library.cpp)
target_include_directories(reader_font_library PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(reader_font_library PUBLIC EXReader::exr_core Threads::Threads)
if(MSVC)
  target_compile_options(reader_font_library PRIVATE /W4 /WX)
  target_compile_definitions(reader_font_library PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(reader_font_library PRIVATE -Wall -Wextra -Werror)
endif()
include(CMakePackageConfigHelpers)
install(TARGETS reader_font_library EXPORT EXReaderFontLibraryTargets ARCHIVE DESTINATION lib)
install(FILES include/reader_font_library.h DESTINATION include)
install(EXPORT EXReaderFontLibraryTargets NAMESPACE EXReaderFontLibrary:: DESTINATION lib/cmake/EXReaderFontLibrary)
configure_package_config_file(EXReaderFontLibraryConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EXReaderFontLibraryConfig.cmake" INSTALL_DESTINATION lib/cmake/EXReaderFontLibrary)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EXReaderFontLibraryConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderFontLibraryConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EXReaderFontLibraryConfigVersion.cmake" DESTINATION lib/cmake/EXReaderFontLibrary)
