cmake_minimum_required(VERSION 3.24)
option(EXR_READER_SELECTOR_BYTES_CANDIDATE "Explicit raw store companion candidate; never default" OFF)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  project(EXReaderSyncView VERSION 0.2.0 LANGUAGES CXX)
else()
  project(EXReaderSyncView VERSION 0.1.0 LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(EXR_SESSION_VERSION 0.1.0)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  set(EXR_SESSION_VERSION 0.2.0)
endif()
find_package(EXReaderSession ${EXR_SESSION_VERSION} EXACT CONFIG REQUIRED)
get_target_property(_raw_dependency_defs EXReaderSession::reader_session INTERFACE_COMPILE_DEFINITIONS)
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  if(NOT "EXR_READER_SELECTOR_BYTES_CANDIDATE=1" IN_LIST _raw_dependency_defs)
    message(FATAL_ERROR "Exact dependency target belongs to the legacy capability family")
  endif()
else()
  if("EXR_READER_SELECTOR_BYTES_CANDIDATE=1" IN_LIST _raw_dependency_defs)
    message(FATAL_ERROR "Default profile cannot consume a raw candidate dependency target")
  endif()
endif()
add_library(reader_sync_view STATIC src/sync_view.cpp)
target_include_directories(reader_sync_view PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(reader_sync_view PUBLIC EXReaderSession::reader_session)
if(MSVC)
  target_compile_options(reader_sync_view PRIVATE /W4 /WX)
else()
  target_compile_options(reader_sync_view PRIVATE -Wall -Wextra -Werror)
endif()
include(CMakePackageConfigHelpers)
install(TARGETS reader_sync_view EXPORT EXReaderSyncViewTargets ARCHIVE DESTINATION lib)
install(FILES include/reader_sync_view.h DESTINATION include)
install(EXPORT EXReaderSyncViewTargets NAMESPACE EXReaderSyncView:: DESTINATION lib/cmake/EXReaderSyncView)
configure_package_config_file(EXReaderSyncViewConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncViewConfig.cmake"
  INSTALL_DESTINATION lib/cmake/EXReaderSyncView)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncViewConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncViewConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSyncViewConfigVersion.cmake"
  DESTINATION lib/cmake/EXReaderSyncView)

if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  target_compile_definitions(reader_sync_view PUBLIC EXR_READER_SELECTOR_BYTES_CANDIDATE=1)
endif()

# RAW-CANDIDATE ORIGIN GUARD: append-only; never changes imported targets.
if(EXR_READER_SELECTOR_BYTES_CANDIDATE)
  _exr_raw_guard_target(EXReaderSession::reader_session "${EXReaderSession_DIR}"
    reader_session.h READER_SESSION_VERSION 0.2.0 TRUE)
endif()
