cmake_minimum_required(VERSION 3.24)
project(OriginalExportS15Consumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
# The full probe needs the unchanged product packages; the C consumer needs only
# the installed original-export package and its exact SDK dependency.
option(S15_PROBE_CONSUMER "Build the C++ probe against the installed product packages" ON)
# Fresh installed consumers: no source include path, only installed packages.
find_package(EXReaderOriginalExport 0.1.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(original-export-c-consumer api.c)
set_target_properties(original-export-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(original-export-c-consumer PRIVATE EXReaderOriginalExport::reader_original_export)
if(S15_PROBE_CONSUMER)
  find_package(EXReaderCommands 0.1.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderSideload CONFIG REQUIRED)
  add_executable(original-export-probe probe.cpp)
  target_link_libraries(original-export-probe PRIVATE EXReaderOriginalExport::reader_original_export EXReaderCommands::reader_commands EXReaderSideload::exr_sideload Threads::Threads)
  if(MSVC)
    target_link_options(original-export-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/original-export-link.map")
  elseif(APPLE)
    target_link_options(original-export-probe PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/original-export-link.map")
  else()
    target_link_options(original-export-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/original-export-link.map")
  endif()
endif()
