cmake_minimum_required(VERSION 3.24)
project(WindowsRestartReaderConsumer LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(EXReader 3.42.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_executable(windows-restart-reader-check probe.c)
target_include_directories(windows-restart-reader-check PRIVATE "${RESTART_PREFIX}/include/exreader-migration-candidate" "${LEASE_PREFIX}/include/exreader-migration-candidate" "${BRIDGE_PREFIX}/include/exreader-migration-candidate" "${PUBLISH_PREFIX}/include/exreader-migration-candidate" "${RECORD_PREFIX}/include/exreader-migration-candidate")
target_compile_definitions(windows-restart-reader-check PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00 "SOURCE_SHA=\"${SOURCE_SHA}\"")
target_compile_options(windows-restart-reader-check PRIVATE /W4 /WX /utf-8)
target_link_libraries(windows-restart-reader-check PRIVATE "${RESTART_PREFIX}/lib/exreader-migration-windows-restart-reader.lib" "${LEASE_PREFIX}/lib/exreader-migration-windows-lease.lib" "${BRIDGE_PREFIX}/lib/exreader-migration-windows-record-bridge.lib" "${PUBLISH_PREFIX}/lib/exreader-migration-windows-publish.lib" "${RECORD_PREFIX}/lib/exreader-migration-windows-record.lib" EXReader::exr_library advapi32 psapi)
target_link_options(windows-restart-reader-check PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/restart-consumer.map")
