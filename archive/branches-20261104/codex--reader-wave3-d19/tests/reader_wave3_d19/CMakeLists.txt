cmake_minimum_required(VERSION 3.24)
project(D19CatalogReadback LANGUAGES C)
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
find_package(EXReaderProductCatalog CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
add_executable(d19-catalog catalog.c ../../apps/reader/desktop/catalog_sdk.c)
target_include_directories(d19-catalog PRIVATE ../../apps/reader/desktop)
target_link_libraries(d19-catalog PRIVATE EXReader::exr_library EXReaderProductCatalog::reader_catalog PkgConfig::GLIB)
target_compile_options(d19-catalog PRIVATE -Wall -Wextra -Werror)
