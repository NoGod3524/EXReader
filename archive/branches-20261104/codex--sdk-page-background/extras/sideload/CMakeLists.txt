cmake_minimum_required(VERSION 3.24)
project(EXReaderSideload VERSION 0.1.0 LANGUAGES C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
if(EXR_SDK)
  list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
endif()
find_package(EXReader CONFIG REQUIRED COMPONENTS library)
add_library(exr_sideload STATIC src/sideload.c)
target_include_directories(exr_sideload PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(exr_sideload PUBLIC EXReader::exr_library)
set_target_properties(exr_sideload PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
install(TARGETS exr_sideload EXPORT EXReaderSideloadTargets ARCHIVE DESTINATION lib)
install(FILES include/exr_sideload.h DESTINATION include)
install(EXPORT EXReaderSideloadTargets NAMESPACE EXReaderSideload:: DESTINATION lib/cmake/EXReaderSideload)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSideloadConfig.cmake"
  "include(CMakeFindDependencyMacro)\nfind_dependency(EXReader CONFIG COMPONENTS library)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/EXReaderSideloadTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EXReaderSideloadConfig.cmake" DESTINATION lib/cmake/EXReaderSideload)
if(EXR_SIDELOAD_TEST_FIXTURES)
  enable_testing()
  add_executable(sideload-check ../../tests/sideload/check_sideload.c)
  target_link_libraries(sideload-check PRIVATE exr_sideload)
  set_target_properties(sideload-check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  add_test(NAME sideload-public-consumer COMMAND sideload-check
    "${EXR_SIDELOAD_TEST_FIXTURES}" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
