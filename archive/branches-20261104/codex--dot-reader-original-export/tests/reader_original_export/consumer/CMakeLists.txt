cmake_minimum_required(VERSION 3.24)
project(OriginalExportConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderOriginalExport 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderCommands 0.1.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(original-export-c-consumer api.c)
set_target_properties(original-export-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(original-export-c-consumer PRIVATE EXReaderOriginalExport::reader_original_export)
add_executable(original-export-probe probe.cpp)
target_link_libraries(original-export-probe PRIVATE EXReaderOriginalExport::reader_original_export EXReaderCommands::reader_commands Threads::Threads)
if(MSVC)
  target_link_options(original-export-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/original-export-link.map")
elseif(APPLE)
  target_link_options(original-export-probe PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/original-export-link.map")
else()
  target_link_options(original-export-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/original-export-link.map")
endif()
