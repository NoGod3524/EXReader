cmake_minimum_required(VERSION 3.24)
project(WebDAVCleanupRetry LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "GNU link wrappers and non-root Linux permissions required")
endif()
find_package(EXReaderWebDAV 0.2.0 EXACT CONFIG REQUIRED)
add_executable(webdav-cleanup-retry probe.cpp)
target_compile_features(webdav-cleanup-retry PRIVATE cxx_std_17)
target_compile_options(webdav-cleanup-retry PRIVATE -Wall -Wextra -Werror)
target_link_libraries(webdav-cleanup-retry PRIVATE EXReaderWebDAV::exr_webdav)
target_link_options(webdav-cleanup-retry PRIVATE "-Wl,--wrap=linkat" "-Wl,--wrap=unlinkat" "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/retry.map")
target_compile_definitions(webdav-cleanup-retry PRIVATE "SOURCE_SHA=\"${SOURCE_SHA}\"")
