if(EXR_READER_BUILD_GTK)
  if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "The GTK product host requires Linux")
  endif()
  add_subdirectory(../../../examples/linux gtk-host)
  if(EXR_READER_GTK_BATCH_IMPORT)
    list(PREPEND CMAKE_PREFIX_PATH "${READER_BATCH_IMPORT_PREFIX}" "${EXR_SDK}")
    find_package(EXReaderBatchImport 0.1.0 EXACT CONFIG REQUIRED)
    target_compile_definitions(exreader-linux-reading PUBLIC EXR_READER_BATCH_IMPORT=1)
    target_compile_definitions(exreader-linux PRIVATE EXR_READER_BATCH_IMPORT_UI=1)
    target_link_libraries(exreader-linux-reading PUBLIC EXReaderBatchImport::reader_batch_import PRIVATE PkgConfig::GLIB)

      target_compile_options(reader-gtk-drop-backend PRIVATE -Wall -Wextra -Werror)
      target_link_libraries(reader-gtk-drop-backend PRIVATE exreader-linux-reading reader_catalog_sdk reader_catalog_sqlite PkgConfig::GLIB)

      message(FATAL_ERROR "GTK account integration requires the installed metadata and account module prefixes")
    endif()
    list(PREPEND CMAKE_PREFIX_PATH "${READER_ACCOUNT_PREFIX}")
    find_package(EXReaderCredentials 0.1.0 EXACT CONFIG REQUIRED)
    if(EXR_READER_GTK_RAW3)
      find_package(EXReaderSession 0.2.0 EXACT CONFIG REQUIRED)
      find_package(EXReaderReadingState 0.2.0 EXACT CONFIG REQUIRED)
    else()
      find_package(EXReaderSession 0.1.0 EXACT CONFIG REQUIRED)
      find_package(EXReaderReadingState 0.1.0 EXACT CONFIG REQUIRED)
    endif()
    find_package(EXReaderSideload CONFIG REQUIRED)
    target_compile_definitions(exreader-linux-reading PRIVATE EXR_READER_ACCOUNT=1)
    target_compile_definitions(exreader-linux PRIVATE EXR_READER_ACCOUNT=1)
    target_link_libraries(exreader-linux-reading PRIVATE EXReaderCredentials::exr_credentials
      EXReaderSideload::exr_sideload EXReaderWebDAV::exr_webdav EXReaderReadingState::reader_reading_state)
    # Like the installed reading-state consumer, make the product provider a
    # direct dependency before the SDK's transitive vendor archives.
    target_link_libraries(exreader-linux PRIVATE EXReaderSession::reader_session EXReaderCredentials::exr_credentials
      EXReaderProductCatalog::reader_catalog EXReaderProductCatalog::reader_catalog_sqlite)
    target_link_options(exreader-linux PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/gtk-account-host.map")
  endif()

  if(EXR_READER_GTK_RAW3)
    if(NOT EXR_READER_GTK_ACCOUNT OR NOT EXR_READER_GTK_READING_RECORDS OR NOT EXR_SDK_VERSION MATCHES "^3\\.(41|42)\\.0$")
      message(FATAL_ERROR "GTK raw3 requires explicit SDK3.41 or SDK3.42 and complete account/records integration")
    endif()
    target_compile_definitions(exreader-linux-reading PUBLIC EXR_READER_GTK_RAW3=1)
    target_compile_definitions(exreader-linux PRIVATE EXR_READER_GTK_RAW3=1)
    file(REAL_PATH "${READER_GTK_RAW3_PREFIX}" _gtk_raw_prefix)
    foreach(_gtk_raw_spec
        "EXReaderRevisions|reader_revisions|READER_REVISIONS_VERSION|0.4.0"
        "EXReaderCommands|reader_commands|READER_COMMANDS_VERSION|0.2.0"
        "EXReaderReadingState|reader_reading_state|READER_READING_STATE_VERSION|0.2.0"
        "EXReaderReadingCommands|reader_reading_commands|READER_READING_COMMANDS_VERSION|0.2.0"
        "EXReaderAnnotationView|reader_annotation_view|READER_ANNOTATION_VIEW_VERSION|0.2.0"
        "EXReaderSync|reader_sync|READER_SYNC_VERSION|0.2.0"
        "EXReaderSession|reader_session|READER_SESSION_VERSION|0.2.0"
        "EXReaderDesktopMetadata|reader_metadata|READER_METADATA_VERSION|0.2.0")
      string(REPLACE "|" ";" _gtk_raw_fields "${_gtk_raw_spec}")
      list(GET _gtk_raw_fields 0 _gtk_raw_package)
      list(GET _gtk_raw_fields 1 _gtk_raw_header)
      list(GET _gtk_raw_fields 2 _gtk_raw_macro)
      list(GET _gtk_raw_fields 3 _gtk_raw_version)
      set(_gtk_raw_target "${_gtk_raw_header}")
      if(_gtk_raw_package STREQUAL "EXReaderDesktopMetadata")
        set(_gtk_raw_target reader_desktop_metadata)
      endif()
      file(REAL_PATH "${${_gtk_raw_package}_DIR}/../../.." _gtk_raw_origin)
      if(NOT _gtk_raw_origin STREQUAL _gtk_raw_prefix)
        message(FATAL_ERROR "GTK raw3 family origin differs: ${_gtk_raw_package}")
      endif()
      set(_gtk_raw_definition TRUE)
      if(_gtk_raw_package STREQUAL "EXReaderReadingState")
        _exr_raw_guard_target(EXReaderReadingState::reader_reading_state_codec "${EXReaderReadingState_DIR}"
          reader_reading_state.h READER_READING_STATE_VERSION 0.2.0 TRUE)
        set(_gtk_raw_definition FALSE)
      endif()
      _exr_raw_guard_target(${_gtk_raw_package}::${_gtk_raw_target} "${${_gtk_raw_package}_DIR}"
        "${_gtk_raw_header}.h" "${_gtk_raw_macro}" "${_gtk_raw_version}" ${_gtk_raw_definition})
    endforeach()
  endif()
endif()
