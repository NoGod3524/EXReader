cmake_minimum_required(VERSION 3.24)
project(ReaderNextS13Probe LANGUAGES C)
# Installed public SDK consumer for the S13 CBZ leading-slash candidate.
# EXR_SDK must name the install prefix under test; nothing here reaches into the
# SDK source tree, and no private header is included.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
if(NOT EXR_SDK)
  message(FATAL_ERROR "EXR_SDK must name the installed public SDK prefix")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED COMPONENTS core pdf library)
# One stable location for multi-config and single-config generators alike.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(reader-next-s13-probe reader_next_s13_probe.c)
target_link_libraries(reader-next-s13-probe PRIVATE EXReader::exreader)
set_target_properties(reader-next-s13-probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(NOT MSVC)
  target_compile_options(reader-next-s13-probe PRIVATE -Wall -Wextra)
endif()
if(UNIX)
  target_link_libraries(reader-next-s13-probe PRIVATE m)
endif()
