cmake_minimum_required(VERSION 3.24)
project(EXReaderWindowsPublish LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
find_package(EXReader CONFIG REQUIRED COMPONENTS core library)
find_path(RECORD_INCLUDE lm_recovery_win32_record.h PATHS "${RECORD_PREFIX}/include/exreader-migration-candidate" NO_DEFAULT_PATH REQUIRED)
find_library(RECORD_ARCHIVE exreader-migration-windows-record PATHS "${RECORD_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(exreader-migration-windows-publish STATIC src/lm_win_publish.c)
target_include_directories(exreader-migration-windows-publish PUBLIC include "${RECORD_INCLUDE}")
target_compile_definitions(exreader-migration-windows-publish PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(exreader-migration-windows-publish PRIVATE /W4 /WX /utf-8)
target_link_libraries(exreader-migration-windows-publish PUBLIC "${RECORD_ARCHIVE}" EXReader::exr_library advapi32 psapi)
set_target_properties(exreader-migration-windows-publish PROPERTIES C_STANDARD 11)
install(TARGETS exreader-migration-windows-publish ARCHIVE DESTINATION lib)
install(FILES include/lm_win_publish.h DESTINATION include/exreader-migration-candidate)
