cmake_minimum_required(VERSION 3.24)
project(F07DesktopProviders LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
file(REAL_PATH "${SDK_PREFIX}" SDK_REAL)
file(REAL_PATH "${PRODUCT_PREFIX}" PRODUCT_REAL)
if(NOT IS_ABSOLUTE "${SDK_PREFIX}" OR NOT "${SDK_PREFIX}" STREQUAL "${SDK_REAL}" OR NOT "${PRODUCT_PREFIX}" STREQUAL "${PRODUCT_REAL}")
 message(FATAL_ERROR "F07_PREFIX")
endif()
find_package(EXReader 3.42.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${SDK_PREFIX}" NO_DEFAULT_PATH)
file(REAL_PATH "${EXReader_DIR}/../../.." FOUND_SDK)
if(NOT "${FOUND_SDK}" STREQUAL "${SDK_REAL}")
 message(FATAL_ERROR "F07_PREFIX")
endif()
find_package(EXReaderReadingState 0.2.0 EXACT CONFIG REQUIRED)
find_package(EXReaderProductCatalog 0.1.0 EXACT CONFIG REQUIRED)
foreach(t codec sdk)
 add_executable(f07-${t} probe.c)
 if(MSVC)
  target_link_options(f07-${t} PRIVATE "/MAP:${CMAKE_BINARY_DIR}/${t}.map" "/VERBOSE:LIB")
 elseif(APPLE)
  target_link_options(f07-${t} PRIVATE "-Wl,-map,${CMAKE_BINARY_DIR}/${t}.map")
 else()
  message(FATAL_ERROR "Only MSVC and Apple evidence supported")
 endif()
endforeach()
target_compile_definitions(f07-codec PRIVATE F07_CODEC=1)
target_link_libraries(f07-codec PRIVATE EXReaderReadingState::reader_reading_state_codec EXReaderProductCatalog::reader_catalog_sqlite)
target_link_libraries(f07-sdk PRIVATE EXReader::exr_library EXReader::exr_core)
target_include_directories(f07-sdk PRIVATE "${PRODUCT_PREFIX}/include/exreader-product")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/targets.txt" CONTENT "product=$<TARGET_FILE:EXReaderProductCatalog::reader_catalog_sqlite>\nsdk=$<TARGET_FILE:EXReader::vendor_file_exr_sqlite>\nbase=$<TARGET_FILE:EXReader::exr_base>\nrender=$<TARGET_FILE:EXReader::exr_render>\ncodec=$<TARGET_FILE:EXReaderReadingState::reader_reading_state_codec>\ncore_type=$<TARGET_PROPERTY:EXReader::exr_core,TYPE>\ncore_links=$<TARGET_PROPERTY:EXReader::exr_core,INTERFACE_LINK_LIBRARIES>\n")
