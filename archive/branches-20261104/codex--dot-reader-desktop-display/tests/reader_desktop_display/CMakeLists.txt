cmake_minimum_required(VERSION 3.24)
project(DesktopDisplayConsumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(EXReaderDesktopDisplay 0.1 CONFIG REQUIRED)
foreach(kind c cpp)
  add_executable(consumer_${kind} consumer.${kind})
  target_link_libraries(consumer_${kind} PRIVATE EXReaderDesktopDisplay::display)
endforeach()
if(WIN32)
  add_executable(display_probe probe_windows.c)
else()
  add_executable(display_probe probe_linux.c)
endif()
target_link_libraries(display_probe PRIVATE EXReaderDesktopDisplay::display)
