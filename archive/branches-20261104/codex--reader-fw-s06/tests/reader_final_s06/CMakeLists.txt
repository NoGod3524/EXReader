# S06 consumes only installed packages: the frozen EXReader SDK, the independent
# EXReaderSideload import module and the independent EXReaderDesktopDrop adapter.
cmake_minimum_required(VERSION 3.24)
project(EXReaderFinalS06 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  if(NOT CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_compile_options(/utf-8)
else()
  add_compile_options(-Wall -Wextra)
endif()

if(WIN32)
  set(S06_PLATFORM windows)
  find_package(EXReaderDesktopDrop 0.1 CONFIG REQUIRED COMPONENTS win32)
  set(S06_DROP_TARGET EXReaderDesktopDrop::win32)
  set(S06_HANDOFF_SOURCE src/s06_handoff_windows.c)
  set(S06_EXTRA_LIBS user32)
elseif(APPLE)
  set(S06_PLATFORM macos)
  enable_language(OBJC)
  find_package(EXReaderDesktopDrop 0.1 CONFIG REQUIRED COMPONENTS apple)
  set(S06_DROP_TARGET EXReaderDesktopDrop::apple)
  set(S06_HANDOFF_SOURCE src/s06_handoff_apple.m)
  set(S06_EXTRA_LIBS "")
else()
  set(S06_PLATFORM linux)
  find_package(EXReaderDesktopDrop 0.1 CONFIG REQUIRED COMPONENTS gtk)
  set(S06_DROP_TARGET EXReaderDesktopDrop::gtk)
  set(S06_HANDOFF_SOURCE src/s06_handoff_gtk.c)
  set(S06_EXTRA_LIBS "")
endif()

find_package(EXReader CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderSideload CONFIG REQUIRED)

add_executable(s06-matrix-probe
  src/s06_probe.c src/sha256.c src/handoff_common.c ${S06_HANDOFF_SOURCE})
target_link_libraries(s06-matrix-probe PRIVATE
  EXReaderSideload::exr_sideload ${S06_DROP_TARGET} ${S06_EXTRA_LIBS})
set_target_properties(s06-matrix-probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(APPLE)
  set_target_properties(s06-matrix-probe PROPERTIES LINKER_LANGUAGE CXX)
endif()
message(STATUS "S06 matrix probe platform: ${S06_PLATFORM}")
