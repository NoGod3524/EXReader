cmake_minimum_required(VERSION 3.24)
project(EXReaderUnixDistributionConsumer LANGUAGES C CXX)
# The only SDK input is the installed, relocated prefix. Nothing from the source
# tree is linked into these programs except the existing public probe source.
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core pdf)
foreach(component audio pdf_edit textindex library)
  if(TARGET EXReader::exr_${component})
    message(FATAL_ERROR "reading-core prefix unexpectedly exports ${component}")
  endif()
endforeach()
add_executable(unix-c-consumer main.c)
target_link_libraries(unix-c-consumer PRIVATE EXReader::exreader)
set_target_properties(unix-c-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(UNIX)
  target_link_libraries(unix-c-consumer PRIVATE m)
endif()
add_executable(unix-cxx-consumer main.cpp)
target_link_libraries(unix-cxx-consumer PRIVATE EXReader::exr_core)
set_target_properties(unix-cxx-consumer PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
if(EXR_REPO)
  add_executable(unix-document "${EXR_REPO}/tools/check_document.c")
  target_link_libraries(unix-document PRIVATE EXReader::exreader)
  set_target_properties(unix-document PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(UNIX)
    target_link_libraries(unix-document PRIVATE m)
  endif()
endif()
