cmake_minimum_required(VERSION 3.24)
project(EXReader VERSION 0.11.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
option(EXR_BUILD_TOOLS "Build the standalone native renderer" ON)
option(EXR_BUILD_TESTS "Build native integration tests" ON)
option(EXR_SANITIZE "Address and undefined behavior sanitizers" OFF)
if(EXR_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/deps.cmake)
target_compile_definitions(miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(exreader SHARED src/core.c src/text.c src/document.c src/construct.c src/anchor.c src/publication.c src/formats.c src/style.c src/images.c src/layout.c src/memory.c src/flow.c src/pager.c src/pager_table.c src/raster.c src/library.c src/transfer.c)
target_include_directories(exreader SYSTEM PRIVATE ${stb_SOURCE_DIR})
target_compile_definitions(exreader PRIVATE EXR_BUILD)
if(ANDROID)
  target_link_options(exreader PRIVATE -Wl,-z,max-page-size=16384)
endif()
target_include_directories(exreader PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(exreader PRIVATE freetype expat miniz exr_sqlite)
if(MINGW)
  target_link_options(exreader PRIVATE -static-libgcc -static)
endif()
if(MSVC)
  target_compile_options(exreader PRIVATE /W4)
else()
  target_compile_options(exreader PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(exreader PRIVATE m)
endif()
if(EXR_BUILD_TOOLS)
  add_executable(exreader-render tools/render.c)
  target_link_libraries(exreader-render PRIVATE exreader)
  if(APPLE)
    set_target_properties(exreader-render PROPERTIES INSTALL_RPATH "@loader_path/../lib")
  elseif(UNIX)
    set_target_properties(exreader-render PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  if(WIN32)
    target_link_libraries(exreader-render PRIVATE shell32)
  endif()
  install(TARGETS exreader-render RUNTIME DESTINATION bin)
  add_executable(exreader-benchmark tools/benchmark.c)
  target_link_libraries(exreader-benchmark PRIVATE exreader)
  if(WIN32)
    target_link_libraries(exreader-benchmark PRIVATE psapi shell32)
  elseif(APPLE)
    set_target_properties(exreader-benchmark PROPERTIES INSTALL_RPATH "@loader_path/../lib")
  else()
    set_target_properties(exreader-benchmark PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  install(TARGETS exreader-benchmark RUNTIME DESTINATION bin)
  install(FILES tools/run_benchmarks.py DESTINATION share/exreader/benchmarks)
endif()
if(EXR_BUILD_TESTS)
  enable_testing()
  add_executable(exreader-test tests/test_core.c)
  target_link_libraries(exreader-test PRIVATE exreader)
  add_test(NAME native-core COMMAND exreader-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures ${CMAKE_CURRENT_BINARY_DIR})
  add_executable(exreader-flow-test tests/test_flow.c)
  target_link_libraries(exreader-flow-test PRIVATE exreader)
  add_test(NAME native-flow COMMAND exreader-flow-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures ${CMAKE_CURRENT_BINARY_DIR})
  add_executable(exreader-construct-test tests/test_construct.c)
  target_link_libraries(exreader-construct-test PRIVATE exreader)
  add_test(NAME native-construct COMMAND exreader-construct-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  add_executable(exreader-anchor-test tests/test_anchor.c)
  target_link_libraries(exreader-anchor-test PRIVATE exreader)
  add_test(NAME native-anchor COMMAND exreader-anchor-test)
  add_executable(exreader-anchor-library-test tests/test_anchor_library.c)
  target_link_libraries(exreader-anchor-library-test PRIVATE exreader exr_sqlite)
  add_test(NAME native-anchor-library COMMAND exreader-anchor-library-test ${CMAKE_CURRENT_BINARY_DIR})
  add_executable(exreader-lazy-test tests/test_lazy.c)
  target_link_libraries(exreader-lazy-test PRIVATE exreader miniz)
  add_test(NAME native-lazy COMMAND exreader-lazy-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  add_executable(exreader-text-test tests/test_text.c)
  target_link_libraries(exreader-text-test PRIVATE exreader)
  add_test(NAME native-text COMMAND exreader-text-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/unicode)
  add_executable(exreader-font-test tests/test_font.c)
  target_link_libraries(exreader-font-test PRIVATE exreader)
  add_test(NAME native-font COMMAND exreader-font-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fonts)
  add_executable(exreader-budget-test tests/test_budget.c)
  target_link_libraries(exreader-budget-test PRIVATE exreader)
  add_test(NAME native-budget COMMAND exreader-budget-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fonts)
  add_executable(exreader-pager-test tests/test_pager.c)
  target_link_libraries(exreader-pager-test PRIVATE exreader)
  add_test(NAME native-pager COMMAND exreader-pager-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fonts)
  add_executable(exreader-page-cache-test tests/test_page_cache.c)
  target_link_libraries(exreader-page-cache-test PRIVATE exreader)
  add_test(NAME native-page-cache COMMAND exreader-page-cache-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fonts)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    if(EXR_BUILD_TOOLS)
      add_test(NAME native-benchmark COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_benchmark.py $<TARGET_FILE:exreader-benchmark> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fonts/primary.ttf)
    endif()
    add_test(NAME native-png COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_png.py ${CMAKE_CURRENT_BINARY_DIR}/native-page.png)
    set_tests_properties(native-png PROPERTIES DEPENDS native-core)
    add_test(NAME native-color-png COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_png.py ${CMAKE_CURRENT_BINARY_DIR}/native-styled-page.png ${CMAKE_CURRENT_BINARY_DIR}/native-image-page.png ${CMAKE_CURRENT_BINARY_DIR}/native-image-page.bmp)
    set_tests_properties(native-color-png PROPERTIES DEPENDS native-flow)
  endif()
endif()
install(TARGETS exreader RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/exreader.h DESTINATION include)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/exreader)

install(FILES ${freetype_SOURCE_DIR}/docs/FTL.TXT ${freetype_SOURCE_DIR}/LICENSE.TXT DESTINATION share/exreader/licenses/freetype)
install(FILES ${expat_SOURCE_DIR}/expat/COPYING DESTINATION share/exreader/licenses/expat)
install(FILES ${miniz_SOURCE_DIR}/LICENSE DESTINATION share/exreader/licenses/miniz)
install(FILES ${stb_SOURCE_DIR}/LICENSE DESTINATION share/exreader/licenses/stb)
install(FILES tests/fixtures/unicode/LICENSE.txt DESTINATION share/exreader/licenses/unicode)

install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES docs/NATIVE_API.md docs/NATIVE_VALIDATION.md docs/PERFORMANCE.md docs/native-page.png docs/styled-page.png DESTINATION docs)
install(DIRECTORY examples DESTINATION .)
install(DIRECTORY tests/fixtures/fonts DESTINATION examples)
