cmake_minimum_required(VERSION 3.24)
project(EXReader VERSION 0.26.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)

  add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/deps.cmake)

if(ANDROID)
  target_link_options(exreader PRIVATE -Wl,-z,max-page-size=16384)
endif()

target_link_libraries(exreader PRIVATE freetype expat miniz exr_sqlite)
if(MINGW)
  target_link_options(exreader PRIVATE -static-libgcc -static)
endif()
if(MSVC)
  target_compile_options(exreader PRIVATE /W4)
else()
  target_compile_options(exreader PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(exreader PRIVATE m)
endif()

    target_link_libraries(exreader-render PRIVATE shell32)
  endif()
