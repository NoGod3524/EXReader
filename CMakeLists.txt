if(EXR_ENABLE_ICC)
  set(default_cmyk_profile "${CMAKE_CURRENT_SOURCE_DIR}/src/profiles/default_cmyk.icc.zlib")
  file(SHA256 "${default_cmyk_profile}" default_cmyk_hash)
  if(NOT default_cmyk_hash STREQUAL "20f63c910decd30bf87d7ce2461f133d0305f552279ccdc388cdcfc8762753b2")
    message(FATAL_ERROR "Default CMYK profile checksum mismatch")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${default_cmyk_profile}")
  file(READ "${default_cmyk_profile}" default_cmyk_hex HEX)
  string(REGEX REPLACE "(..)" "0x\\1," default_cmyk_bytes "${default_cmyk_hex}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/default_cmyk_profile.h" CONTENT
    "/* Public-domain Argyll CMYK profile, zlib compressed. */\nstatic const unsigned char default_cmyk_data[] = {${default_cmyk_bytes}};\n")

endif()
