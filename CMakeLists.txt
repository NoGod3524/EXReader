cmake_minimum_required(VERSION 3.24)
project(EXReader VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
option(EXR_BUILD_TOOLS "Build the standalone native renderer" ON)
option(EXR_BUILD_TESTS "Build native integration tests" ON)
option(EXR_SANITIZE "Address and undefined behavior sanitizers" OFF)
if(EXR_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/deps.cmake)
target_compile_definitions(miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(exreader SHARED src/core.c src/document.c src/formats.c src/layout.c src/library.c src/transfer.c)
target_compile_definitions(exreader PRIVATE EXR_BUILD)
if(ANDROID)
  target_link_options(exreader PRIVATE -Wl,-z,max-page-size=16384)
endif()
target_include_directories(exreader PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(exreader PRIVATE freetype expat miniz exr_sqlite)
if(MSVC)
  target_compile_options(exreader PRIVATE /W4)
else()
  target_compile_options(exreader PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(exreader PRIVATE m)
endif()
if(EXR_BUILD_TOOLS)
  add_executable(exreader-render tools/render.c)
  target_link_libraries(exreader-render PRIVATE exreader)
  if(APPLE)
    set_target_properties(exreader-render PROPERTIES INSTALL_RPATH "@loader_path/../lib")
  elseif(UNIX)
    set_target_properties(exreader-render PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  if(WIN32)
    target_link_libraries(exreader-render PRIVATE shell32)
  endif()
  install(TARGETS exreader-render RUNTIME DESTINATION bin)
endif()
if(EXR_BUILD_TESTS)
  enable_testing()
  add_executable(exreader-test tests/test_core.c)
  target_link_libraries(exreader-test PRIVATE exreader)
  add_test(NAME native-core COMMAND exreader-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures ${CMAKE_CURRENT_BINARY_DIR})
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME native-png COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_png.py ${CMAKE_CURRENT_BINARY_DIR}/native-page.png)
    set_tests_properties(native-png PROPERTIES DEPENDS native-core)
  endif()
endif()
install(TARGETS exreader RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/exreader.h DESTINATION include)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/exreader)

install(FILES ${freetype_SOURCE_DIR}/docs/FTL.TXT ${freetype_SOURCE_DIR}/LICENSE.TXT DESTINATION share/exreader/licenses/freetype)
install(FILES ${expat_SOURCE_DIR}/expat/COPYING DESTINATION share/exreader/licenses/expat)
install(FILES ${miniz_SOURCE_DIR}/LICENSE DESTINATION share/exreader/licenses/miniz)

install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES docs/NATIVE_API.md docs/NATIVE_VALIDATION.md docs/native-page.png DESTINATION docs)
