cmake_minimum_required(VERSION 3.24)
project(ReaderReadingV2Consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderReadingState 0.1.0 EXACT CONFIG REQUIRED)
add_executable(reading-v2-probe probe.cpp api.c)
target_link_libraries(reading-v2-probe PRIVATE EXReaderReadingState::reader_reading_state
  EXReaderProductCatalog::reader_catalog EXReaderProductCatalog::reader_catalog_sqlite)
option(READING_V2_WEBDAV "Exercise real native WebDAV with installed modules" OFF)
if(READING_V2_WEBDAV)
  find_package(EXReaderSync 0.1.0 EXACT CONFIG REQUIRED)
  find_package(EXReaderSideload CONFIG REQUIRED)
  target_compile_definitions(reading-v2-probe PRIVATE READER_V2_WEBDAV_TEST=1)
  target_link_libraries(reading-v2-probe PRIVATE EXReaderSync::reader_sync EXReaderSideload::exr_sideload)
endif()
if(MSVC)
  target_link_options(reading-v2-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/reading-v2-link.map")
else()
  target_link_options(reading-v2-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/reading-v2-link.map" "-Wl,--trace-symbol=sqlite3_open_v2")
endif()
