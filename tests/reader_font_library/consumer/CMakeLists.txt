cmake_minimum_required(VERSION 3.24)
project(InstalledFontLibraryConsumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /WX)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  add_compile_options(-Wall -Wextra -Werror)
endif()
find_package(EXReaderFontLibrary 0.1.0 EXACT CONFIG REQUIRED)
add_executable(font-c api.c)
set_target_properties(font-c PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(font-c PRIVATE EXReaderFontLibrary::reader_font_library)
add_executable(font-probe probe.cpp)
target_link_libraries(font-probe PRIVATE EXReaderFontLibrary::reader_font_library)
if(MSVC)
  target_link_options(font-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/font-probe.map")
elseif(APPLE)
  target_link_options(font-probe PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/font-probe.map")
else()
  target_link_options(font-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/font-probe.map")
endif()
