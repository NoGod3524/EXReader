cmake_minimum_required(VERSION 3.24)
project(EXReaderSideloadInstalledConsumer LANGUAGES C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
if(EXR_SDK)
  list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
endif()
if(EXR_SIDELOAD_PREFIX)
  list(PREPEND CMAKE_PREFIX_PATH "${EXR_SIDELOAD_PREFIX}")
endif()
find_package(EXReaderSideload CONFIG REQUIRED)
if(NOT EXR_SIDELOAD_TEST_FIXTURES)
  message(FATAL_ERROR "EXR_SIDELOAD_TEST_FIXTURES is required")
endif()
add_executable(sideload-installed-check check_sideload.c)
target_link_libraries(sideload-installed-check PRIVATE EXReaderSideload::exr_sideload)
set_target_properties(sideload-installed-check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
enable_testing()
add_test(NAME sideload-installed-public-consumer COMMAND sideload-installed-check
  "${EXR_SIDELOAD_TEST_FIXTURES}" "${CMAKE_CURRENT_BINARY_DIR}")
