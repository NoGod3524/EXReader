cmake_minimum_required(VERSION 3.24)
project(ReaderConflictActionsInstalledConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderConflictActions 0.1.0 EXACT CONFIG REQUIRED)
add_executable(conflict-consumer main.cpp)
target_link_libraries(conflict-consumer PRIVATE EXReaderConflictActions::reader_conflict_actions EXReaderProductCatalog::reader_catalog_sqlite)
if(MSVC)
  target_link_options(conflict-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/conflict-link.map")
  target_link_libraries(conflict-consumer PRIVATE shell32)
elseif(APPLE)
  target_link_options(conflict-consumer PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/conflict-link.map")
else()
  target_link_options(conflict-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/conflict-link.map" "-Wl,--trace-symbol=sqlite3_open_v2")
endif()
add_executable(conflict-c-consumer api.c)
set_target_properties(conflict-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(conflict-c-consumer PRIVATE EXReaderConflictActions::reader_conflict_actions)
