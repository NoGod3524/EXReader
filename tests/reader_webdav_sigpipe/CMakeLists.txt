cmake_minimum_required(VERSION 3.24)
project(WebDAVSigpipeInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderWebDAV 0.2.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(webdav-sigpipe probe.cpp)
target_link_libraries(webdav-sigpipe PRIVATE EXReaderWebDAV::exr_webdav Threads::Threads)
if(MSVC)
  target_compile_options(webdav-sigpipe PRIVATE /W4 /WX)
else()
  target_compile_options(webdav-sigpipe PRIVATE -Wall -Wextra -Werror)
endif()
