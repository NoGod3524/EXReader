cmake_minimum_required(VERSION 3.24)
project(EXReaderL07Acceptance LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "L07 validates the Windows host")
endif()
get_filename_component(L07_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT L07_HOST)
  message(FATAL_ERROR "Run tests/docedit_a11y/run.py to prepare the host overlay")
endif()
set(EXR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXR_BUILD_JNI OFF CACHE BOOL "" FORCE)
set(EXR_BUILD_FUZZ OFF CACHE BOOL "" FORCE)
set(EXR_BUILD_SHARED_COMPAT OFF CACHE BOOL "" FORCE)
foreach(feature PDF PDF_CRYPTO PDF_JPX PDF_EDIT AUDIO MP3 AAC LCP ICC IMAGE_WEBP IMAGE_TIFF IMAGE_AVIF LIBRARY TEXT_INDEX)
  set(EXR_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(EXR_BUILD_TEXT_INDEX OFF CACHE BOOL "" FORCE)
add_subdirectory("${L07_ROOT}" sdk)
set(host_sources "${L07_HOST}/examples/windows/shell_d2d.cpp"
  "${L07_HOST}/examples/windows/shell_epub.cpp" "${L07_HOST}/examples/windows/shell_speech.cpp")
add_executable(check_shell_docedit_a11y "${L07_HOST}/examples/windows/check_shell_docedit_a11y.c" ${host_sources})
add_executable(EXReaderL07 WIN32 "${L07_HOST}/examples/reader.c" ${host_sources})
foreach(target check_shell_docedit_a11y EXReaderL07)
  target_compile_definitions(${target} PRIVATE EXR_READER_WINDOWS_SHELL=1)
  target_compile_features(${target} PRIVATE c_std_11 cxx_std_17)
  target_link_libraries(${target} PRIVATE exreader miniz expat comdlg32 gdi32 uiautomationcore
    shell32 user32 winhttp crypt32 d2d1 dwrite advapi32 ole32 oleaut32 uuid)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
    target_link_options(${target} PRIVATE -static)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
enable_testing()
add_test(NAME windows-docedit-a11y-headless COMMAND check_shell_docedit_a11y
  "${L07_ROOT}/tests/fixtures/rich.docx" "${CMAKE_CURRENT_BINARY_DIR}/a11y-edited.docx")
set_tests_properties(windows-docedit-a11y-headless PROPERTIES LABELS headless TIMEOUT 90)
