cmake_minimum_required(VERSION 3.24)
project(EXReaderWorkerReview LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
if(NOT EXR_REVIEW_SOURCE)
  set(EXR_REVIEW_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../examples/reader.c")
endif()
add_executable(EXReader WIN32 "${EXR_REVIEW_SOURCE}")
target_link_libraries(EXReader PRIVATE EXReader::exreader comdlg32 gdi32 shell32 user32)
if(NOT MSVC)
  target_link_libraries(EXReader PRIVATE m)
endif()
target_compile_definitions(EXReader PRIVATE EXR_HOST_WORKER_REVIEW=1
  EXR_HOST_WORKER_REVIEW_HEADER="${CMAKE_CURRENT_LIST_DIR}/hooks.h")
find_file(EXR_RUNTIME NAMES exreader.dll libexreader.dll PATHS "${EXR_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
add_custom_command(TARGET EXReader POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EXR_RUNTIME}" "$<TARGET_FILE_DIR:EXReader>" VERBATIM)
