cmake_minimum_required(VERSION 3.24)
project(RawStoreCompanionInstalledConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderCommands 0.2.0 EXACT CONFIG REQUIRED)
find_package(EXReaderSync 0.2.0 EXACT CONFIG REQUIRED)
find_package(EXReaderSession 0.2.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(raw-bootstrap bootstrap.cpp)
add_executable(raw-sync sync.cpp)
add_executable(raw-revision revision.cpp)
add_executable(raw-session session.cpp api.c)
add_executable(raw-commands commands.cpp)
add_executable(raw-commands-c commands_api.c)
add_executable(raw-guard guard.cpp)
set_target_properties(raw-commands-c PROPERTIES LINKER_LANGUAGE CXX)
foreach(target raw-bootstrap raw-sync raw-revision raw-session raw-commands raw-commands-c raw-guard)
  target_link_libraries(${target} PRIVATE EXReaderCommands::reader_commands EXReaderSync::reader_sync
    EXReaderSession::reader_session EXReaderProductCatalog::reader_catalog_sqlite Threads::Threads)
  if(WIN32)
    target_link_libraries(${target} PRIVATE shell32)
  endif()
  if(MSVC)
    target_link_options(${target} PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
  else()
    target_link_options(${target} PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
    if(target STREQUAL "raw-commands")
      target_link_options(${target} PRIVATE "-Wl,--trace-symbol=sqlite3_open_v2")
    endif()
  endif()
endforeach()
