cmake_minimum_required(VERSION 3.24)
project(ReaderSyncInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderSync CONFIG REQUIRED)
add_executable(sync-consumer main.cpp)
target_link_libraries(sync-consumer PRIVATE EXReaderSync::reader_sync)
if(WIN32)
  target_link_libraries(sync-consumer PRIVATE shell32)
endif()
