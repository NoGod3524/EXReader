cmake_minimum_required(VERSION 3.24)
project(ReaderMigrationProbe LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(Threads REQUIRED)
find_package(EXReaderReadingCommands 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderCommands 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderNavigation 0.1.0 EXACT CONFIG REQUIRED)
add_executable(migration-probe probe.cpp)
target_link_libraries(migration-probe PRIVATE EXReaderReadingCommands::reader_reading_commands EXReaderCommands::reader_commands EXReaderReadingState::reader_reading_state EXReaderNavigation::reader_navigation EXReaderProductCatalog::reader_catalog_sqlite Threads::Threads)
if(MSVC)
  target_link_options(migration-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/migration-link.map")
  target_link_libraries(migration-probe PRIVATE shell32)
else()
  target_link_options(migration-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/migration-link.map")
endif()
add_executable(migration-c-consumer api.c)
set_target_properties(migration-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(migration-c-consumer PRIVATE EXReaderReadingCommands::reader_reading_commands EXReaderReadingState::reader_reading_state EXReaderNavigation::reader_navigation EXReaderProductCatalog::reader_catalog_sqlite)
install(TARGETS migration-probe migration-c-consumer RUNTIME DESTINATION bin)
