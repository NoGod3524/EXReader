cmake_minimum_required(VERSION 3.16)
project(CoordinatorCASInstalledConsumer LANGUAGES C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /WX)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Werror)
endif()
add_library(installed-hardening STATIC IMPORTED)
if(MSVC)
  set(_base "${WH_PREFIX}/lib/exreader-migration-coordinator-hardening.lib")
else()
  set(_base "${WH_PREFIX}/lib/libexreader-migration-coordinator-hardening.a")
endif()
set_target_properties(installed-hardening PROPERTIES IMPORTED_LOCATION "${_base}"
  INTERFACE_INCLUDE_DIRECTORIES "${WH_PREFIX}/include/exreader-migration-coordinator-hardening")

add_library(installed-cas STATIC IMPORTED)
if(MSVC)
  set(_cas "${WH_PREFIX}/lib/exreader-coordinator-cas.lib")
else()
  set(_cas "${WH_PREFIX}/lib/libexreader-coordinator-cas.a")
endif()
set_target_properties(installed-cas PROPERTIES IMPORTED_LOCATION "${_cas}"
  INTERFACE_INCLUDE_DIRECTORIES "${WH_PREFIX}/include/exreader-coordinator-cas")
add_executable(coordinator-cas-check wh3_test.c)
target_compile_definitions(coordinator-cas-check PRIVATE "CANDIDATE_SOURCE_SHA=\"${SOURCE_SHA}\"")
target_link_libraries(coordinator-cas-check PRIVATE installed-cas installed-hardening)
set_target_properties(coordinator-cas-check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
