cmake_minimum_required(VERSION 3.24)
project(WindowsPublishInstalledConsumer LANGUAGES C)
if(NOT WIN32)
 message(FATAL_ERROR "Native Windows required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(EXReader 3.42.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_executable(windows-publish-check probe.c)
target_include_directories(windows-publish-check PRIVATE "${PUBLISH_PREFIX}/include/exreader-migration-candidate" "${RECORD_PREFIX}/include/exreader-migration-candidate")
target_compile_definitions(windows-publish-check PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00 "SOURCE_SHA=\"${SOURCE_SHA}\"")
target_compile_options(windows-publish-check PRIVATE /W4 /WX /utf-8)
target_link_libraries(windows-publish-check PRIVATE "${PUBLISH_PREFIX}/lib/exreader-migration-windows-publish.lib" "${RECORD_PREFIX}/lib/exreader-migration-windows-record.lib" EXReader::exr_library advapi32 psapi)
target_link_options(windows-publish-check PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/publish-consumer.map")
