cmake_minimum_required(VERSION 3.16)
project(CoordinatorInstalledConsumer LANGUAGES C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /WX)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Werror)
endif()
# Consumes only the installed archive and headers (no source-tree include path).
add_library(installed-coordinator STATIC IMPORTED)
if(MSVC)
  set(_lib "${COORD_PREFIX}/lib/exreader-migration-coordinator.lib")
else()
  set(_lib "${COORD_PREFIX}/lib/libexreader-migration-coordinator.a")
endif()
set_target_properties(installed-coordinator PROPERTIES IMPORTED_LOCATION "${_lib}"
  INTERFACE_INCLUDE_DIRECTORIES "${COORD_PREFIX}/include/exreader-migration-coordinator")
add_executable(coordinator-check wc_test.c)
target_compile_definitions(coordinator-check PRIVATE "CANDIDATE_SOURCE_SHA=\"${SOURCE_SHA}\"")
target_link_libraries(coordinator-check PRIVATE installed-coordinator)
set_target_properties(coordinator-check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
