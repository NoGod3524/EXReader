cmake_minimum_required(VERSION 3.24)
project(EXReaderRollbackObservation LANGUAGES C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
if(EXR_SDK)
  list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
endif()
find_package(EXReader CONFIG REQUIRED COMPONENTS core library)
# Header only. All SQLite implementation symbols come from the installed SDK;
# this consumer does not build a second SQLite or replace its VFS/return values.
if(NOT EXISTS "${SQLITE_HEADER_DIR}/sqlite3.h")
  message(FATAL_ERROR "Pass the matching pinned migration SQLite header directory")
endif()
add_executable(migration-rollback-observe observe.c)
target_include_directories(migration-rollback-observe PRIVATE "${SQLITE_HEADER_DIR}")
target_link_libraries(migration-rollback-observe PRIVATE EXReader::exr_library)
set_target_properties(migration-rollback-observe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(NOT WIN32)
  target_link_libraries(migration-rollback-observe PRIVATE m)
endif()
