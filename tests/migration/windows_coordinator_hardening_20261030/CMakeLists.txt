cmake_minimum_required(VERSION 3.16)
project(CoordinatorHardeningInstalledConsumer LANGUAGES C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /WX)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Werror)
endif()
# Consumes only the installed archive and headers (no source-tree include path).
add_library(installed-hardening STATIC IMPORTED)
if(MSVC)
  set(_lib "${WH_PREFIX}/lib/exreader-migration-coordinator-hardening.lib")
else()
  set(_lib "${WH_PREFIX}/lib/libexreader-migration-coordinator-hardening.a")
endif()
set_target_properties(installed-hardening PROPERTIES IMPORTED_LOCATION "${_lib}"
  INTERFACE_INCLUDE_DIRECTORIES "${WH_PREFIX}/include/exreader-migration-coordinator-hardening")
add_executable(coordinator-hardening-check wh_test.c)
target_compile_definitions(coordinator-hardening-check PRIVATE "CANDIDATE_SOURCE_SHA=\"${SOURCE_SHA}\"")
target_link_libraries(coordinator-hardening-check PRIVATE installed-hardening)
set_target_properties(coordinator-hardening-check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
