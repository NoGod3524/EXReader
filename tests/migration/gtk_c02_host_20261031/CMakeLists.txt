cmake_minimum_required(VERSION 3.24)
project(C02ActualGTKHost LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Actual GTK custody checks require Linux")
endif()
set(EXR_READER_BUILD_GTK ON CACHE BOOL "" FORCE)
set(EXR_READER_GTK_SEARCH ON CACHE BOOL "" FORCE)
set(EXR_READER_GTK_NAVIGATION ON CACHE BOOL "" FORCE)
set(EXR_SDK_VERSION "3.42.0" CACHE STRING "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../apps/reader/desktop" actual-host)
add_executable(c02-gtk-host-check host.c)
target_compile_options(c02-gtk-host-check PRIVATE -Wall -Wextra -Werror)
target_link_libraries(c02-gtk-host-check PRIVATE exreader-linux-reading)
