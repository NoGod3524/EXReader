cmake_minimum_required(VERSION 3.24)
project(PosixC02RepairHelpersCheck LANGUAGES C)
if(WIN32)
 message(FATAL_ERROR "Native POSIX helper fixtures are required")
endif()
if(NOT IS_ABSOLUTE "${WH_PREFIX}" OR NOT EXISTS "${WH_PREFIX}/lib/libexreader-migration-coordinator-hardening.a")
 message(FATAL_ERROR "WH_PREFIX must be the installed frozen ABI2 coordinator")
endif()
if(NOT SOURCE_SHA MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "SOURCE_SHA must identify this Actions source")
endif()
string(LENGTH "${SOURCE_SHA}" source_length)
if(NOT source_length EQUAL 40)
 message(FATAL_ERROR "SOURCE_SHA must have 40 lowercase hex characters")
endif()
set(native "${CMAKE_CURRENT_SOURCE_DIR}/../../../extras/migration/posix_c02_admission_20261031")
add_library(repair-helpers-under-test STATIC
 "${native}/src/pc02_repair_scan.c" "${native}/src/pc02_repair_store.c")
target_compile_features(repair-helpers-under-test PUBLIC c_std_11)
target_compile_options(repair-helpers-under-test PRIVATE -Wall -Wextra -Werror)
target_include_directories(repair-helpers-under-test PUBLIC
 "${native}/include" "${WH_PREFIX}/include/exreader-migration-coordinator-hardening")
target_link_libraries(repair-helpers-under-test PUBLIC
 "${WH_PREFIX}/lib/libexreader-migration-coordinator-hardening.a")
add_executable(posix-c02-repair-helpers-check repair_helpers_test.c)
target_compile_features(posix-c02-repair-helpers-check PRIVATE c_std_11)
target_compile_options(posix-c02-repair-helpers-check PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(posix-c02-repair-helpers-check PRIVATE SOURCE_SHA="${SOURCE_SHA}")
target_link_libraries(posix-c02-repair-helpers-check PRIVATE repair-helpers-under-test)
