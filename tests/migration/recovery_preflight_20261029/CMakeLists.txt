# Parse whole exported entries, never harvest a plausible target substring from
# an unsupported/malformed generator expression. Empty LINK_ONLY is emitted for
# a build-only dependency eliminated from an installed static target interface.
function(preflight_decode_link entry output valid)
  set(token "")
  set(ok TRUE)
  if("${entry}" MATCHES "^([A-Za-z0-9_.-]+::[A-Za-z0-9_.-]+)$")
    set(token "${CMAKE_MATCH_1}")
  elseif("${entry}" MATCHES "^\\$<LINK_ONLY:([A-Za-z0-9_.-]+::[A-Za-z0-9_.-]+)>$")
    set(token "${CMAKE_MATCH_1}")
  elseif("${entry}" STREQUAL "$<LINK_ONLY:>")
    set(token "")
  elseif("${entry}" MATCHES "\\$<|::")
    set(ok FALSE)
  endif()
  set(${output} "${token}" PARENT_SCOPE)
  set(${valid} "${ok}" PARENT_SCOPE)
endfunction()
# Actions-configure controls: exact genex/plain, empty/build-only, ordinary system
# library, malformed/trailing and ambiguous multi-target expressions.
foreach(entry "$<LINK_ONLY:EXReader::vendor_exr_sqlite>" "EXReader::vendor_exr_sqlite")
  preflight_decode_link("${entry}" token ok)
  if(NOT ok OR NOT token STREQUAL "EXReader::vendor_exr_sqlite")
    message(FATAL_ERROR "Imported-target positive control failed")
  endif()
endforeach()
foreach(entry "$<LINK_ONLY:>" "m")
  preflight_decode_link("${entry}" token ok)
  if(NOT ok OR NOT token STREQUAL "")
    message(FATAL_ERROR "Imported-target empty/system control failed")
  endif()
endforeach()
foreach(entry "$<LINK_ONLY:EXReader::one" "$<LINK_ONLY:EXReader::one>junk"

  if(ok OR NOT token STREQUAL "")
    message(FATAL_ERROR "Imported-target malformed/ambiguous refusal failed")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/link-closure.txt" "Installed target dependency inventory\n")
function(record_link_target name)
  get_property(seen GLOBAL PROPERTY PREFLIGHT_SEEN)
  if("${name}" IN_LIST seen)
    return()
  endif()
  set_property(GLOBAL APPEND PROPERTY PREFLIGHT_SEEN "${name}")
  foreach(property TYPE IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE INTERFACE_LINK_LIBRARIES)
    get_target_property(value "${name}" "${property}")
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/link-closure.txt" "${name} ${property}=${value}\n")
  endforeach()
  get_target_property(deps "${name}" INTERFACE_LINK_LIBRARIES)
  foreach(entry IN LISTS deps)
    preflight_decode_link("${entry}" dependency ok)
    if(NOT ok)
      message(FATAL_ERROR "Uninspectable imported link entry: ${entry}")
    endif()
    if(dependency)
      if(NOT TARGET "${dependency}")
        message(FATAL_ERROR "Unresolved imported dependency: ${dependency}")
      endif()
      record_link_target("${dependency}")
    endif()
  endforeach()
endfunction()
