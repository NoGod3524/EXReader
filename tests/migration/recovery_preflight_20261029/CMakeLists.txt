function(record_link_target name)
  get_property(seen GLOBAL PROPERTY PREFLIGHT_SEEN)
  if("${name}" IN_LIST seen)
    return()
  endif()
  set_property(GLOBAL APPEND PROPERTY PREFLIGHT_SEEN "${name}")
  foreach(property TYPE IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE INTERFACE_LINK_LIBRARIES)
    get_target_property(value "${name}" "${property}")
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/link-closure.txt" "${name} ${property}=${value}\n")
  endforeach()
  get_target_property(deps "${name}" INTERFACE_LINK_LIBRARIES)
  string(REGEX MATCHALL "[A-Za-z0-9_:.-]+" tokens "${deps}")
  foreach(dependency IN LISTS tokens)
    if(TARGET "${dependency}")
      record_link_target("${dependency}")
    endif()
  endforeach()
endfunction()
