cmake_minimum_required(VERSION 3.24)
project(EXRCrossOsRelayConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
# Installed package only: nothing of the component source is compiled here.
find_package(EXRCrossOsRelay 0.1.0 EXACT CONFIG REQUIRED)
add_executable(relay-tool tool.cpp)
target_link_libraries(relay-tool PRIVATE EXRCrossOsRelay::exr_relay)
add_executable(relay-consumer consumer.cpp)
target_link_libraries(relay-consumer PRIVATE EXRCrossOsRelay::exr_relay)
add_executable(relay-c-consumer api.c)
set_target_properties(relay-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(relay-c-consumer PRIVATE EXRCrossOsRelay::exr_relay)
