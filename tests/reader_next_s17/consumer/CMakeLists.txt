cmake_minimum_required(VERSION 3.24)
project(ReaderNextS17RawConsumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
# Explicit raw install family only: a legacy install is refused by the exact
# version and by the installed capability guard.
find_package(EXReaderSyncView 0.2.0 EXACT CONFIG REQUIRED)
find_package(EXReaderRevisions 0.4.0 EXACT CONFIG REQUIRED)
find_package(EXReaderProductCatalog 0.1.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(raw-view-consumer view_main.cpp raw_api.c)
set_target_properties(raw-view-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(raw-view-consumer PRIVATE EXReaderSyncView::reader_sync_view)
add_executable(raw-view-c-consumer raw_api.c c_main.c)
set_target_properties(raw-view-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(raw-view-c-consumer PRIVATE EXReaderSyncView::reader_sync_view)
add_executable(raw-revision raw_revision.cpp)
target_link_libraries(raw-revision PRIVATE EXReaderRevisions::reader_revisions
  EXReaderProductCatalog::reader_catalog_sqlite Threads::Threads)
foreach(target raw-view-consumer raw-view-c-consumer raw-revision)
  if(WIN32)
    target_link_libraries(${target} PRIVATE shell32)
  endif()
  if(MSVC)
    target_link_options(${target} PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
  endif()
endforeach()
