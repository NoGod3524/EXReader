cmake_minimum_required(VERSION 3.24)
project(cloud_readback LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8)
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderSideload CONFIG REQUIRED)
find_package(EXReaderWebDAV CONFIG REQUIRED)
find_package(EXReaderCloudImport CONFIG REQUIRED)
add_executable(cloud-probe ../reader_webdav_import/consumer/cloud_probe.c)
target_link_libraries(cloud-probe PRIVATE EXReaderCloudImport::reader_cloud_import shell32)
