cmake_minimum_required(VERSION 3.24)
project(EXReaderReleaseIdentity C)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
if(EXR_RELEASE_LIBRARY)
  find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS library)
else()
  find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core)
endif()
add_executable(release-identity identity.c)
target_link_libraries(release-identity PRIVATE EXReader::exr_core)
if(EXR_RELEASE_LIBRARY)
  target_compile_definitions(release-identity PRIVATE EXR_RELEASE_LIBRARY=1)
  target_link_libraries(release-identity PRIVATE EXReader::exr_library)
endif()
set_target_properties(release-identity PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
