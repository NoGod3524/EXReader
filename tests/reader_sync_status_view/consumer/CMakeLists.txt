cmake_minimum_required(VERSION 3.24)
project(ReaderSyncViewInstalledConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderSyncView 0.1.0 EXACT CONFIG REQUIRED)
add_executable(status-view-consumer main.cpp api.c)
target_link_libraries(status-view-consumer PRIVATE EXReaderSyncView::reader_sync_view)
if(WIN32)
  target_link_libraries(status-view-consumer PRIVATE shell32)
endif()
if(MSVC)
  target_link_options(status-view-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/status-view-link.map")
endif()
add_executable(status-view-c-consumer api.c c_main.c)
set_target_properties(status-view-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(status-view-c-consumer PRIVATE EXReaderSyncView::reader_sync_view)
