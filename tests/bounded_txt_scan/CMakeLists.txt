cmake_minimum_required(VERSION 3.24)
project(bounded_txt_scan C CXX)
find_package(EXReader CONFIG REQUIRED)
add_executable(bounded_txt_scan check.c)
target_link_libraries(bounded_txt_scan PRIVATE EXReader::exr_core)
add_executable(bounded_txt_cpp cpp.cpp)
target_link_libraries(bounded_txt_cpp PRIVATE EXReader::exr_core)
enable_testing()
add_test(NAME bounded_txt_cpp COMMAND bounded_txt_cpp)
add_test(NAME bounded_txt_scan COMMAND bounded_txt_scan)
add_executable(bounded_txt_view view.c)
target_link_libraries(bounded_txt_view PRIVATE EXReader::exr_core)
add_test(NAME bounded_txt_view COMMAND bounded_txt_view)

if(WIN32)
  add_executable(bounded_txt_large_pager large_pager.c)
  target_link_libraries(bounded_txt_large_pager PRIVATE EXReader::exr_core psapi)
endif()
