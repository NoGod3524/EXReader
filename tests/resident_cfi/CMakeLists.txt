cmake_minimum_required(VERSION 3.24)
project(EXReaderResidentCFIExperiment LANGUAGES C)
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core)
add_executable(resident-cfi main.c)
target_link_libraries(resident-cfi PRIVATE EXReader::exr_core)
set_target_properties(resident-cfi PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_compile_options(resident-cfi PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(existing-cfi-text ../test_cfi_text.c)
target_link_libraries(existing-cfi-text PRIVATE EXReader::exr_core)
