cmake_minimum_required(VERSION 3.24)
project(ReaderStartupProfile LANGUAGES C)
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED COMPONENTS core library PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_subdirectory(../../apps/reader/store/native catalog)
add_library(profile_adapter STATIC ../../apps/reader/desktop/catalog_sdk.c)
target_include_directories(profile_adapter PUBLIC ../../apps/reader/desktop)
target_link_libraries(profile_adapter PUBLIC reader_catalog EXReader::exr_library)
if(WIN32)
  target_link_libraries(profile_adapter PRIVATE bcrypt normaliz)
endif()
add_executable(startup_profile profile.c)
target_link_libraries(startup_profile PRIVATE profile_adapter reader_catalog_sqlite)
add_executable(seed_shelf ../windows_host/seed_shelf.c)
target_link_libraries(seed_shelf PRIVATE EXReader::exr_library)
