cmake_minimum_required(VERSION 3.24)
project(reader_local_downloads_consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(EXR_SDK_VERSION "3.40.0" CACHE STRING "Exact installed reading SDK")
find_package(EXReader ${EXR_SDK_VERSION} EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderSideload CONFIG REQUIRED)
find_package(EXReaderProductCatalog 0.1.0 CONFIG REQUIRED)
find_package(EXReaderLocalDownloads 0.1.0 EXACT CONFIG REQUIRED)
add_executable(ld-probe ld_probe.c)
target_link_libraries(ld-probe PRIVATE EXReaderLocalDownloads::reader_local_downloads)
set_target_properties(ld-probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
add_executable(ld-cpp-consumer ld_cpp_consumer.cpp)
target_link_libraries(ld-cpp-consumer PRIVATE EXReaderLocalDownloads::reader_local_downloads)
