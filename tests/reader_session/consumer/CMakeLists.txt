cmake_minimum_required(VERSION 3.24)
project(ReaderSessionConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderSession 0.1.0 EXACT CONFIG REQUIRED)
add_executable(session-consumer main.cpp api.c)
target_link_libraries(session-consumer PRIVATE EXReaderSession::reader_session)
if(WIN32)
  target_link_libraries(session-consumer PRIVATE shell32)
endif()
