cmake_minimum_required(VERSION 3.24)
project(ReaderSessionAndroidAcceptance LANGUAGES C CXX)
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE BOTH)
list(PREPEND CMAKE_PREFIX_PATH "${EXR_DAV}")
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderSideload CONFIG REQUIRED)
add_library(session-sdk-owner SHARED sdk_owner.c)
target_compile_features(session-sdk-owner PRIVATE c_std_11)
target_link_libraries(session-sdk-owner PRIVATE EXReaderSideload::exr_sideload)
target_compile_options(session-sdk-owner PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(session-sdk-owner PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic
  -Wl,-z,max-page-size=16384 "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/sdk-owner.map")
find_package(EXReaderSession 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderDesktopMetadata 0.1.0 EXACT CONFIG REQUIRED)
add_library(session-product-owner SHARED product_owner.cpp)
target_compile_features(session-product-owner PRIVATE cxx_std_17)
target_compile_options(session-product-owner PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(session-product-owner PRIVATE EXReaderSession::reader_session
  EXReaderDesktopMetadata::reader_desktop_metadata EXReaderProductCatalog::reader_catalog
  EXReaderProductCatalog::reader_catalog_sqlite dl)
target_link_options(session-product-owner PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic
  -Wl,-z,max-page-size=16384 "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/product-owner.map")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(session-product-owner PRIVATE EXR_WEBDAV_ANDROID_TESTING=1)
endif()
