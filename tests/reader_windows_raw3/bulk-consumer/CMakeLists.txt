cmake_minimum_required(VERSION 3.24)
project(RawBulkInstalledConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
option(PROBE_RAW "Explicit raw bulk consumer" OFF)
if(PROBE_RAW)
  set(_version 0.2.0)
else()
  set(_version 0.1.0)
endif()
find_package(EXReaderBulkOrganization ${_version} EXACT CONFIG REQUIRED)
get_filename_component(TESTS "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(bulk-probe "${TESTS}/reader_bulk_organization/consumer/bulk_probe.cpp")
add_executable(bulk-c-consumer "${TESTS}/reader_bulk_organization/consumer/api.c")
set_target_properties(bulk-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
foreach(target bulk-probe bulk-c-consumer)
  target_link_libraries(${target} PRIVATE EXReaderBulkOrganization::reader_bulk
    EXReaderProductCatalog::reader_catalog EXReaderProductCatalog::reader_catalog_sqlite)
  if(PROBE_RAW)
    if(MSVC)
      target_compile_options(${target} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/raw_aliases.h")
    else()
      target_compile_options(${target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/raw_aliases.h")
    endif()
  endif()
  if(MSVC)
    target_link_libraries(${target} PRIVATE shell32)
  endif()
endforeach()
if(MSVC)
  target_link_options(bulk-probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/bulk-link.map")
else()
  target_link_options(bulk-probe PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/bulk-link.map")
endif()
