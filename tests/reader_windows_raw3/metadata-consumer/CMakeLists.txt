cmake_minimum_required(VERSION 3.24)
project(RawMetadataInstalledConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
option(RAW_METADATA "Consume the explicit raw metadata package" OFF)
if(RAW_METADATA)
  set(_version 0.2.0)
else()
  set(_version 0.1.0)
endif()
find_package(EXReaderDesktopMetadata ${_version} EXACT CONFIG REQUIRED)
get_filename_component(TESTS "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(metadata-consumer "${TESTS}/reader_metadata/consumer/main.cpp")
add_executable(metadata-c-consumer "${TESTS}/reader_metadata/consumer/api.c")
set_target_properties(metadata-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
add_executable(command-consumer "${TESTS}/reader_commands/consumer/main.cpp")
foreach(target metadata-consumer metadata-c-consumer command-consumer)
  target_link_libraries(${target} PRIVATE EXReaderDesktopMetadata::reader_desktop_metadata
    EXReaderProductCatalog::reader_catalog EXReaderProductCatalog::reader_catalog_sqlite)
  if(RAW_METADATA)
    if(MSVC)
      target_compile_options(${target} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/raw_aliases.h")
    else()
      target_compile_options(${target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/raw_aliases.h")
    endif()
  endif()
  if(MSVC)
    target_link_libraries(${target} PRIVATE shell32)
  endif()
endforeach()
if(MSVC)
  target_link_options(metadata-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/metadata-link.map")
elseif(APPLE)
  target_link_options(metadata-consumer PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/metadata-link.map")
else()
  target_link_options(metadata-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/metadata-link.map" "-Wl,--trace-symbol=sqlite3_open_v2")
endif()
add_executable(metadata-profile profile.cpp)
target_link_libraries(metadata-profile PRIVATE EXReaderDesktopMetadata::reader_desktop_metadata)
