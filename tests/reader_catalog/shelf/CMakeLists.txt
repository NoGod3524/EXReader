cmake_minimum_required(VERSION 3.24)
project(EXReaderDesktopShelfQuery LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(Threads REQUIRED)
add_subdirectory(../../../apps/reader/store/native native-catalog)
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/original" "${CMAKE_CURRENT_BINARY_DIR}/shelf"
  "${CMAKE_CURRENT_BINARY_DIR}/organization")
add_executable(reader-catalog-check ../native_catalog_check.c)
target_link_libraries(reader-catalog-check PRIVATE reader_catalog reader_catalog_sqlite Threads::Threads)
add_test(NAME original-twelve-catalog-groups COMMAND reader-catalog-check
  "${CMAKE_CURRENT_BINARY_DIR}/original" "${CMAKE_CURRENT_SOURCE_DIR}/../../../apps/reader/store/001_catalog.sql")
add_executable(reader-shelf-check ../native_shelf_check.c)
target_link_libraries(reader-shelf-check PRIVATE reader_catalog reader_catalog_sqlite)
if(MSVC)
  target_compile_options(reader-shelf-check PRIVATE /W4 /WX)
else()
  target_compile_options(reader-shelf-check PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME bidirectional-filtered-keyset COMMAND reader-shelf-check "${CMAKE_CURRENT_BINARY_DIR}/shelf")
add_executable(reader-organization-check ../native_organization_check.c)
target_link_libraries(reader-organization-check PRIVATE reader_catalog reader_catalog_sqlite Threads::Threads)
if(MSVC)
  target_compile_options(reader-organization-check PRIVATE /W4 /WX)
else()
  target_compile_options(reader-organization-check PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME bounded-organization-views COMMAND reader-organization-check "${CMAKE_CURRENT_BINARY_DIR}/organization")
