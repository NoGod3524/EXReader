cmake_minimum_required(VERSION 3.24)
project(EXReaderReadingConsumer LANGUAGES C CXX)
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core pdf)
foreach(component audio pdf_edit textindex)
  if(TARGET EXReader::exr_${component})
    message(FATAL_ERROR "Reading profile unexpectedly exports ${component}")
  endif()
endforeach()
if(EXR_EXPECT_LIBRARY)
  if(NOT TARGET EXReader::exr_library)
    message(FATAL_ERROR "Bookshelf profile is missing the library component")
  endif()
elseif(TARGET EXReader::exr_library)
  message(FATAL_ERROR "Core reading profile unexpectedly exports SQLite library")
endif()
enable_testing()
# Reuse installed C consumers and C++ header compilation, including the bounded
# chapter consumer. No source-tree engine target is linked into these programs.
add_subdirectory(../sdk_components components)
add_executable(reading-document ../../tools/check_document.c)
add_executable(reading-benchmark ../../tools/benchmark.c)
foreach(target reading-document reading-benchmark)
  target_link_libraries(${target} PRIVATE EXReader::exreader)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  if(WIN32)
    target_link_libraries(${target} PRIVATE shell32 psapi)
  elseif(UNIX)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
