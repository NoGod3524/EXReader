cmake_minimum_required(VERSION 3.24)
project(ReaderReadingCodecConsumer LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(EXReaderReadingState 0.1.0 EXACT CONFIG REQUIRED)
add_executable(reading-codec-consumer codec.c)
set_target_properties(reading-codec-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(reading-codec-consumer PRIVATE EXReaderReadingState::reader_reading_state_codec)
if(MSVC)
  target_link_options(reading-codec-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/codec.map")
else()
  target_link_options(reading-codec-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/codec.map")
endif()
