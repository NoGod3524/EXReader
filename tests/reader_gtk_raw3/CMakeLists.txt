cmake_minimum_required(VERSION 3.24)
project(GTKRaw3ActualOwner LANGUAGES C CXX)
set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL TRUE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET GLOBAL glib-2.0)
if(NOT HOST_SOURCE OR NOT EXR_SDK OR NOT MODULE_PREFIX)
  message(FATAL_ERROR "Explicit actual owner and installed prefixes required")
endif()
set(READER_METADATA_PREFIX "${MODULE_PREFIX}" CACHE PATH "" FORCE)
set(READER_ACCOUNT_PREFIX "${MODULE_PREFIX}" CACHE PATH "" FORCE)
set(READER_READING_RECORDS_PREFIX "${MODULE_PREFIX}" CACHE PATH "" FORCE)
set(EXR_READER_BUILD_GTK ON CACHE BOOL "" FORCE)
set(EXR_READER_GTK_ACCOUNT ON CACHE BOOL "" FORCE)
set(EXR_READER_GTK_READING_RECORDS ON CACHE BOOL "" FORCE)
set(EXR_READER_GTK_COMPACT ON CACHE BOOL "" FORCE)
set(EXR_READER_FIXTURES "${HOST_SOURCE}/tests/fixtures" CACHE PATH "" FORCE)
if(NOT LEGACY_PROFILE)
  set(EXR_READER_GTK_RAW3 ON CACHE BOOL "" FORCE)
  set(READER_GTK_RAW3_PREFIX "${MODULE_PREFIX}" CACHE PATH "" FORCE)
endif()
add_subdirectory("${HOST_SOURCE}/apps/reader/desktop" actual-owner)
get_target_property(owner_definitions exreader-linux-reading COMPILE_DEFINITIONS)
get_target_property(owner_links exreader-linux-reading LINK_LIBRARIES)
if(LEGACY_PROFILE)
  add_executable(gtk-raw3-owner ../reader_gtk_record_compatibility/probe.c)
else()
  add_executable(gtk-raw3-owner probe.c)
endif()
target_compile_definitions(gtk-raw3-owner PRIVATE ${owner_definitions}
  "REAL_READING_SOURCE=\"${HOST_SOURCE}/examples/linux/reading.c\"")
target_link_libraries(gtk-raw3-owner PRIVATE exreader-linux-reading ${owner_links}
  reader_catalog_sdk reader_catalog_sqlite PkgConfig::GLIB)
target_compile_options(gtk-raw3-owner PRIVATE -Wall -Wextra -Werror)
target_link_options(gtk-raw3-owner PRIVATE "-Wl,--wrap=reader_rc_apply"
  "-Wl,--wrap=reader_rc_apply_raw_v3" "-Wl,--wrap=reader_revisions_open_raw_v3"
  "-Wl,--wrap=reader_revision_entities_page_raw_v3" "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/owner.map")
