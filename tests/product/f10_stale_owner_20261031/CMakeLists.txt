cmake_minimum_required(VERSION 3.24)
project(F10StaleOwnerProbe LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(EXReaderWebDAVCleanup 0.1.0 CONFIG REQUIRED)
add_executable(f10-stale-probe probe.c)
target_link_libraries(f10-stale-probe PRIVATE EXReaderWebDAVCleanup::exr_webdav_cleanup)
target_compile_options(f10-stale-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
# Existing production hook sites only; a separate binary for deterministic
# real SIGKILL at receipt/cleanup boundaries, not the installed binary.
add_library(f10-boundary-core STATIC ../../../extras/webdav-cleanup/src/cleanup_posix.c)
target_include_directories(f10-boundary-core PUBLIC ../../../extras/webdav-cleanup/include)
target_compile_definitions(f10-boundary-core PRIVATE EXR_WC_TEST_HOOKS)
target_compile_options(f10-boundary-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(f10-stale-boundary-probe probe.c)
target_link_libraries(f10-stale-boundary-probe PRIVATE f10-boundary-core)
target_compile_options(f10-stale-boundary-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(EXReaderSync 0.2.0 EXACT CONFIG REQUIRED)
find_package(EXReaderWebDAV 0.2.0 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(LibXml2 REQUIRED)
find_package(Threads REQUIRED)
add_executable(f10-sync-probe sync_probe.cpp)
target_link_libraries(f10-sync-probe PRIVATE EXReaderSync::reader_sync EXReaderWebDAVCleanup::exr_webdav_cleanup)
# Separately labelled existing registry hooks select real SIGKILL boundaries.
add_library(f10-managed-boundary-webdav STATIC ../../../extras/webdav/src/webdav.cpp)
target_include_directories(f10-managed-boundary-webdav PUBLIC ../../../extras/webdav/include)
target_compile_definitions(f10-managed-boundary-webdav PUBLIC EXR_WEBDAV_MANAGED=1 PRIVATE EXR_WEBDAV_MANAGED_TEST_HOOKS=1)
target_link_libraries(f10-managed-boundary-webdav PRIVATE EXReaderWebDAVCleanup::exr_webdav_cleanup PkgConfig::CURL LibXml2::LibXml2 Threads::Threads)
add_executable(f10-managed-boundary-probe sync_probe.cpp)
target_compile_definitions(f10-managed-boundary-probe PRIVATE F10_BOUNDARY=1)
target_link_libraries(f10-managed-boundary-probe PRIVATE f10-managed-boundary-webdav EXReaderWebDAVCleanup::exr_webdav_cleanup)
# The original b5e7 implementation is materialized by Actions git worktree.
if(NOT EXISTS "${F10_OLD_SOURCE}/extras/webdav/src/webdav.cpp")
  message(FATAL_ERROR "Fixed old source missing")
endif()
add_library(f10-old-webdav STATIC "${F10_OLD_SOURCE}/extras/webdav/src/webdav.cpp")
target_include_directories(f10-old-webdav PUBLIC "${F10_OLD_SOURCE}/extras/webdav/include")
target_compile_definitions(f10-old-webdav PUBLIC EXR_WEBDAV_MANAGED=1)
target_link_libraries(f10-old-webdav PRIVATE EXReaderWebDAVCleanup::exr_webdav_cleanup PkgConfig::CURL LibXml2::LibXml2 Threads::Threads)
add_executable(f10-old-client sync_probe.cpp)
target_compile_definitions(f10-old-client PRIVATE F10_OLD_CLIENT=1)
target_link_libraries(f10-old-client PRIVATE f10-old-webdav)
