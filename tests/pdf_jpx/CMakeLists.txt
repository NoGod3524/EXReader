cmake_minimum_required(VERSION 3.24)
project(EXReaderJPXRegression LANGUAGES C)
include(FetchContent)
FetchContent_Declare(openjpeg URL https://codeload.github.com/uclouvain/openjpeg/tar.gz/refs/tags/v2.5.4 URL_HASH SHA256=a695fbe19c0165f295a8531b1e4e855cd94d0875d2f88ec4b61080677e27188a SOURCE_SUBDIR exreader-library-only)
FetchContent_MakeAvailable(openjpeg)
# Match the shipping dependency's portable allocator and threadless C sources.
set(OPENJPEG_VERSION_MAJOR 2)
set(OPENJPEG_VERSION_MINOR 5)
set(OPENJPEG_VERSION_BUILD 4)
set(PACKAGE_VERSION "2.5.4")
set(OPJ_BIG_ENDIAN OFF)
if(CMAKE_C_BYTE_ORDER STREQUAL "BIG_ENDIAN")
  set(OPJ_BIG_ENDIAN ON)
endif()
configure_file("${openjpeg_SOURCE_DIR}/src/lib/openjp2/opj_config.h.cmake.in" "${openjpeg_BINARY_DIR}/opj_config.h" @ONLY)
configure_file("${openjpeg_SOURCE_DIR}/src/lib/openjp2/opj_config_private.h.cmake.in" "${openjpeg_BINARY_DIR}/opj_config_private.h" @ONLY)
set(jpx_sources thread bio cio dwt event ht_dec image invert j2k jp2 mct mqc openjpeg opj_clock pi t1 t2 tcd tgt function_list opj_malloc sparse_array)
list(TRANSFORM jpx_sources PREPEND "${openjpeg_SOURCE_DIR}/src/lib/openjp2/")
list(TRANSFORM jpx_sources APPEND ".c")
add_library(exr_jpx STATIC ${jpx_sources})
target_include_directories(exr_jpx SYSTEM PUBLIC "${openjpeg_SOURCE_DIR}/src/lib/openjp2" "${openjpeg_BINARY_DIR}")
target_compile_definitions(exr_jpx PUBLIC OPJ_STATIC PRIVATE MUTEX_stub)
target_link_libraries(exr_jpx PUBLIC m)
add_executable(jpx-regress regress.c ../../src/pdf_jpx.c)
target_include_directories(jpx-regress PRIVATE ../../include ../../src)
target_compile_definitions(jpx-regress PRIVATE EXR_STATIC OPJ_STATIC EXR_JPX_STANDALONE EXR_ENABLE_PDF_JPX=1)
target_link_libraries(jpx-regress PRIVATE exr_jpx)
target_compile_options(jpx-regress PRIVATE -Wall -Wextra -Werror)
set_target_properties(jpx-regress PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
enable_testing()
add_test(NAME packed-jpx-regression COMMAND jpx-regress)
add_executable(jpx-off off.c ../../src/pdf_jpx.c)
target_include_directories(jpx-off PRIVATE ../../include ../../src)
target_compile_definitions(jpx-off PRIVATE EXR_STATIC)
target_compile_options(jpx-off PRIVATE -Wall -Wextra -Werror)
add_test(NAME disabled-jpx-regression COMMAND jpx-off)
