cmake_minimum_required(VERSION 3.24)
project(ReaderCommandInstalledConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderCommands 0.1.0 EXACT CONFIG REQUIRED)
add_executable(command-consumer main.cpp)
target_link_libraries(command-consumer PRIVATE EXReaderCommands::reader_commands EXReaderProductCatalog::reader_catalog_sqlite)
if(MSVC)
  target_link_options(command-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/command-link.map")
  target_link_libraries(command-consumer PRIVATE shell32)
elseif(APPLE)
  target_link_options(command-consumer PRIVATE "-Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/command-link.map")
else()
  target_link_options(command-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/command-link.map" "-Wl,--trace-symbol=sqlite3_open_v2")
endif()
add_executable(command-c-consumer api.c)
set_target_properties(command-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(command-c-consumer PRIVATE EXReaderCommands::reader_commands)
