cmake_minimum_required(VERSION 3.24)
project(Wave3JointInstallConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
# One CMake graph imports the unchanged SDK and all five optional packages from
# the installed default (0.1.0) or explicit raw (0.2.0) family prefix.
find_package(EXReader 3.41.0 EXACT CONFIG REQUIRED COMPONENTS library)
find_package(EXReaderAnnotationExport ${FAMILY_VERSION} EXACT CONFIG REQUIRED)
find_package(EXReaderAnnotationImport ${FAMILY_VERSION} EXACT CONFIG REQUIRED)
find_package(EXReaderFontLibrary 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderOriginalExport 0.1.0 EXACT CONFIG REQUIRED)
find_package(EXReaderJumpHistory ${FAMILY_VERSION} EXACT CONFIG REQUIRED)
find_package(EXReaderSyncView ${FAMILY_VERSION} EXACT CONFIG REQUIRED)
set(modules EXReaderAnnotationExport::reader_annotation_export
  EXReaderAnnotationImport::reader_annotation_import EXReaderFontLibrary::reader_font_library
  EXReaderOriginalExport::reader_original_export EXReaderJumpHistory::reader_jump_history
  EXReaderSyncView::reader_sync_view)
foreach(name FAMILY_PREFIX SHARED_PREFIX SDK_PREFIX)
  if(NOT IS_ABSOLUTE "${${name}}")
    message(FATAL_ERROR "The declared install prefix ${name} must be absolute")
  endif()
endforeach()
get_property(imports DIRECTORY PROPERTY IMPORTED_TARGETS)
include("${CMAKE_CURRENT_LIST_DIR}/ImportedPaths.cmake")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/origins.txt" "")
foreach(target IN LISTS imports)
  if(NOT target MATCHES "^EXReader")
    continue()
  endif()
  foreach(property IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB_RELEASE INTERFACE_INCLUDE_DIRECTORIES)
    s18_imported_paths(paths "${target}" "${property}" "")
    if(NOT paths)
      continue()
    endif()
    foreach(path IN LISTS paths)
      if(path STREQUAL "")
        continue()
      endif()
      s18_owned_path(actual "${target}" "${property}" "${path}")
      file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/origins.txt" "${target}|${property}|${actual}\n")
    endforeach()
  endforeach()
endforeach()
get_property(expressions GLOBAL PROPERTY S18_IMPORTED_PATH_EXPRESSIONS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/genexpr-origins.txt" "")
foreach(expression IN LISTS expressions)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/genexpr-origins.txt" "${expression}\n")
endforeach()
# The one SDK provider this graph links must live in the declared SDK prefix.
get_target_property(sdk_location EXReader::exr_library IMPORTED_LOCATION_RELEASE)
if(NOT sdk_location)
  get_target_property(sdk_location EXReader::exr_library IMPORTED_LOCATION)
endif()
if(NOT sdk_location)
  message(FATAL_ERROR "EXReader::exr_library has no imported location")
endif()
file(REAL_PATH "${sdk_location}" sdk_actual)
file(REAL_PATH "${SDK_PREFIX}" sdk_root)
cmake_path(IS_PREFIX sdk_root "${sdk_actual}" NORMALIZE sdk_inside)
if(NOT sdk_inside)
  message(FATAL_ERROR "The linked SDK provider is outside the declared SDK prefix: ${sdk_actual}")
endif()
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/origins.txt" "EXReader::exr_library|resident_sdk|${sdk_actual}\n")
add_executable(foundation-consumer main.cpp)
target_compile_definitions(foundation-consumer PRIVATE FAMILY_VERSION="${FAMILY_VERSION}")
target_link_libraries(foundation-consumer PRIVATE ${modules} EXReader::exr_library)
