cmake_minimum_required(VERSION 3.24)
project(GTKManagedWebDAVChecks LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(EXReaderWebDAVManaged 0.1.0 EXACT CONFIG REQUIRED)
add_executable(managed-consumer probe.cpp)
target_compile_options(managed-consumer PRIVATE -Wall -Wextra -Werror)
target_link_libraries(managed-consumer PRIVATE EXReaderWebDAV::exr_webdav)
target_link_options(managed-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/consumer.map")
if(NOT HOST_SOURCE)
  message(FATAL_ERROR "Exact candidate checkout is required for the uninstalled fault worker")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(LibXml2 REQUIRED)
find_package(Threads REQUIRED)
add_executable(managed-fault-worker probe.cpp "${HOST_SOURCE}/extras/webdav/src/webdav.cpp"
  "${HOST_SOURCE}/extras/webdav-cleanup/src/cleanup_posix.c")
target_include_directories(managed-fault-worker PRIVATE "${HOST_SOURCE}/extras/webdav/include" "${HOST_SOURCE}/extras/webdav-cleanup/include")
target_compile_definitions(managed-fault-worker PRIVATE EXR_WEBDAV_MANAGED=1 EXR_WC_TEST_HOOKS=1 EXR_WEBDAV_MANAGED_TEST_HOOKS=1 EXR_MANAGED_FAULT_WORKER=1)
target_compile_options(managed-fault-worker PRIVATE -Wall -Wextra -Werror)
target_link_libraries(managed-fault-worker PRIVATE PkgConfig::CURL LibXml2::LibXml2 Threads::Threads)
target_link_options(managed-fault-worker PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/fault.map")
