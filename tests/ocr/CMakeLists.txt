cmake_minimum_required(VERSION 3.24)
project(EXReaderOCRConsumer LANGUAGES C CXX)
if(NOT TARGET exreader_ocr)
  find_package(EXReaderOCR CONFIG REQUIRED)
  set(OCR_TARGET EXReader::exreader_ocr)
else()
  set(OCR_TARGET exreader_ocr)
endif()
add_executable(exreader-ocr-probe probe.cpp)
target_link_libraries(exreader-ocr-probe PRIVATE ${OCR_TARGET})
target_compile_features(exreader-ocr-probe PRIVATE cxx_std_17)
