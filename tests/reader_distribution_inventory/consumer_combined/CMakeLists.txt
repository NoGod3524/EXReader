cmake_minimum_required(VERSION 3.24)
project(EXReaderCombinedConsumers LANGUAGES C CXX)
# Every consumer links exactly one installed optional package and nothing else.
# The recorded link command is the real interface of that package inside the
# composed prefix; no source tree and no test helper is referenced here.
list(PREPEND CMAKE_PREFIX_PATH "${EXR_MODULE_PREFIX}")
function(exr_consumer name package target)
  find_package(${package} CONFIG REQUIRED)
  add_executable(consumer-${name} main.c)
  target_link_libraries(consumer-${name} PRIVATE ${target})
  set_target_properties(consumer-${name} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(UNIX AND NOT APPLE)
    target_link_options(consumer-${name} PRIVATE "LINKER:--no-as-needed")
  endif()
  if(MSVC)
    set_property(TARGET consumer-${name} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endfunction()
exr_consumer(catalog EXReaderProductCatalog EXReaderProductCatalog::reader_catalog)
exr_consumer(revisions EXReaderRevisions EXReaderRevisions::reader_revisions)
exr_consumer(commands EXReaderCommands EXReaderCommands::reader_commands)
exr_consumer(desktop-metadata EXReaderDesktopMetadata EXReaderDesktopMetadata::reader_desktop_metadata)
exr_consumer(webdav EXReaderWebDAV EXReaderWebDAV::exr_webdav)
exr_consumer(session EXReaderSession EXReaderSession::reader_session)
exr_consumer(sync EXReaderSync EXReaderSync::reader_sync)
exr_consumer(credentials EXReaderCredentials EXReaderCredentials::exr_credentials)
exr_consumer(reading-state EXReaderReadingState EXReaderReadingState::reader_reading_state)
exr_consumer(search EXReaderSearch EXReaderSearch::reader_search)
exr_consumer(preferences EXReaderPreferences EXReaderPreferences::reader_preferences)
exr_consumer(batch-import EXReaderBatchImport EXReaderBatchImport::reader_batch_import)
exr_consumer(sideload EXReaderSideload EXReaderSideload::exr_sideload)
