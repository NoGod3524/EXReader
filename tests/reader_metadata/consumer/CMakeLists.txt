cmake_minimum_required(VERSION 3.24)
project(ReaderMetadataInstalledConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderDesktopMetadata 0.1.0 EXACT CONFIG REQUIRED)
add_executable(metadata-consumer main.cpp)
target_link_libraries(metadata-consumer PRIVATE EXReaderDesktopMetadata::reader_desktop_metadata
  EXReaderProductCatalog::reader_catalog EXReaderProductCatalog::reader_catalog_sqlite)
if(MSVC)
  target_link_options(metadata-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/metadata-link.map")
  target_link_libraries(metadata-consumer PRIVATE shell32)
else()
  target_link_options(metadata-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/metadata-link.map" "-Wl,--trace-symbol=sqlite3_open_v2")
endif()
add_executable(metadata-c-consumer api.c)
set_target_properties(metadata-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(metadata-c-consumer PRIVATE EXReaderDesktopMetadata::reader_desktop_metadata EXReaderProductCatalog::reader_catalog)
