cmake_minimum_required(VERSION 3.24)
project(PageBackgroundInstalledConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(EXReader CONFIG REQUIRED COMPONENTS core)
add_executable(page-background probe.c)
target_link_libraries(page-background PRIVATE EXReader::exr_core)
if(BACKGROUND_BASELINE)
  target_compile_definitions(page-background PRIVATE BACKGROUND_BASELINE)
endif()
if(MSVC)
  target_compile_options(page-background PRIVATE /utf-8)
else()
  target_link_libraries(page-background PRIVATE m)
  target_compile_options(page-background PRIVATE -Wall -Wextra -Werror)
endif()
