cmake_minimum_required(VERSION 3.24)
project(EXReaderCppConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(EXReader 0.18.0 EXACT CONFIG REQUIRED PATHS "${EXR_SDK}" NO_DEFAULT_PATH)
add_executable(exreader-cpp-test main.cpp)
target_link_libraries(exreader-cpp-test PRIVATE EXReader::exreader)
if(WIN32)
  add_custom_command(TARGET exreader-cpp-test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:EXReader::exreader>" "$<TARGET_FILE_DIR:exreader-cpp-test>")
endif()
