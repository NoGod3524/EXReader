cmake_minimum_required(VERSION 3.24)
project(EXReaderComponentConsumer LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
if(EXR_SDK)
  list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
endif()
find_package(EXReader CONFIG REQUIRED COMPONENTS core)
foreach(document SUPPORT_MATRIX.md WP7-SCRIPTS.md WP8_COLR_V1.md WP8_SVG_FONTS.md WP12A-FORMATS.md WP12A-STREAMING.md CONVERSION.md WP6-BOUNDED-READER.md)
  if(NOT EXISTS "${EXReader_DIR}/../../../docs/${document}")
    message(FATAL_ERROR "Installed SDK is missing docs/${document}")
  endif()
endforeach()
add_executable(core-consumer ../test_core_component.c)
add_executable(chapter-consumer ../../examples/window.c)
target_link_libraries(chapter-consumer PRIVATE EXReader::exr_core)
set_target_properties(chapter-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_link_libraries(core-consumer PRIVATE EXReader::exr_core)
set_target_properties(core-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
get_target_property(include_dirs EXReader::exr_base INTERFACE_INCLUDE_DIRECTORIES)
foreach(include_dir IN LISTS include_dirs)
  file(GLOB headers "${include_dir}/exr/*.h")
  foreach(header IN LISTS headers)
    get_filename_component(name "${header}" NAME_WE)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${name}.cpp" "#include \"exr/${name}.h\"\n")
    add_library(header-${name} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header_${name}.cpp")
    target_link_libraries(header-${name} PRIVATE EXReader::exr_core)
    set_target_properties(header-${name} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
  endforeach()
endforeach()
if(TARGET EXReader::exr_audio)
  add_executable(audio-consumer ../test_audio.c)
  target_link_libraries(audio-consumer PRIVATE EXReader::exr_audio)
endif()
if(TARGET EXReader::exr_library)
  add_executable(library-consumer ../test_library_component.c)
  target_link_libraries(library-consumer PRIVATE EXReader::exr_library)
  if(TARGET EXReader::exr_pdf)
    target_compile_definitions(library-consumer PRIVATE EXR_TEST_PDF=1)
  endif()
endif()
if(TARGET EXReader::exr_textindex)
  add_executable(textindex-consumer ../test_textindex_component.c)
  target_link_libraries(textindex-consumer PRIVATE EXReader::exr_textindex)
endif()
if(TARGET EXReader::exreader)
  add_executable(convert-consumer ../test_convert_component.c)
  target_link_libraries(convert-consumer PRIVATE EXReader::exreader)
  get_target_property(exr_consumer_library_type EXReader::exreader TYPE)
  if(WIN32 AND exr_consumer_library_type STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET convert-consumer POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:EXReader::exreader>" "$<TARGET_FILE_DIR:convert-consumer>")
  endif()
endif()

enable_testing()
if(EXR_TEST_FONT)
  add_test(NAME installed-core COMMAND core-consumer "${EXR_TEST_FONT}" "${EXR_TEST_COLOR_FONT}")
endif()
if(EXR_TEST_FONT AND EXR_TEST_FIXTURES)
  add_test(NAME installed-chapter COMMAND chapter-consumer "${EXR_TEST_FONT}" "${EXR_TEST_FIXTURES}/table-fragments-12.epub")
endif()
if(TARGET convert-consumer AND EXR_TEST_FIXTURES)
  add_test(NAME installed-convert COMMAND convert-consumer "${EXR_TEST_FIXTURES}" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
foreach(component audio library textindex)
  if(TARGET ${component}-consumer)
    add_test(NAME installed-${component} COMMAND ${component}-consumer "${CMAKE_CURRENT_BINARY_DIR}")
  endif()
endforeach()
