cmake_minimum_required(VERSION 3.24)
project(reader_book_information_consumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
# Only the installed package and its declared public dependencies.
find_package(EXReaderBookInformation 0.1.0 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(book-info-probe probe.c)
target_link_libraries(book-info-probe PRIVATE EXReaderBookInformation::reader_book_information)
add_executable(book-info-compare compare.cpp)
target_link_libraries(book-info-compare PRIVATE EXReaderBookInformation::reader_book_information Threads::Threads)
if(WIN32)
  target_link_libraries(book-info-probe PRIVATE shell32)
endif()
if(MSVC)
  foreach(target book-info-probe book-info-compare)
    target_compile_options(${target} PRIVATE /W4 /WX)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endforeach()
else()
  foreach(target book-info-probe book-info-compare)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endforeach()
endif()
