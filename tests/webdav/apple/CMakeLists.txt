cmake_minimum_required(VERSION 3.24)
project(AppleInstalledWebDAVProbe LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
find_package(EXReaderWebDAV CONFIG REQUIRED)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(WebDAVProbe MACOSX_BUNDLE ios_main.mm probe.cpp)
  target_link_libraries(WebDAVProbe PRIVATE "-framework UIKit")
  set_target_properties(WebDAVProbe PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
else()
  add_executable(WebDAVProbe probe.cpp)
endif()
target_compile_options(WebDAVProbe PRIVATE -Wall -Wextra -Werror $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
target_link_libraries(WebDAVProbe PRIVATE EXReaderWebDAV::exr_webdav)
