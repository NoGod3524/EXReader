cmake_minimum_required(VERSION 3.24)
project(ReaderRevisionInstalledConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
find_package(EXReaderRevisions CONFIG REQUIRED)
add_executable(revision-consumer main.cpp)
target_link_libraries(revision-consumer PRIVATE EXReaderRevisions::reader_revisions
  EXReaderProductCatalog::reader_catalog_sqlite)
if(MSVC)
  target_link_options(revision-consumer PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/revision-link.map")
  target_link_libraries(revision-consumer PRIVATE shell32)
else()
  target_link_options(revision-consumer PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/revision-link.map" "-Wl,--trace-symbol=sqlite3_open_v2")
endif()
add_executable(revision-c-consumer api.c)
set_target_properties(revision-c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(revision-c-consumer PRIVATE EXReaderRevisions::reader_revisions)
