cmake_minimum_required(VERSION 3.24)
project(txt_forward_installed_sdk_regressions C CXX)
find_package(EXReader CONFIG REQUIRED)
enable_testing()
# Reuse existing consumer test sources unchanged, linked to the same installed
# SDK that built both native Windows hosts in this run.
get_filename_component(EXR_TESTS "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(bounded_txt_cpp ${EXR_TESTS}/bounded_txt_scan/cpp.cpp)
add_executable(bounded_txt_scan ${EXR_TESTS}/bounded_txt_scan/check.c)
add_executable(bounded_txt_view ${EXR_TESTS}/bounded_txt_scan/view.c)
add_executable(bounded_txt_library_boundary ${EXR_TESTS}/bounded_txt_scan/library_boundary.c)
foreach(target bounded_txt_cpp bounded_txt_scan bounded_txt_view)
  target_link_libraries(${target} PRIVATE EXReader::exr_core)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
target_link_libraries(bounded_txt_library_boundary PRIVATE EXReader::exr_library)
add_test(NAME bounded_txt_library_boundary COMMAND bounded_txt_library_boundary)
add_executable(bounded_native_shaping ${EXR_TESTS}/test_shaping.c)
add_executable(bounded_native_complex_layout ${EXR_TESTS}/test_complex_layout.c)
add_executable(bounded_native_line_boundary ${EXR_TESTS}/test_line_boundary.c)
foreach(target bounded_native_shaping bounded_native_complex_layout bounded_native_line_boundary)
  target_link_libraries(${target} PRIVATE EXReader::exr_core)
endforeach()
add_test(NAME bounded_native_shaping COMMAND bounded_native_shaping ${EXR_TESTS}/fixtures/shaping)
add_test(NAME bounded_native_complex_layout COMMAND bounded_native_complex_layout ${EXR_TESTS}/fixtures/shaping)
add_test(NAME bounded_native_line_boundary COMMAND bounded_native_line_boundary ${EXR_TESTS}/fixtures/shaping ${EXR_TESTS}/fixtures/fonts)
