cmake_minimum_required(VERSION 3.24)
project(NativeOwnerReadback LANGUAGES C CXX)
set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL TRUE)
add_subdirectory(../reader_windows_raw3/peer-owner original-peer)
get_target_property(_links windows-peer-owner LINK_LIBRARIES)
get_target_property(_includes windows-peer-owner INCLUDE_DIRECTORIES)
get_target_property(_definitions windows-peer-owner COMPILE_DEFINITIONS)
add_executable(owner-public-state public_state.cpp ../../apps/reader/desktop/windows_records_owner.cpp)
target_compile_features(owner-public-state PRIVATE cxx_std_17)
target_include_directories(owner-public-state PRIVATE ${_includes})
target_compile_definitions(owner-public-state PRIVATE ${_definitions})
target_link_libraries(owner-public-state PRIVATE ${_links})
if(MSVC)
  set_property(TARGET owner-public-state PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(owner-public-state PRIVATE /utf-8 /W4 /EHsc)
  target_link_options(owner-public-state PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/owner-public-state.map")
endif()
