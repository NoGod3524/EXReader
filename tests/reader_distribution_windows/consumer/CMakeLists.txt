cmake_minimum_required(VERSION 3.24)
project(EXReaderDistributionConsumer LANGUAGES C CXX)
# Only an extracted distribution prefix may be used: the SDK location is an
# input, and the exported package must resolve inside that same prefix.
if(NOT EXR_SDK)
  message(FATAL_ERROR "EXR_SDK must point at an extracted EXReader distribution prefix")
endif()
if(MSVC)
  # Keep the default dynamic CRT so the consumer uses the same runtime as the
  # installed SDK; a mismatched runtime would prove nothing about relocation.
  add_compile_options(/utf-8 /W4)
endif()
list(PREPEND CMAKE_PREFIX_PATH "${EXR_SDK}")
find_package(EXReader 3.40.0 EXACT CONFIG REQUIRED COMPONENTS core pdf)
get_filename_component(exr_expected_prefix "${EXR_SDK}" ABSOLUTE)
get_filename_component(exr_resolved_prefix "${EXReader_DIR}/../../.." ABSOLUTE)
if(NOT exr_resolved_prefix STREQUAL exr_expected_prefix)
  message(FATAL_ERROR "EXReader config resolved to ${exr_resolved_prefix}, not ${exr_expected_prefix}")
endif()
foreach(component audio library textindex pdf_edit)
  if(TARGET EXReader::exr_${component})
    message(FATAL_ERROR "The reading distribution must not export the ${component} component")
  endif()
endforeach()
if(NOT TARGET EXReader::exreader OR NOT TARGET EXReader::exr_pdf)
  message(FATAL_ERROR "The distribution is missing the aggregate or PDF component target")
endif()
add_executable(distribution-consumer-c consumer.c)
target_link_libraries(distribution-consumer-c PRIVATE EXReader::exreader EXReader::exr_pdf)
target_compile_definitions(distribution-consumer-c PRIVATE _CRT_SECURE_NO_WARNINGS)
set_target_properties(distribution-consumer-c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
add_executable(distribution-consumer-cpp consumer.cpp)
target_link_libraries(distribution-consumer-cpp PRIVATE EXReader::exreader)
set_target_properties(distribution-consumer-cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
if(WIN32)
  add_executable(relocation-probe ../probe/relocation_probe.c)
  target_link_libraries(relocation-probe PRIVATE EXReader::exreader EXReader::exr_pdf psapi)
  target_compile_definitions(relocation-probe PRIVATE _CRT_SECURE_NO_WARNINGS)
  set_target_properties(relocation-probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
endif()
