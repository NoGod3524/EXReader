cmake_minimum_required(VERSION 3.24)
project(ReaderPureCaptureInstalledConsumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PURE_CAPTURE_MODE "on" CACHE STRING "on, off or default installed candidate")
set(PURE_HEADER_MODE "normal" CACHE STRING "normal, zero, undefined, raw-zero or raw-undefined header consumer")
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
if(PURE_CAPTURE_MODE STREQUAL "default")
  find_package(EXReaderReadingState 0.1.0 EXACT CONFIG REQUIRED)
elseif(PURE_CAPTURE_MODE STREQUAL "on" OR PURE_CAPTURE_MODE STREQUAL "off")
  find_package(EXReaderReadingState 0.2.0 EXACT CONFIG REQUIRED)
else()
  message(FATAL_ERROR "Unknown PURE_CAPTURE_MODE")
endif()
if(NOT PURE_HEADER_MODE STREQUAL "normal")
  if(NOT PURE_HEADER_MODE MATCHES "^(zero|undefined|raw-zero|raw-undefined)$")
    message(FATAL_ERROR "Unknown PURE_HEADER_MODE")
  endif()
  add_executable(pure-header-probe header_probe.c)
  target_link_libraries(pure-header-probe PRIVATE EXReaderReadingState::reader_reading_state_codec)
  set_target_properties(pure-header-probe PROPERTIES LINKER_LANGUAGE CXX)
  if(PURE_HEADER_MODE STREQUAL "zero")
    target_compile_definitions(pure-header-probe PRIVATE PURE_HEADER_ZERO=1)
  elseif(PURE_HEADER_MODE STREQUAL "raw-zero")
    target_compile_definitions(pure-header-probe PRIVATE PURE_HEADER_RAW_ZERO=1)
  elseif(PURE_HEADER_MODE STREQUAL "raw-undefined")
    target_compile_definitions(pure-header-probe PRIVATE PURE_HEADER_RAW_UNDEFINED=1)
  endif()
else()
  add_executable(pure-capture-c api.c)
  add_executable(pure-capture-cpp probe.cpp)
  set_target_properties(pure-capture-c PROPERTIES LINKER_LANGUAGE CXX)
  foreach(target pure-capture-c pure-capture-cpp)
    target_link_libraries(${target} PRIVATE EXReaderReadingState::reader_reading_state
      EXReaderProductCatalog::reader_catalog_sqlite)
    if(PURE_CAPTURE_MODE STREQUAL "on")
      target_compile_definitions(${target} PRIVATE PURE_CAPTURE_TEST_ON=1)
    endif()
    if(NOT PURE_CAPTURE_MODE STREQUAL "default")
      target_compile_definitions(${target} PRIVATE PURE_CAPTURE_TEST_RAW=1)
    endif()
  endforeach()
endif()
